\section{Coherent measures and Fourier transport}

\begin{definition}[Coherent measure on the unit sphere]
    \label{def:replica_coherent_sphere_measure}
    \lean{TensorPower.CoherentSphere, TensorPower.coherentSphereMap,
        TensorPower.coherentDensity, TensorPower.coherentMeasure,
        TensorPower.continuous_coherentSphere_coe, TensorPower.coherentSphereMap_apply,
        TensorPower.continuous_coherentSphereMap, TensorPower.coherentMeasure_apply}
    \leanok
    Let $\Theta$ be the unit sphere of the one-copy space $\mathcal V$, let
    $e_a$ be a coordinate unit vector and let $H$ be Haar probability measure
    on $U(\mathcal V)$. For a matrix $\rho$ on $\mathcal V^{\otimes k}$, set
    $q_\rho(U)=D_k\operatorname{Re}\operatorname{Tr}(\rho P_{Ue_a,k})$. Define
    \begin{align}
        \mu_\rho & =(U\mapsto Ue_a)_*\bigl((q_\rho)_+H\bigr).
        \label{eq:replica_coherent_measure}
    \end{align}
    For $\rho\ge0$, the density is nonnegative and
    (\ref{eq:replica_coherent_measure}) is the coherent measure
    of~\cite[Proposition~7.4]{OpenAI2026AreaLaw}.
\end{definition}

\begin{theorem}[Coherent integral representation]
    \label{thm:replica_coherent_measure_integral}
    \lean{TensorPower.isFiniteMeasure_coherentMeasure,
        TensorPower.integral_coherentMeasure, TensorPower.coherentMeasure_real_univ,
        TensorPower.coherentMeasure_zero, TensorPower.coherentDensity_zero,
        TensorPower.continuous_coherentDensity_joint, TensorPower.continuous_coherentDensity,
        TensorPower.coherentDensity_nonneg, TensorPower.integrable_coherentDensity,
        TensorPower.integral_coherentDensity_mul, TensorPower.integrable_coherentMeasure}
    \leanok
    \uses{def:replica_coherent_sphere_measure}
    The measure $\mu_\rho$ is finite for every matrix $\rho$. For $\rho\ge0$
    and every continuous real function $f$ on $\mathcal V$,
    \begin{align}
        \int_\Theta f(\theta)\,d\mu_\rho(\theta)
         & =D_k\int_{U(\mathcal V)}f(Ue_a)
        \operatorname{Re}\operatorname{Tr}(\rho P_{Ue_a,k})\,dH(U).
        \label{eq:replica_coherent_integral}
    \end{align}
    If in addition $\operatorname{Tr}\rho=1$ and $\Pi_k\rho=\rho$, then
    $\mu_\rho(\Theta)=1$. Also $\mu_0=0$.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:coherent_resolution}
    Continuity bounds the density on the compact unitary group. For $\rho\ge0$,
    $\operatorname{Tr}(\rho P_{\theta,k})
        =\langle\theta^{\otimes k},\rho\theta^{\otimes k}\rangle\ge0$.
    Change of measure gives (\ref{eq:replica_coherent_integral}). Taking $f=1$
    and using $D_k\int P_{Ue_a,k}\,dH(U)=\Pi_k$ gives the mass statement.
\end{proof}

\begin{definition}[Fourier-weighted terminal measure]
    \label{def:replica_fourier_leaf_measure}
    \lean{TensorPower.ReplicaTransport.TransportData.transportLeafDensity,
        TensorPower.ReplicaTransport.TransportData.transportLeafMeasure,
        TensorPower.ReplicaTransport.TransportData.transportLeafDensity_eq}
    \leanok
    \uses{def:replica_coherent_sphere_measure}
    For the terminal state $\sigma_{j,u}$ of the transport tree, let
    $d_j(u,U)=m_{1/4}(u)q_{\sigma_{j,u}}(U)$ and define
    \begin{align}
        \nu_j & =((u,U)\mapsto Ue_a)_*
        \bigl((d_j)_+(du\otimes H)\bigr).
        \label{eq:replica_fourier_measure}
    \end{align}
    Neither a terminal weight nor a factor of two occurs in this definition.
    The parameter is clamped to $[0,1]$; a zero terminal weight gives the zero
    normalized derivative map, using the convention $0^{-1}=0$.
\end{definition}

\begin{theorem}[Fourier-coherent integral representation]
    \label{thm:replica_fourier_leaf_integral}
    \lean{TensorPower.ReplicaTransport.TransportData.measurable_transportLeafDensity,
        TensorPower.ReplicaTransport.TransportData.integrable_transportLeafDensity_mul,
        TensorPower.ReplicaTransport.TransportData.integrable_transportLeafDensity,
        TensorPower.ReplicaTransport.TransportData.isFiniteMeasure_transportLeafMeasure,
        TensorPower.ReplicaTransport.TransportData.integral_transportLeafMeasure,
        TensorPower.ReplicaTransport.TransportData.transportLeafMeasure_real_univ,
        TensorPower.ReplicaTransport.TransportData.transportLeafMeasure_real_univ_eq_ite,
        TensorPower.ReplicaTransport.TransportData.transportLeafMeasure_zero_pre,
        TensorPower.ReplicaTransport.TransportData.transportLeafMeasure_new_zero,
        TensorPower.ReplicaTransport.TransportData.transportLeafMeasure_old_one,
        TensorPower.ReplicaTransport.TransportData.continuous_state,
        TensorPower.ReplicaTransport.TransportData.integrable_transportLeafMeasure,
        TensorPower.ReplicaTransport.TransportData.transportLeafDensity_nonneg,
        TensorPower.ReplicaTransport.TransportData.state_eq_zero_of_weight_eq_zero,
        TensorPower.ReplicaTransport.TransportData.transportLeafMeasure_eq_zero_of_weight_eq_zero,
        TensorPower.ReplicaTransport.TransportData.transportLeafMeasure_real_univ_of_weight_ne_zero}
    \leanok
    \uses{def:replica_fourier_leaf_measure}
    For fixed real $p$, replica number $k$ and supplied vector, $d_j$ is jointly
    measurable and absolutely integrable, and $\nu_j$ is finite. For admissible
    data, nonnegative regulator parameter and cross-band commutation on
    $\mathcal S_k$, every continuous real $f$ on $\mathcal V$ satisfies
    \begin{align}
        \int_\Theta f\,d\nu_j
         & =\int_{\mathbb R}m_{1/4}(u)
        \int_\Theta f\,d\mu_{\sigma_{j,u}}\,du.
        \label{eq:replica_fourier_integral}
    \end{align}
    Under the same structural hypotheses, a nonzero symmetric supplied vector
    gives $\nu_j(\Theta)=1/2$ if the terminal weight is nonzero, and $\nu_j=0$
    if that weight is zero. Thus every interior
    leaf has mass $1/2$, the new leaves vanish at $p=0$, and the old leaves vanish
    at $p=1$. A zero supplied vector gives zero measure. These assertions include
    $k=0$.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:replica_coherent_measure_integral, lem:replica_transport_states}
    Write $\sigma_{j,u}=\Phi_j^*(|M^{-iu}v\rangle\langle M^{-iu}v|)$. The function
    $(W,U)\mapsto q_{\Phi_j^*(|Wv\rangle\langle Wv|)}(U)$ is continuous on
    $U(\mathcal V^{\otimes k})\times U(\mathcal V)$, hence bounded. Since
    $M^{-iu}$ is unitary and $\int m_{1/4}=1/2$, this bounds the absolute joint
    density by an integrable function. The same argument includes a continuous
    spherical symbol. Positivity of the actual terminal state removes the positive
    part, and Fubini's theorem gives (\ref{eq:replica_fourier_integral}). For
    nonzero terminal weight the trace of the state is one; at zero weight the
    normalized derivative map is zero. Apply the coherent mass formula and
    $\int m_{1/4}=1/2$. Compactness at fixed $p$ makes no uniform assertion in $p$.
\end{proof}
