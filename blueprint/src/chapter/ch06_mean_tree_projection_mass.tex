\section{Retained projection mass}

\begin{theorem}[The logarithmic loss from discarded projection mass]
\label{thm:mean_tree_projection_mass_log_bound}
\lean{Matrix.MeanTree.re_dotProduct_log_eval_ge_of_projection_mass}
\leanok
\uses{def:mean_tree}
Let $P$ be an orthogonal projection on a finite-dimensional complex
Hilbert space, and let $T$ be a weighted binary tree with labels in a
finite set $J$ and terminal weights $w_j$. Suppose $b>0$ and, for each
$j\in J$, the real number $d_j$ and matrix $A_j$ satisfy
\begin{align}
 b&\le d_j,\notag\\
 b(\Id-P)+d_jP&\le A_j.
 \notag
\end{align}
Let $v$ be a unit vector and let $\delta\in\mathbb R$ satisfy
$1-\delta\le\langle v,Pv\rangle$. Put $\ell=\sum_{j\in J}w_j\log d_j$.
Then
\begin{align}
 \ell-\delta(\ell-\log b)
 \le\operatorname{Re}\langle v,\log(T(A))v\rangle.
 \notag
\end{align}
No upper bound on $\delta$ or commutation of $P$ with the inputs is
required. This is the expectation step in the lower comparison of
\cite[Section~8]{OpenAI2026AreaLaw}.
\end{theorem}

\begin{proof}
\leanok
\uses{thm:mean_tree_projection_log_lower_bound,lem:mean_tree_weights}
The terminal-weight identities give $\ell\ge\log b$. Set
$q=\langle v,Pv\rangle$. Taking the expectation in
Theorem~\ref{thm:mean_tree_projection_log_lower_bound} gives
\begin{align}
 \operatorname{Re}\langle v,\log(T(A))v\rangle
 &\ge(\log b)(1-q)+\ell q\notag\\
 &=\log b+(\ell-\log b)q
 \ge\ell-\delta(\ell-\log b).
 \notag
\end{align}
The last inequality uses $q\ge1-\delta$ and $\ell-\log b\ge0$.
\end{proof}
