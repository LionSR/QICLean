\section{Random sampling of a viable subspace}

Let $M$ and $F$ be finite-dimensional complex inner product spaces, where $F$
stands for $\mathcal H_L\otimes\mathcal H_R$ and $M$ for $\mathcal H_M$ in
the setting of~\cite[lines 532--547]{Arad2016RigorousRG}; viability is read
up to the associator and the exchange of the first two factors.

\begin{theorem}[Net step]
    \label{thm:viable_random_sampling_net}
    \lean{Submodule.isCloseTo_of_net}
    \leanok
    \uses{thm:subspace_close_iff}
    Let $S$ and $T$ be subspaces of a finite-dimensional inner product space
    and let $c\ge0$. If every unit vector $t\in T$ lies within distance
    $\sqrt c$ of some $v\in T$ with $\lVert P_Sv\rVert^2\ge4c$, then
    $\lVert P_St\rVert^2\ge c\lVert t\rVert^2$ for all $t\in T$, that is, $S$
    is $(1-c)$-close to $T$.
\end{theorem}

\begin{proof}
    \leanok
    For a unit vector $u\in T$ choose $v$ as in the hypothesis. Then
    $\lVert P_Su\rVert\ge\lVert P_Sv\rVert-\lVert u-v\rVert
        \ge2\sqrt c-\sqrt c=\sqrt c$. A general nonzero $t\in T$ is
    $\lVert t\rVert$ times a unit vector of $T$.
\end{proof}

\begin{definition}[Random subspace]
    \label{def:viable_random_subspace}
    \lean{Submodule.randomSubspace, Submodule.finrank_randomSubspace}
    \leanok
    Let $W\subseteq M$ have an orthonormal basis indexed by a finite set
    $\iota$, and let $A\subseteq\iota$ have $s$ elements. For a unitary $U$
    on $\mathbb C^\iota$, the subspace $W'_U\subseteq W$ is the image under
    the basis isometry $\mathbb C^\iota\to W$ of $U$ applied to the span of
    the standard basis vectors indexed by $A$; it has dimension $s$. When $U$ is distributed by the
    Haar measure, $W'_U$ is a uniformly random $s$-dimensional subspace of
    $W$.
\end{definition}

\begin{theorem}[One vector]
    \label{thm:viable_random_sampling_one_vector}
    \lean{Submodule.unitaryHaar_norm_sq_starProjection_randomSubspace_le}
    \leanok
    \uses{def:viable_random_subspace, thm:haar_projection_law,
        thm:complex_gaussian_ratio_lower_tail}
    Let $q=\lvert\iota\rvert$. For every unit vector $f\in W$, the Haar
    probability that $\lVert P_{W'_U}f\rVert^2\le s/(2q)$ is at most
    $e^{-s/16}$.
\end{theorem}

\begin{proof}
    \leanok
    The basis isometry carries projections onto subspaces of
    $\mathbb C^\iota$ to projections onto their images, so the event is the
    event that the projection of a unit vector of $\mathbb C^\iota$ onto
    $U$ applied to the coordinate subspace of $A$ has squared norm at most
    $s/(2q)$. Its probability is the probability that
    $\sum_{i\in A}\lvert g_i\rvert^2\le\frac{s}{2q}\sum_i\lvert g_i\rvert^2$
    for independent standard complex Gaussians $g_i$, which is at most
    $e^{-s/16}$.
\end{proof}

\begin{theorem}[Random sampling]
    \label{thm:viable_random_sampling}
    \lean{Submodule.unitaryHaar_not_isViable_randomSubspace_le}
    \leanok
    \uses{thm:viable_random_sampling_net, thm:viable_random_sampling_one_vector,
        thm:tensor_weights, thm:ball_net, def:subspace_viable}
    Let $T\subseteq M\otimes F$ have dimension $r$, let $W\subseteq M$ be
    $\delta$-viable for $T$, and let $W'_U$ be as above with
    $q=\dim W$. Put
    \[
        \delta'=\frac{(1-\delta)s}{8q},\qquad\nu=\sqrt{\delta'},\qquad
        \eta=\Big(1+\frac2\nu\Big)^{2r}qe^{-s/16}.
    \]
    Then the Haar probability that $W'_U$ is not $(1-\delta')$-viable for
    $T$ is at most $\eta$. This is Lemma~l:rs
    of~\cite[lines 532--547]{Arad2016RigorousRG} with the failure
    probability corrected: the printed value
    $(1+4\sqrt{q/((1-\delta)s)})^rqe^{-s/16}$ undercounts the net of the
    unit sphere of $T$, a real space of dimension
    $2r$~\cite{gap:alvv16_random_sampling_net_count}.
\end{theorem}

\begin{proof}
    \leanok
    If $\delta'=0$ every subspace is $1$-viable. Otherwise choose a
    $\nu$-net $C$ of the unit sphere of $T$ with at most $(1+2/\nu)^{2r}$
    points. For $v\in C$ let $w$ be the projection of $v$ onto $W\otimes F$,
    so $\lVert w\rVert^2\ge1-\delta$ and
    $P_{W'_U\otimes F}v=P_{W'_U\otimes F}w$. Write
    $\lVert P_{W'_U\otimes F}w\rVert^2=\sum_k\mu_k\lVert P_{W'_U}f_k\rVert^2$
    along the weights of $w$. Outside the union of the events
    $\lVert P_{W'_U}f_k\rVert^2\le s/(2q)$, over $v\in C$ and the $q$ basis
    vectors $f_k$, every net point satisfies
    $\lVert P_{W'_U\otimes F}v\rVert^2\ge\frac{s}{2q}(1-\delta)=4\delta'$,
    and the net step gives $(1-\delta')$-viability. The union has
    probability at most $\lvert C\rvert qe^{-s/16}\le\eta$.
\end{proof}

\begin{theorem}[Existence of a viable sample]
    \label{thm:viable_random_sampling_exists}
    \lean{Submodule.exists_isViable_randomSubspace}
    \leanok
    \uses{thm:viable_random_sampling}
    In the setting of the random sampling theorem, if $\eta<1$ then some
    unitary $U$ makes $W'_U$ a $(1-\delta')$-viable subspace for $T$ of
    dimension $s$.
\end{theorem}

\begin{proof}
    \leanok
    Otherwise the failure event is the whole unitary group, of probability
    one, which exceeds $\eta$.
\end{proof}


\begin{lemma}[Sampling or retaining the whole space]
    \label{thm:viable_sample_or_full}
    \lean{Submodule.sampleIndices, Submodule.card_sampleIndices,
        Submodule.randomSubspace_eq_of_card_eq,
        Submodule.IsViable.randomSubspace_min_of_le,
        Submodule.exists_isViable_randomSubspace_min,
        Submodule.exists_measurable_randomSubspace_min_event}
    \leanok
    \uses{thm:viable_random_sampling}
    Fix a $\delta$-viable space $W$ of dimension $Q$ and an orthonormal
    basis, where $\delta\le1$. Sample $\min\{s,Q\}$ coordinate vectors
    after a Haar unitary. There is a measurable event on which the
    resulting space is $(1-c)$-viable, where
    $c=(1-\delta)\min\{s,Q\}/(8Q)$.
    Its complement has measure at most
    \begin{align}
      \eta&=\begin{cases}
        0,&Q\le s,\\
        \bigl(1+2/\sqrt{(1-\delta)s/(8Q)}\bigr)^{2\dim T}
          Qe^{-s/16},&s<Q.
      \end{cases}\notag
    \end{align}
    The event has measure at least $1-\max\{\eta,0\}$.
    For $Q\le s$ the sample is exactly $W$ for every unitary.
    This assertion concerns a fixed input space; it does not assert
    measurable dependence on a varying input.
\end{lemma}
\begin{proof}\leanok
    A full coordinate sample has dimension $Q$ and is contained in $W$.
    In the other case take the complement of a measurable hull of the
    bad set in the corrected sampling estimate. The hull has the same
    measure as the bad set, and its complement consists of successful
    samples.
\end{proof}


\begin{lemma}[The actual sampled word image]
    \label{thm:sampled_operator_word_image}
    \lean{Submodule.sampledWordImage,
        Submodule.finrank_sampledWordImage_le,
        Submodule.exists_measurable_sampledWordImage_event}
    \leanok
    \uses{thm:viable_sample_or_full,thm:spectral_operator_word_powers}
    Suppose a fixed family of $N$ operators has the following property:
    the sum of all length-$a$ word images of a $\delta$-viable space for $T$ is
    $\Delta^a/(1-\delta)^2$-viable for $T$, whenever $\delta<1$.
    Apply the family to the actual sample above. On its measurable
    success event the span of the actual length-$a$ images is
    $\Delta^a/c^2$-viable for the same $T$ and has dimension at most
    $N^a\min\{s,Q\}$, provided $a,s\ge1$, $W\ne0$ and $\delta<1$.
    The operators and target precede the input and the Haar unitary.
\end{lemma}
\begin{proof}\leanok
    Apply the assumed word-improvement property to each successful
    sample. The dimension of each image is at most the sample dimension;
    summing over the $N^a$ ordered words gives the count.
\end{proof}


\begin{lemma}[A uniform interval trimming radius]
    \label{lem:viable_node_trimming_radius}
    \lean{Real.viable_node_trimming_radius_budget}
    \leanok

\end{lemma}
\begin{proof}\leanok

\end{proof}


\begin{lemma}[A common sampling size at the cubic dimension scale]
    \label{thm:viable_node_uniform_sampling_size}
    \lean{Real.viable_node_sample_size_log_bound,
        Real.viable_node_sampling_failure_le_exp,
        Real.viable_node_uniform_sampling_size}
    \leanok
    \uses{thm:viable_sample_or_full}
    Let $D\ge2$ and $r\ge1$ be integers and let $0<\beta\le1$. Put
    \begin{align}
      A&=1+\log r+\log D+\log(1/\beta), &
      s&=\lceil4096rA\rceil.\notag
    \end{align}
    Then $1\le s\le4097rA$. For every $Q\le D^{12}s^6$, the
    corrected failure bound for a $1/4$-viable input of dimension $Q$
    and a target of dimension $r$ satisfies
    \begin{align}
      \eta_Q&\le e^{-s/64}\le\beta.\notag
    \end{align}
    Thus the same size is chosen before every parent space.
    This is an auxiliary choice for the powered cubic profile, rather
    than an identification with the source's printed tree parameters.
\end{lemma}
\begin{proof}\leanok
    The ceiling estimate gives $\log s\le13A$.
    When $Q>s$, the net factor is at most $16\sqrt Q$, so
    \begin{align}
      \log\eta_Q
      &\le32r\log D+12r\log s-s/16
      \le188rA-s/16\le-s/64.\notag
    \end{align}
    Also $s/64\ge\log(1/\beta)$. For $Q\le s$ the failure bound is
    zero because the whole input is retained.
\end{proof}


\begin{lemma}[Counts and ranks at the common node scales]
    \label{lem:sampled_node_count_rank}
    \lean{Submodule.finrank_sampledWordImage_le_cubic,
        Real.viable_node_trimming_rank_bound}
    \leanok
    \uses{thm:sampled_operator_word_image,lem:viable_node_trimming_radius}
    Suppose $D\ge2$, $s\ge1$ and a fixed family has at most $D^2$
    operators. At $a=\lceil\log_D(D^2s)\rceil$, the literal sampled-word
    image has dimension at most $D^6s^3$, for every unitary, before imposing
    a success event. Moreover, if $D_{\rm ref}\ge1$ and
    $\xi=(1000D_{\rm ref}\sqrt{L+1})^{-1}$, then for nonnegative integers $R,M$,
    $\xi^2R\le M$ implies the bound
    $R\le10^6D_{\rm ref}^2(L+1)M$.
\end{lemma}
\begin{proof}\leanok
    The image dimension is at most $(D^a)^2s\le D^6s^3$.
    For the rank estimate, square the positive radius and multiply the
    weighted inequality by its reciprocal.
\end{proof}


\begin{lemma}[Borel Gram--Schmidt transformations]
    \label{lem:borel_gram_schmidt}
    \lean{InnerProductSpace.measurable_gramSchmidt,
        InnerProductSpace.measurable_gramSchmidtNormed}
    \leanok
    Let $E$ be a separable real or complex inner product space. For every
    finite family $x=(x_0,\ldots,x_{n-1})$, the Gram--Schmidt transformation
    $x\mapsto(y_0,\ldots,y_{n-1})$ is Borel. So is its normalized version
    $x\mapsto(\|y_i\|^{-1}y_i)_{i<n}$, with the convention $0^{-1}=0$.
    Linear independence is not required: zero vectors are retained, and
    the empty family is permitted.
\end{lemma}
\begin{proof}\leanok
    Induct over the finite ordered indices in
    \[
        y_i=x_i-\sum_{j<i}
          \frac{\langle y_j,x_i\rangle}{\|y_j\|^2}y_j.
    \]
    Each term is Borel because scalar products, finite sums and scalar
    multiplication are continuous, while reciprocation extended by zero
    is Borel. Norms and the same reciprocal convention give the normalized
    statement. This is a prerequisite for measurable varying frames; it
    does not itself select a frame or establish an adaptive sampling law.
\end{proof}


\begin{definition}[Ordered orthonormal frames from finite generators]
    \label{def:finite_ordered_orthonormal_frame}
    \lean{InnerProductSpace.finiteOrthonormalFrame,
        Matrix.rangeOrthonormalFrame}
    \leanok
    For a finite ordered family $f$ whose span has dimension $r$, apply
    normalized Gram--Schmidt, remove its zero vectors and retain the original
    order. Denote the resulting $r$-vector frame by $\mathcal F_r(f)$.
    For a rank-$r$ matrix $A$ on $\C^d$, apply this construction to the
    ordered columns $Ae_0,\ldots,Ae_{d-1}$, obtaining
    $\mathcal R_r(A)$. Rank zero and empty families are permitted.
\end{definition}

\begin{lemma}[Borel frame on finite-family rank strata]
    \label{lem:borel_finite_ordered_frame}
    \lean{InnerProductSpace.orthonormal_finiteOrthonormalFrame,
        InnerProductSpace.span_finiteOrthonormalFrame,
        InnerProductSpace.measurable_finiteOrthonormalFrame}
    \leanok
    On finite generating families in a separable real or complex inner
    product space, $\mathcal F_r(f)$ is orthonormal and spans the same
    space as $f$. The map $f\mapsto\mathcal F_r(f)$ is Borel on the
    stratum where the span has dimension $r$.
\end{lemma}
\begin{proof}\leanok
    \uses{def:finite_ordered_orthonormal_frame,lem:borel_gram_schmidt}
    The nonzero normalized Gram--Schmidt vectors form an orthonormal family
    with the original span, so their number is $r$. Their increasing
    enumeration gives the claimed frame. For measurability, partition the
    domain according to the subset of indices at which these vectors are
    nonzero. Each of the finitely many pieces is Borel. On a fixed piece
    the frame is a fixed coordinate restriction of the Borel normalized
    Gram--Schmidt map. These restrictions agree with the defined ordered
    frame, and hence assemble to a Borel map. No arbitrary choice of an
    orthonormal basis is used.
\end{proof}

\begin{lemma}[Borel orthonormal matrix-range frame]
    \label{lem:borel_matrix_range_frame}
    \lean{Matrix.orthonormal_rangeOrthonormalFrame,
        Matrix.span_rangeOrthonormalFrame,
        Matrix.measurable_rangeOrthonormalFrame}
    \leanok
    On the finite complex matrices of rank $r$, the frame
    $\mathcal R_r(A)$ is orthonormal, spans exactly $\operatorname{ran}A$,
    and depends in a Borel manner on $A$. In particular this holds on
    each rank-$r$ stratum of orthogonal projections, including $r=0$ and
    the empty ambient index set.
\end{lemma}
\begin{proof}\leanok
    \uses{lem:borel_finite_ordered_frame}
    Matrix columns depend continuously on the matrix. Their span is its
    actual range and has dimension equal to the matrix rank. Apply the
    preceding lemma to that family. No global continuity or running-time
    assertion is made, and an adaptive joint sampling law remains separate.
\end{proof}


\begin{definition}[Actual range basis and rotated columns]
    \label{def:canonical_range_sampling_basis}
    \lean{Matrix.rangeOrthonormalBasis,Matrix.rotatedRangeFrame}
    \leanok
    For a rank-$r$ matrix $P$ on $\C^d$, let $W_P=\operatorname{ran}P$.
    Restrict the ordered frame $C(P)=\mathcal R_r(P)$ to $W_P$, obtaining
    its actual orthonormal basis $b_P$. For $U\in\mathrm U(r)$ put
    \[
        z_j(P,U)=\sum_{i<r}U_{ij}C(P)_i.
    \]
    The basis object is used pointwise; no measurable structure on a
    varying space of basis objects is assumed.
\end{definition}

\begin{lemma}[Projection of the existing sampled subspace]
    \label{lem:existing_sampler_rank_one_projection}
    \lean{Submodule.basisEmbedding_apply_eq_sum,
        Submodule.randomSubspace_eq_span_rotated_basis,
        Submodule.starProjection_randomSubspace_eq_sum_rankOne}
    \leanok
    Let $b=(b_i)_{i\in\iota}$ be an orthonormal basis of $W\subseteq M$,
    with $\iota$ finite, and write $E_bx=\sum_i x_i b_i$ for its ambient
    isometric embedding. For $I\subseteq\iota$ and a unitary $U$, the
    existing sampler of Definition~\ref{def:viable_random_subspace} satisfies
    \[
        W'_{b,I,U}=\operatorname{span}\{E_bUe_j:j\in I\},\qquad
        P_{W'_{b,I,U}}=\sum_{j\in I}|E_bUe_j\rangle\langle E_bUe_j|.
    \]
    Empty retained sets are allowed.
\end{lemma}
\begin{proof}\leanok
    \uses{def:viable_random_subspace}
    The two linear images in the existing sampler take its coordinate
    span to the displayed span. The rotated and embedded full family is
    orthonormal. Its retained subfamily is an orthonormal basis of that
    span, so its rank-one sum is the actual ambient orthogonal projection.
\end{proof}

\begin{lemma}[The canonical basis gives the actual sampled projector]
    \label{lem:canonical_sampler_projector_formula}
    \lean{Matrix.coe_rangeOrthonormalBasis,
        Matrix.basisEmbedding_rangeOrthonormalBasis_unitaryAction_eq,
        Matrix.randomSubspace_rangeOrthonormalBasis_projection_eq_sum}
    \leanok
    The ambient coordinates of $b_{P,i}$ are exactly $C(P)_i$, and
    $E_{b_P}Ue_j=z_j(P,U)$. Consequently the ambient matrix of the
    projection of the existing sampler $W'_{b_P,I,U}$ is
    \[
        \sum_{j\in I}z_j(P,U)z_j(P,U)^*.
    \]
    No different sampler is introduced.
\end{lemma}
\begin{proof}\leanok
    \uses{def:canonical_range_sampling_basis,lem:existing_sampler_rank_one_projection}
    Restricting the frame to its exact range preserves each ambient
    coordinate. Expand $Ue_j$ in the coordinate basis and apply the
    finite basis expansion for $E_{b_P}$. The preceding projector identity
    gives the stated matrix Gram sum.
\end{proof}

\begin{lemma}[A joint Borel projector for the canonical-frame sampler]
    \label{lem:joint_borel_canonical_sample_projector}
    \lean{Matrix.measurable_rotatedRangeFrame,
        Matrix.measurable_randomSubspace_rangeOrthonormalBasis_projection}
    \leanok
    Fix $d,r$ and a retained coordinate set $I\subseteq\{0,\ldots,r-1\}$.
    On the product of the rank-$r$ matrix stratum and $\mathrm U(r)$,
    both $(P,U)\mapsto(z_j(P,U))_{j<r}$ and the actual ambient projector
    $(P,U)\mapsto P_{W'_{b_P,I,U}}$ are Borel.
    In particular $I$ may be the first $\min\{s,r\}$ coordinates.
    Rank zero, $s=0$, empty retained sets and empty ambient dimension
    are included.
\end{lemma}
\begin{proof}\leanok
    \uses{lem:borel_matrix_range_frame,lem:canonical_sampler_projector_formula}
    The frame coordinates are Borel and the unitary entries are continuous.
    Their finite linear combinations are therefore jointly Borel. Each
    entry of the projector Gram sum is a finite sum of products of a
    coordinate and a conjugate coordinate, and hence is Borel.
\end{proof}

This is a statement about one fixed-rank sampling step. It neither asserts
measurability of the old arbitrary varying basis choice, proves a comparison
of Haar laws under basis changes, nor constructs an adaptive tree law or a
running-time estimate.

\subsection{A common finite unitary space for varying ranks}

\begin{lemma}[Borel gluing along matrix-rank strata]
    \label{lem:borel_matrix_rank_gluing}
    \lean{Matrix.measurable_rank,measurable_of_matrix_rank_strata}
    \leanok
    Matrix rank is Borel on finite complex matrices. If $P:X\to M_{m,n}(\C)$
    is Borel and $f:X\to Y$ is Borel on each set
    $\{x:\operatorname{rank}P(x)=r\}$, then $f$ is Borel on $X$.
\end{lemma}
\begin{proof}\leanok
    Matrix rank is lower semicontinuous. Its level sets are therefore
    Borel and form a countable partition. The inverse image of each
    measurable subset of $Y$ is the union of its inverse images in the
    rank strata.
\end{proof}

\begin{lemma}[The actual sample, family image and trim are jointly Borel]

    \leanok
    Fix the operator family and collective-trimming parameters before the

\begin{lemma}[Actual projections in relabeled tensor coordinates]
    \label{lem:borel_actual_tensor_projection_coordinates}
    \lean{Submodule.projectionMatrix_map_piLpCongrLeft,
        Submodule.measurable_projectionMatrix_map_piLpCongrLeft,
        Submodule.projectionMatrix_tensorProd_coordinates,
        Submodule.measurable_projectionMatrix_tensorProd_coordinates}
    \leanok
    Relabeling finite coordinates relabels both indices of the actual
    orthogonal projection by the same equivalence. In the fixed product
    coordinates, the projection onto $S\otimes R$ is exactly
    $P_S\otimes P_R$. Consequently Borel actual factor projections give
    a Borel actual tensor-product projection. Empty coordinate sets and
    zero subspaces are included.
\end{lemma}
\begin{proof}\leanok
    Orthogonal projections commute with fixed coordinate isometries.
    The tensor-product projection is the tensor product of the factor
    projections, and the fixed product orthonormal basis represents this
    operator by the Kronecker matrix. Coordinate relabeling and the
    Kronecker map are continuous on finite matrix spaces.
\end{proof}

\begin{lemma}[Borel true viability with the full exterior]

    \leanok

\end{lemma}
\begin{proof}\leanok
    \uses{lem:borel_actual_tensor_projection_coordinates,
        lem:borel_actual_sample_family_trim_closeness}

    $P_{S(x)}\otimes\Id$. Carry the fixed target back through the fixed
    tensor-coordinate isometry and apply the true closeness test. This
    is exactly the original viability predicate; the target is unchanged
    throughout and need not be a product subspace.
\end{proof}

\begin{lemma}[The full short-block dimension fits the common node count]
    \label{lem:viable_tree_full_leaf_count}
    \lean{Nat.viable_tree_leaf_count_le}

\begin{lemma}[Sampling at a size determined by the input rank]
    \label{lem:borel_rank_size_sample_family_trim}
    \lean{Matrix.measurable_randomRangeSubspace_projection_rank_size,
        Matrix.measurable_familySupportProj_randomRangeSubspace_rank_size,
        Entropy.measurable_projectionMatrix_trim_sample_family_image_rank_size,
        Entropy.measurableSet_isViable_trim_sample_family_image_rank_size}
    \leanok
    Fix a function $s:\mathbb N\to\mathbb N$, a finite operator family
    $F_i:\mathbb C^a\to\mathcal H$, and the collective-trimming cuts,
    threshold and number of cuts on the finite physical space
    $\mathcal H$. Let $P$ be a matrix on $\mathbb C^a$ and put
    $Q=\operatorname{rank}P$. From the $Q$th coordinate of the fixed
    product $\Omega_a=\prod_{j=0}^a\mathrm U(j)$, take the canonical
    range sample $W(P,U)$ of dimension $\min\{s(Q),Q\}$. Set
    \[
       V(P,U)=\sum_i F_iW(P,U),\qquad
       S(P,U)=\operatorname{Trim}_\xi V(P,U).
    \]
    The actual orthogonal projections onto $W(P,U)$, $V(P,U)$ and
    $S(P,U)$ are Borel in $(P,U)$. For every fixed target
    $T\subseteq\mathcal H\otimes\mathcal E$ and every real $\delta$,
    the true event that $S(P,U)$ is $\delta$-viable for $T$ is Borel.
    The entire finite-dimensional exterior $\mathcal E$ is retained.
\end{lemma}
\begin{proof}\leanok
    \uses{def:canonical_variable_rank_sample,
        lem:borel_matrix_rank_gluing,
        lem:borel_canonical_collective_trim_matrix_range,
        lem:borel_true_regional_viability_event}
    On the stratum $\operatorname{rank}P=Q$, the requested size is the
    fixed integer $s(Q)$. The existing canonical sample therefore has
    a Borel projection there. Joining the rank strata gives the first
    assertion. Apply the fixed operator family and the actual collective
    trim, then the true viability test for the fixed target.
\end{proof}

This permits, for example, the actual quotient
$s(Q)=\lfloor Q/(2D^2)\rfloor$ in repeated amplification associated with
\cite{Arad2016RigorousRG}. It gives measurability of these operations;
a bound for the probability of an entire adaptive construction requires
a separate argument.
