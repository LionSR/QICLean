\chapter{Scalar Estimates for Buffered Filters}
\label{ch:buffered_filters}

The conditional-entropy discount in a buffered rectangle
of~\cite[Lemma~3.2]{OpenAI2026AreaLaw} optimizes positive filters on nested
contours. This chapter collects two scalar ingredients: the weights of the
contours and the iteration of the safe-box recurrence
of~\cite[Proposition~3.3]{OpenAI2026AreaLaw}.

\begin{lemma}[Harmonic weights]
    \label{lem:harmonic_weights}
    \lean{Entropy.sum_harmonicWeight, Entropy.sum_mul_harmonicWeight_sq,
        Entropy.harmonic_cost_le, Entropy.log_div_le_sum_inv_contour,
        Entropy.log_div_le_sum_inv_contour_of_le}
    \leanok
    For positive costs $c_j$ put $Z=\sum_j c_j^{-1}$ and $a_j=2/(Zc_j)$. Then
    $\sum_ja_j=2$ and $\sum_jc_ja_j^2=4/Z$, and every family of weights of
    total mass two has quadratic cost at least $4/Z$. For $c_j=2r+jD$,
    $1\le j\le m$, with $r\ge1$, $D>0$, $P\ge1$ and $D(m+1)\ge(P-1)r$,
    \[
        \sum_{j=1}^m\frac1{2r+jD}\ge\frac1D\log\frac{P+1}{2+D}.
    \]
\end{lemma}
\begin{proof}\leanok
    The mass and cost are direct computations, and optimality is
    Cauchy--Schwarz. For the lower bound, $\log(x+D)-\log x\le D/x$
    telescopes to $\frac1D\log\frac{2r+D(m+1)}{2r+D}$, and
    $2r+D(m+1)\ge(P+1)r$ while $2r+D\le(2+D)r$.
\end{proof}

\begin{lemma}[Power bound from a scale recurrence]
    \label{lem:scale_recurrence}
    \lean{Entropy.le_pow_of_scale_recurrence,
        Entropy.exists_le_mul_rpow_of_scale_recurrence,
        Entropy.one_lt_log_div_log_lt_two}
    \leanok
    Let $F$ be nondecreasing on the positive integers, $M\ge2$ an integer,
    $\lambda>M$ and $K\ge0$, with $F(Mr)\le\lambda F(r)+Kr$ for all
    $r\ge1$. Then $F(r)\le Cr^{\log_M\lambda}$ for all $r\ge1$. If
    $M<\lambda<M^2$, then $1<\log_M\lambda<2$.
\end{lemma}
\begin{proof}\leanok
    By induction,
    $F(M^j)\le\lambda^jF(1)+K(\lambda^j-M^j)/(\lambda-M)\le C\lambda^j$.
    For $M^{j-1}<r\le M^j$, monotonicity gives
    $F(r)\le C\lambda^j=C(M^j)^{\log_M\lambda}\le C(Mr)^{\log_M\lambda}$.
\end{proof}
