\section{Trace pairings weighted by a positive semidefinite matrix}

For a positive semidefinite matrix $\sigma$, the map
$(X,Y)\mapsto\operatorname{Tr}(\sigma X^\dagger Y)$ is a positive semidefinite
sesquilinear form on matrices. The following two facts are used for the
Cauchy--Schwarz step of the expectation bound in Hastings' one-dimensional area
law \cite[lines 440--461]{Hastings2007AreaLaw}. The norm $\|A\|$ is the operator
norm, and $A\le B$ means that $B-A$ is positive semidefinite.

\begin{theorem}[Cauchy--Schwarz for the expectation in a positive matrix]
    \label{thm:state_weighted_cauchy_schwarz}
    \lean{Matrix.PosSemidef.norm_trace_mul_mul_le}
    \leanok
    Let $\sigma$ be positive semidefinite and let $A$, $B$ be complex matrices of
    the same size. Then
    \begin{align*}
        \bigl|\operatorname{Tr}(\sigma AB)\bigr|
        \le\sqrt{\operatorname{Re}\operatorname{Tr}(\sigma AA^\dagger)}\,
        \sqrt{\operatorname{Re}\operatorname{Tr}(\sigma B^\dagger B)}.
    \end{align*}
\end{theorem}

\begin{proof}
    \leanok
    The form $\langle X,Y\rangle_\sigma=\operatorname{Tr}(\sigma X^\dagger Y)$ is
    positive semidefinite, and $\operatorname{Tr}(\sigma AB)=\langle A^\dagger,B\rangle_\sigma$.
    Apply the Cauchy--Schwarz inequality for this form.
\end{proof}

\begin{theorem}[Monotonicity under a positive contraction]
    \label{thm:state_weighted_contraction_square}
    \lean{Matrix.PosSemidef.re_trace_mul_mul_self_le}
    \leanok
    Let $\sigma$ be positive semidefinite and let $A$ be positive semidefinite with
    $\|A\|\le1$. Then
    $\operatorname{Re}\operatorname{Tr}(\sigma AA)\le\operatorname{Re}\operatorname{Tr}(\sigma A)$.
\end{theorem}

\begin{proof}
    \leanok
    Since $0\le A$ and $\|A\|\le1$, one has $A^2\le A$. The trace of $\sigma$ against
    the positive semidefinite matrix $A-A^2$ is nonnegative.
\end{proof}
