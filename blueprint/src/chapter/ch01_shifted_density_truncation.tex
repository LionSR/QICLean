\section{Spectral truncation of shifted powers}
\label{sec:shifted_density_truncation}

Let $I$ be a finite index set. All matrices in this section act on $\C^I$,
and all norms are Euclidean operator norms. The rank is measured on this
space, before tensoring with an identity on any other space.

\begin{definition}[Closed-threshold spectral projection]
    \label{def:closed_threshold_projection}
    \lean{Matrix.spectralProjectionGE}
    \leanok
    For a Hermitian matrix $A$ and $b\in\R$, let
    $\Pi=\mathbf1_{[b,\infty)}(A)$ be the spectral projection onto the
    eigenvalues at least $b$. An eigenvalue equal to $b$ is included.
\end{definition}

\begin{theorem}[Projection and commutation]
    \label{thm:closed_threshold_projection}
    \lean{Matrix.IsHermitian.spectralProjectionGE_eq_cfc,
        Matrix.IsHermitian.isStarProjection_spectralProjectionGE,
        Matrix.IsHermitian.commute_spectralProjectionGE}
    \leanok
    \uses{def:closed_threshold_projection}
    Let $A$ be Hermitian and $b\in\R$. The projection $\Pi$ is the
    functional calculus of $\mathbf1_{[b,\infty)}$ at $A$, and satisfies
    $\Pi^*=\Pi$, $\Pi^2=\Pi$, and $A\Pi=\Pi A$.
\end{theorem}

\begin{proof}
    \leanok
    The spectrum of $A$ is finite, so its indicator is continuous on that
    spectrum. The functional calculus preserves real-valued functions and
    products. The scalar identities $\mathbf1_{[b,\infty)}^2=
        \mathbf1_{[b,\infty)}$ and
    $t\mathbf{1}_{\lbrack b,\infty)}(t)=\mathbf{1}_{\lbrack b,\infty)}(t)t$
    give the assertions.
\end{proof}

\begin{definition}[Head and tail of a shifted power]
    \label{def:shifted_power_head_tail}
    \lean{Matrix.shiftedPowerHead, Matrix.shiftedPowerTail}
    \leanok
    \uses{def:closed_threshold_projection}
    Let $A\ge0$, $b>0$, and $r\in\R$. Write $F=(A+b\Id)^r$ and define
    \begin{align}
        H & =F\Pi, & T & =F(\Id-\Pi).
        \label{eq:shifted_power_split_definition}
    \end{align}
\end{definition}

\begin{theorem}[Functional calculus of the shifted power]
    \label{thm:shifted_power_functional_calculus}
    \lean{Matrix.PosSemidef.add_smul_one_rpow_eq_cfc,
        Matrix.PosSemidef.commute_add_smul_one_rpow_spectralProjectionGE}
    \leanok
    \uses{def:shifted_power_head_tail}
    If $A\ge0$, $b>0$, and $r\in\R$, then $F=f(A)$ for
    $f(t)=(t+b)^r$, and $F\Pi=\Pi F$.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:positive_identity_shift, thm:closed_threshold_projection}
    The matrix $A+b\Id$ is positive definite by
    Theorem~\ref{thm:positive_identity_shift}. Composing the functional
    calculus for $t\mapsto t+b$ and $t\mapsto t^r$ gives $F=f(A)$.
    Both $F$ and $\Pi$ are functions of $A$, so they commute.
\end{proof}

\begin{theorem}[Exact head--tail decomposition]
    \label{thm:shifted_power_split}
    \lean{Matrix.shiftedPowerHead_add_shiftedPowerTail}
    \leanok
    \uses{def:shifted_power_head_tail}
    The matrices in~(\ref{eq:shifted_power_split_definition}) satisfy $H+T=F$.
\end{theorem}

\begin{proof}
    \leanok
    Distributivity gives $F\Pi+F(\Id-\Pi)=F$.
\end{proof}

\begin{theorem}[Scalar descriptions of the head and tail]
    \label{thm:shifted_power_truncated_functions}
    \lean{Matrix.PosSemidef.shiftedPowerHead_eq_cfc,
        Matrix.PosSemidef.shiftedPowerTail_eq_cfc}
    \leanok
    \uses{def:shifted_power_head_tail}
    Let $A\ge0$, $b>0$, and $r\in\R$. Then $H=h(A)$ and $T=t(A)$,
    where
    \begin{align}
        h(\lambda) & =
        \begin{cases}(\lambda+b)^r,&\lambda\ge b,\\0,&\lambda<b,\end{cases}
        \label{eq:shifted_power_head_function} \\
        t(\lambda) & =
        \begin{cases}0,&\lambda\ge b,\\(\lambda+b)^r,&\lambda<b.\end{cases}
        \label{eq:shifted_power_tail_function}
    \end{align}
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:shifted_power_functional_calculus, thm:closed_threshold_projection}
    Multiplication in the functional calculus gives
    $h(\lambda)=(\lambda+b)^r\mathbf1_{[b,\infty)}(\lambda)$.
    Subtraction gives
    $t(\lambda)=(\lambda+b)^r-h(\lambda)$, yielding the two cases.
\end{proof}

\begin{theorem}[Positivity and orthogonality]
    \label{thm:shifted_power_positive_orthogonal}
    \lean{Matrix.PosSemidef.shiftedPowerHead_posSemidef,
        Matrix.PosSemidef.shiftedPowerTail_posSemidef,
        Matrix.PosSemidef.shiftedPowerHead_mul_shiftedPowerTail,
        Matrix.PosSemidef.shiftedPowerTail_mul_shiftedPowerHead}
    \leanok
    \uses{def:shifted_power_head_tail}
    For $A\ge0$, $b>0$, and every $r\in\R$, one has
    $H\ge0$, $T\ge0$, and $HT=TH=0$.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:shifted_power_truncated_functions}
    On the spectrum of $A$, $\lambda+b>0$. Thus both functions in
    (\ref{eq:shifted_power_head_function}) and
    (\ref{eq:shifted_power_tail_function}) are nonnegative, and their
    pointwise product is zero. The functional calculus gives positivity
    and $HT=0$. Taking adjoints gives $TH=0$.
\end{proof}

\begin{theorem}[Inverse shifted powers]
    \label{thm:shifted_power_inverse}
    \lean{Matrix.PosSemidef.add_smul_one_rpow_mul_neg}
    \leanok
    If $A\ge0$, $b>0$, and $r\in\R$, then
    $(A+b\Id)^r(A+b\Id)^{-r}=\Id$.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:positive_identity_shift}
    Strict positivity of $A+b\Id$ permits multiplication of its real
    powers. The exponents add to zero.
\end{proof}

\begin{theorem}[Range and rank of the head]
    \label{thm:shifted_power_head_range}
    \lean{Matrix.PosSemidef.range_shiftedPowerHead,
        Matrix.PosSemidef.rank_shiftedPowerHead}
    \leanok
    \uses{def:shifted_power_head_tail}
    For $A\ge0$, $b>0$, and $r\in\R$, the head satisfies
    $\operatorname{ran}H=\operatorname{ran}\Pi$ and
    $\operatorname{rank}H=\operatorname{rank}\Pi$.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:shifted_power_functional_calculus, thm:shifted_power_inverse}
    Commutation gives $H=\Pi F$, so
    $\operatorname{ran}H\subseteq\operatorname{ran}\Pi$.
    The inverse identity yields $\Pi=H(A+b\Id)^{-r}$, proving the reverse
    inclusion. Taking dimensions gives equality of ranks.
\end{proof}

\begin{theorem}[Spectral counting and trace]
    \label{thm:closed_threshold_rank_trace}
    \lean{Matrix.PosSemidef.mul_rank_spectralProjectionGE_le_trace}
    \leanok
    \uses{def:closed_threshold_projection}
    For every $A\ge0$ and every real $b$,
    $b\operatorname{rank}\Pi\le\operatorname{Re}\tr A$.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:closed_threshold_projection}
    The rank of an orthogonal projection equals its trace. If
    $\lambda_i$ are the eigenvalues of $A$, then
    \begin{align}
        b\operatorname{rank}\Pi
        =\sum_{i\in I}b\mathbf1_{[b,\infty)}(\lambda_i)
        \le\sum_{i\in I}\lambda_i=\operatorname{Re}\tr A.
    \end{align}
    Each scalar inequality follows from $b\le\lambda_i$ on the selected
    spectrum and $0\le\lambda_i$ on its complement.
\end{proof}

\begin{theorem}[Head rank for a density matrix]
    \label{thm:shifted_density_head_rank}
    \lean{Matrix.PosSemidef.rank_shiftedPowerHead_le}
    \leanok
    \uses{def:shifted_power_head_tail}
    If $A\ge0$, $\tr A=1$, $b>0$, and $r\in\R$, then
    $\operatorname{rank}H\le b^{-1}$.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:shifted_power_head_range, thm:closed_threshold_rank_trace}
    Substitute $\operatorname{rank}H=\operatorname{rank}\Pi$ into the
    trace inequality and divide by $b>0$.
\end{proof}

\begin{theorem}[Tail norm for nonnegative powers]
    \label{thm:shifted_power_tail_norm}
    \lean{Matrix.PosSemidef.l2_opNorm_shiftedPowerTail_le}
    \leanok
    \uses{def:shifted_power_head_tail}
    If $A\ge0$, $b>0$, and $r\ge0$, then $\|T\|\le(2b)^r$.
    No trace normalization is required.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:shifted_power_truncated_functions}
    The tail function vanishes for $\lambda\ge b$. For the remaining
    spectral values, $0\le\lambda<b$, so $0<\lambda+b<2b$ and
    $(\lambda+b)^r\le(2b)^r$ by monotonicity for $r\ge0$.
    The norm estimate for functional calculus gives the result.
\end{proof}

For $A\ge0$, $b>0$, and $r\in\R$, consider the orthogonal decomposition
$\C^I=\operatorname{ran}\Pi\oplus\ker\Pi$. Let $F_{\ge b}$ and $F_{<b}$
be the restrictions of $F$ to the two summands. The split has the form
\begin{align}
    H+T
    =\begin{pmatrix}F_{\ge b}&0\\0&0\end{pmatrix}
    +\begin{pmatrix}0&0\\0&F_{<b}\end{pmatrix}
    =\begin{pmatrix}F_{\ge b}&0\\0&F_{<b}\end{pmatrix}=F.
\end{align}
The restriction $F_{\ge b}$ is invertible on $\operatorname{ran}\Pi$,
which explains why truncating the power preserves the projection's rank.

If $I$ is empty, the matrices, ranges, and ranks are zero, and the stated
identities and estimates remain valid. A trace-one matrix cannot occur in
that case. The exponent $r=0$ gives $F=\Id$, $H=\Pi$, and $T=\Id-\Pi$.


\subsection{Borel dependence of the spectral cutoffs}

The hard cutoff is not generally continuous when an eigenvalue crosses the
threshold. Borel measurability nevertheless holds without excluding that
crossing or fixing the rank.

\begin{definition}[Continuous closed-threshold approximants]
    \label{def:closed_threshold_ramp}
    \lean{Real.closedThresholdRamp}
    \leanok
    For $a\in\R$ and $k\in\N$, set
    \[
        f_{a,k}(x)=\max\{0,\min\{1,1+k(x-a)\}\}.
    \]
    In particular $f_{a,k}(a)=1$.
\end{definition}

\begin{lemma}[Finite-spectral stabilization]
    \label{lem:closed_threshold_finite_stabilization}
    \lean{Real.continuous_closedThresholdRamp,
        Real.eventually_closedThresholdRamp_eq,
        Matrix.IsHermitian.eventually_cfc_closedThresholdRamp_eq_spectralProjectionGE,
        Matrix.continuous_cfc_closedThresholdRamp}
    \leanok
    Each $f_{a,k}$ is continuous. At every fixed $x$, it eventually equals
    $\mathbf1_{[a,\infty)}(x)$. For each finite Hermitian matrix $A$,
    $f_{a,k}(A)$ therefore eventually equals
    $\mathbf1_{[a,\infty)}(A)$ exactly. For fixed $a,k$, the map
    $A\mapsto f_{a,k}(A)$ is continuous on the Hermitian matrices.
\end{lemma}
\begin{proof}\leanok
    \uses{def:closed_threshold_ramp,def:closed_threshold_projection}
    If $x\ge a$, the approximant is one for every $k$. If $x<a$, take
    $k\ge\lceil1/(a-x)\rceil$; then $1+k(x-a)\le0$, so it is zero.
    There are only finitely many spectral values, hence a common threshold
    for all of them. Functional calculus gives the exact matrix identity.
    The last assertion is continuity of functional calculus for the
    continuous scalar function $f_{a,k}$.
\end{proof}

\begin{lemma}[Borel closed spectral cutoff]
    \label{lem:borel_closed_spectral_cutoff}
    \lean{Matrix.measurable_spectralProjectionGE_hermitian,
        Matrix.measurable_spectralProjectionGE}
    \leanok
    For every fixed $a\in\R$, the map
    $A\mapsto\mathbf1_{[a,\infty)}(A)$ is Borel on finite Hermitian matrices.
    Its extension by zero to non-Hermitian matrices is also Borel; this is
    the convention of the matrix functional calculus used here.
    No spectral separation, fixed rank or nonempty index set is required.
\end{lemma}
\begin{proof}\leanok
    \uses{lem:closed_threshold_finite_stabilization}
    The continuous matrix approximants converge pointwise, indeed they
    stabilize at each input. Pointwise sequential limits of maps into a
    finite-dimensional matrix space are Borel. The Hermitian matrices
    form a closed set, so extension by zero preserves Borel measurability.
\end{proof}

\begin{lemma}[Borel orthogonal range projection]
    \label{lem:borel_hermitian_range_projection}
    \lean{Matrix.IsHermitian.supportProj_eq_one_sub_spectralProjectionGE_mul,
        Matrix.measurable_supportProj_hermitian}
    \leanok
    For Hermitian $A$, write $P_{\operatorname{ran}A}$ for its orthogonal
    range projection. Then
    \[
        P_{\operatorname{ran}A}
        =\Id-\mathbf1_{[0,\infty)}(A)\mathbf1_{[0,\infty)}(-A).
    \]
    Consequently $A\mapsto P_{\operatorname{ran}A}$ is Borel on the finite
    Hermitian matrices. Zero eigenvalues of arbitrary multiplicity are
    permitted, as are the zero matrix and empty index set.
\end{lemma}
\begin{proof}\leanok
    \uses{lem:borel_closed_spectral_cutoff}
    The product on the right is the projection onto the zero eigenspace:
    its scalar multiplier is one precisely when both $x\ge0$ and $-x\ge0$.
    Functional calculus proves the identity. Matrix multiplication,
    subtraction and negation preserve Borel measurability.
\end{proof}

These are matrix-analysis prerequisites for measurable collective trimming
in \cite[lines 646--655]{Arad2016RigorousRG}. They do not
construct a varying measurable orthonormal basis, a joint adaptive sampling
law or a complete randomized algorithm.


\subsection{Borel joint column supports and operator images}

\begin{definition}[Joint column support of a finite family]
    \label{def:finite_family_column_support}
    \lean{Matrix.familyColumns,Matrix.familyColumnGram,Matrix.familySupportProj}
    \leanok
    For a finite family $A_i:\C^n\to\C^m$, let $C_A$ have the columns of
    all $A_i$, indexed by pairs $(i,j)$. Put $G_A=C_AC_A^*$ and let
    $Q_A$ be its existing orthogonal support projection.
    Empty families and zero-dimensional spaces are allowed.
\end{definition}

\begin{lemma}[Borel joint column support]
    \label{lem:borel_family_column_support}
    \lean{Matrix.continuous_familyColumnGram,
        Matrix.familySupportProj_eq_one_sub_spectralProjectionGE_mul,
        Matrix.measurable_familySupportProj}
    \leanok
    The map $A\mapsto G_A$ is continuous and
    \[
        Q_A=\Id-\mathbf1_{[0,\infty)}(G_A)
                    \mathbf1_{[0,\infty)}(-G_A).
    \]
    Consequently $A\mapsto Q_A$ is Borel on all finite matrix families,
    including at rank changes. No nonzero-family hypothesis is needed.
\end{lemma}
\begin{proof}\leanok
    \uses{def:finite_family_column_support,lem:borel_closed_spectral_cutoff,
        lem:borel_hermitian_range_projection}
    The Gram map is a finite matrix polynomial in entries and their
    conjugates. Its value is positive semidefinite, so the Hermitian
    support identity applies to the actual Gram matrix. The closed cutoffs
    are Borel on the whole matrix space, and multiplication and subtraction
    preserve measurability.
\end{proof}

\begin{lemma}[The actual represented sum of ranges]
    \label{lem:family_column_support_range}
    \lean{Matrix.range_familySupportProj_eq_range_familyColumns,
        Matrix.range_familyColumns_eq_iSup_range,
        Matrix.range_familySupportProj_eq_iSup_range}
    \leanok
    The existing joint support represents exactly the sum of the actual
    family ranges:
    \[
        \operatorname{ran}Q_A=\operatorname{ran}C_A
             =\sum_i\operatorname{ran}A_i.
    \]
    This includes empty and zero families.
\end{lemma}
\begin{proof}\leanok
    \uses{def:finite_family_column_support}
    Absorption $Q_AC_A=C_A$ proves one inclusion. The existing support
    factorization gives $Q_A=G_AT=C_A(C_A^*T)$, proving the other.
    The columns of $C_A$ are precisely the union of the columns of the
    $A_i$, whose span is the sum of their ranges.
\end{proof}

\begin{lemma}[A Borel projector for a finite sum of operator images]
    \label{lem:borel_finite_operator_image_support}
    \lean{Matrix.measurable_familySupportProj_mul,
        Matrix.range_familySupportProj_mul_eq_iSup_map_range}
    \leanok
    Fix a finite family $F_i:\C^n\to\C^m$. The existing joint column
    support of the family $(F_iP)_i$ is Borel in every matrix
    $P:\C^n\to\C^n$, and its range is
    \[
        \sum_i F_i(\operatorname{ran}P).
    \]
    In particular, when $P$ is the orthogonal projection onto $S$, this
    is the actual orthogonal projector onto $\sum_iF_iS$.
\end{lemma}
\begin{proof}\leanok
    \uses{lem:borel_family_column_support,lem:family_column_support_range}
    The family $(F_iP)_i$ depends continuously on $P$. Apply the preceding
    two lemmas and use $\operatorname{ran}(F_iP)=F_i(\operatorname{ran}P)$.
\end{proof}

These statements provide the intrinsic image-space projector for the
improving family in \cite[lines 597--622]{Arad2016RigorousRG}.
They require no choice of an orthonormal basis and assert no adaptive
sampling law or running time.

\begin{lemma}[The actual finite-family image projection]
    \label{lem:actual_finite_family_image_projection}
    \lean{Matrix.range_toEuclideanLin_familySupportProj,
        Matrix.projectionMatrix_iSup_map_eq_familySupportProj,
        Matrix.measurable_projectionMatrix_iSup_map}
    \leanok
    The joint support of a finite matrix family has exactly the sum of
    its Euclidean ranges. In particular, if $F_i$ are fixed operators
    and $P_S$ is the actual orthogonal projection onto $S$, then the
    actual projection onto $\sum_iF_iS$ is the existing joint support of
    $(F_iP_S)_i$. If $P_{S(x)}$ is Borel in $x$, so is the actual
    orthogonal projection onto $\sum_iF_iS(x)$. Empty families and zero
    spaces are included.
\end{lemma}
\begin{proof}\leanok
    \uses{lem:family_column_support_range,lem:borel_finite_operator_image_support}
    The ordinary and Euclidean coordinate equivalence carries each
    actual matrix range to the corresponding Euclidean range and
    preserves sums of subspaces. The range of $F_iP_S$ is $F_iS$.
    Since the existing joint support is an orthogonal projection, it is
    the actual projection onto this sum of ranges. Apply its Borel
    dependence to the measured input projection.
\end{proof}


\begin{theorem}[Finite operator images in prescribed coordinates]
    \label{thm:family_image_prescribed_coordinates}
    \lean{Matrix.range_familySupportProj_comp_linearIsometryEquiv}
    \leanok
    Let $I:\C^d\to\C^n$ be a unitary coordinate identification, let
    $(A_i)$ be a finite operator family on $\C^n$, and put $F_i=A_iI$.
    For every matrix $P$ on $\C^d$, the joint column support of $(F_iP)_i$
    has range $\sum_i A_i(I(\operatorname{ran}P))$.
    No projection assumption on $P$ is required.
\end{theorem}
\begin{proof}\leanok
    \uses{lem:actual_finite_family_image_projection}
    Each product has range
    $\operatorname{ran}(F_iP)=A_i(I(\operatorname{ran}P))$.
    Take the sum of these ranges and apply the joint-support identity.
\end{proof}
