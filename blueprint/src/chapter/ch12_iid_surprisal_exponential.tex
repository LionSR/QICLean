\section{Centered exponential moments of tensor powers}

The scalar product identity gives an exact matrix identity for the
independent-copy step in \cite[Section~8]{OpenAI2026AreaLaw}.

\begin{theorem}[Centering inside the functional calculus]
  \label{thm:hermitian_centered_exponential_cfc}
  \lean{Matrix.IsHermitian.exp_smul_sub_smul_one_eq_cfc}
  \leanok
  Let $A$ be Hermitian on a finite-dimensional complex Hilbert space,
  and let $u,s\in\mathbb R$. Then $e^{u(A-sI)}=f(A)$ for
  $f(x)=e^{u(x-s)}$.
\end{theorem}
\begin{proof}
  \leanok
  The functional calculus preserves subtraction, multiplication by a
  real scalar, and constants. Its composition law then gives the
  stated exponential identity.
\end{proof}

\begin{theorem}[Exact centered moment of a tensor-power density]
  \label{thm:iid_centered_surprisal_exponential}
  \lean{Matrix.PosSemidef.re_trace_finKronecker_exp_centered_surprisal}
  \leanok
  Let $\rho$ be positive semidefinite on a finite-dimensional complex
  Hilbert space, with eigenvalues $p_i$, and put
  $M_p(u)=\sum_i p_i e^{-u\log p_i}$. For $k\geq0$ and $u,s\in\mathbb R$,
  \begin{align}
    \operatorname{Re}\operatorname{Tr}\!\left[
      \rho^{\otimes k}e^{u(-\log\rho^{\otimes k}-sI)}\right]
      &=e^{-us}M_p(u)^k.
  \end{align}
  The logarithm is assigned the value zero on the kernel. No trace
  normalization is assumed. Zero copies give the identity on the
  one-dimensional space of empty configurations.
\end{theorem}
\begin{proof}
  \leanok
  \uses{thm:hermitian_centered_exponential_cfc,thm:surprisal_product_moment}
  The tensor-power spectral weights are $\prod_jp_{x_j}$. Apply the
  centered scalar function to this spectral resolution. Extract the
  factor $e^{-us}$ and apply the product moment identity. Configurations
  with zero product weight make zero contribution.
\end{proof}
