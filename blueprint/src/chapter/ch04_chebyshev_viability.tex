\section{Chebyshev improvement from a common polynomial family}

Let $\mathcal H_M$ and $\mathcal H_E$ be finite-dimensional complex inner
product spaces, and let $H$ be self-adjoint on
$\mathcal H=\mathcal H_M\otimes\mathcal H_E$. Choose an orthonormal eigenbasis
$b_j$ with eigenvalues $h_j$. Suppose that $\eta_0<\eta_1\le u$, that
$h_j\le u$, and that each $h_j$ lies in $(-\infty,\eta_0]\cup[\eta_1,\infty)$.
For a nonnegative integer $k$, set
\begin{align}
    G      & =\operatorname{span}\{b_j:h_j\le\eta_0\},
    \label{eq:cheb_viable_low_space}                                   \\
    \Delta & =\bigl(2e^{-2k\sqrt{(\eta_1-\eta_0)/(u-\eta_0)}}\bigr)^2.
    \label{eq:cheb_viable_error}
\end{align}
These are the closed spectral conditions used in error reduction; positivity
of the unsquared filter is not required
\cite{gap:alvv16_spectral_projector_boundary,gap:alvv16_chebyshev_filter_positivity}.

\begin{theorem}[Chebyshev improvement from a uniform polynomial reduction]
    \label{thm:chebyshev_viability_polynomial_reduction}
    \lean{Submodule.chebyshev_viability_of_polynomial_reduction}
    \leanok
    \uses{thm:chebyshev_spectral_projector, thm:spectral_projector_contract}
    Fix a nonnegative integer $R$, a map $\Gamma$ on subspaces of
    $\mathcal H_M$, and an operator-valued map $\mathcal F$ on complex
    polynomials such that $\mathcal F(Q)=Q(H)$ for every $Q$.
    Suppose the following reduction holds for every $Q$ of degree at most
    $k$: whenever $S$ is $\delta$-viable for a subspace $T$ of $\mathcal H$,
    $\delta<1$, and $d\ge0$, and $\mathcal F(Q)$ preserves $T$ and $T^\perp$
    and satisfies, for every $t\in T$ and $z\in T^\perp$,
    \begin{align}
        \|\mathcal F(Q)t\|   & \ge\|t\|,     \\
        \|\mathcal F(Q)z\|^2 & \le d\|z\|^2,
    \end{align}
    the subspace $\Gamma(S)$ has dimension at most $R\dim S$ and is
    $d/(1-\delta)^2$-viable for $T$.
    Then, under the spectral assumptions above, for every $\delta$-viable
    subspace $S$ for $G$ with $\delta<1$, the subspace $\Gamma(S)$
    has dimension at most $R\dim S$ and is
    $\Delta/(1-\delta)^2$-viable for $G$.
    Here $\Gamma$ and $\mathcal F$ are fixed before the spectral endpoints
    and before $S$. This conditional composition uses
    Theorem~4.2 of~\cite[lines 789--853]{Arad2016RigorousRG}
    and the error-reduction argument of Lemma~2.9.
\end{theorem}

\begin{proof}\leanok
    Choose the normalized real Chebyshev polynomial $p$ of degree $k$ and
    regard it as a complex polynomial. Its evaluation is a spectral
    projector with parameter~(\ref{eq:cheb_viable_error}). The closed spectral
    gap gives invariance, expansion on $G$, and contraction on $G^\perp$.
    Apply the supplied reduction to this polynomial and these bounds.
\end{proof}

\begin{theorem}[A common matrix family gives Chebyshev improvement]
    \label{thm:matrix_family_chebyshev_viability}
    \lean{Matrix.chebyshev_viability_of_conjugated_polynomial_family}
    \leanok
    \uses{thm:viable_tensor_decomposition, thm:chebyshev_viability_polynomial_reduction}
    Let $\Sigma,X,Y$ be finite sets, put $\mathcal H_M=\mathbb C^X$ and
    $\mathcal H_E=\mathbb C^Y$, and fix a unitary linear isometry
    $T:\mathbb C^\Sigma\longrightarrow\mathcal H_M\otimes\mathcal H_E$.
    Fix a matrix $H_0$ on $\mathbb C^\Sigma$ and matrices
    $W_1,\ldots,W_N$ on $\mathbb C^X$. Suppose that for every complex
    polynomial $Q$ of degree at most $k$ there are matrices
    $M_1(Q),\ldots,M_N(Q)$ on $\mathbb C^Y$ such that
    \begin{align}
        TQ(H_0)T^{-1} & =\sum_{r=1}^N W_r\otimes M_r(Q).
        \label{eq:cheb_viable_family}
    \end{align}
    Set $H=TH_0T^{-1}$ and $\Gamma(S)=\sum_{r=1}^N W_rS$.
    If $H$ satisfies the self-adjointness and spectral assumptions stated
    above, then every $\delta$-viable $S$ for
    $G$ of~(\ref{eq:cheb_viable_low_space}), with $\delta<1$, satisfies
    \begin{align}
        \dim\Gamma(S) & \le N\dim S,
        \label{eq:cheb_viable_dimension}                                    \\
        \Gamma(S)     & \text{ is }\Delta/(1-\delta)^2\text{-viable for }G.
        \label{eq:cheb_viable_conclusion}
    \end{align}
    The family $(W_r)$ is fixed before the spectral data and before $S$;
    only the complementary matrices $M_r(Q)$ depend on $Q$.
    The supplied polynomial identity is an explicit hypothesis. This is a
    finite-dimensional consequence supporting the polynomial decomposition
    and error-reduction steps of~\cite[Proposition 5.7 and Lemma 2.9]{Arad2016RigorousRG}.
\end{theorem}

\begin{proof}\leanok
    Polynomial evaluation commutes with the matrix action on Euclidean
    space and with conjugation by $T$. Thus the filter is $Q(H)$.
    For each $Q$, apply error reduction from its tensor decomposition
    in~(\ref{eq:cheb_viable_family}). The uniform polynomial reduction
    then gives~(\ref{eq:cheb_viable_dimension}) and~(\ref{eq:cheb_viable_conclusion}).
\end{proof}

\begin{theorem}[A numerical bound for the image sum]
    \label{thm:matrix_family_chebyshev_viability_card_bound}
    \lean{Matrix.chebyshev_viability_of_conjugated_polynomial_family_of_card_le}
    \leanok
    \uses{thm:matrix_family_chebyshev_viability}
    Under the hypotheses of the common matrix family theorem, let $C$ be a
    nonnegative integer with $N\le C$. The same fixed subspace
    $\Gamma(S)$ then satisfies $\dim\Gamma(S)\le C\dim S$ and is
    $\Delta/(1-\delta)^2$-viable for $G$, for every $\delta$-viable $S$
    with $\delta<1$.
    The numerical bound $C$ is fixed with the family before the spectral
    endpoints and before $S$.
\end{theorem}

\begin{proof}\leanok
    The dimension estimate follows from
    $\dim\Gamma(S)\le N\dim S\le C\dim S$. The viable space and its error
    are unchanged.
\end{proof}

\begin{theorem}[An actual matrix cluster and a fixed polynomial family]
    \label{thm:actual_cluster_polynomial_viability}
    \lean{Matrix.chebyshev_viability_of_actual_cluster_polynomial_family}
    \leanok
    \uses{def:subspace_viable}
    Let $B$ be Hermitian on $\C^\Sigma$, with a supplied orthonormal
    eigenbasis $v_i$ and eigenvalues $\lambda_i$. Write
    $G_c=\operatorname{span}\{v_i:\lambda_i\le c\}$.
    Fix a unitary linear isometry $T:\C^\Sigma\to\C^M\otimes\C^E$ and
    matrices $W_1,\ldots,W_N$ on $\C^M$, where $N\le C$ and $k,C\in\N$.
    Suppose that for every complex polynomial $Q$ of degree at most $k$
    there are matrices $V_r(Q)$ on $\C^E$ such that
    \begin{align}
        TQ(B)T^{-1}=\sum_{r=1}^N W_r\otimes V_r(Q).
    \end{align}
    Assume $\eta_0<\eta_1\le U$, $G_{\eta_0}=G_{\eta_1}$ and $\|B\|\le U$.
    Set
    \begin{align}
        \Delta    & =\bigl(2e^{-2k\sqrt{(\eta_1-\eta_0)/(U-\eta_0)}}\bigr)^2,\notag \\
        \Gamma(S) & =\sum_{r=1}^N W_rS.
    \end{align}
    For every $\delta$-viable $S$ for $TG_{\eta_0}$ with $\delta<1$,
    $\dim\Gamma(S)\le C\dim S$ and $\Gamma(S)$ is
    $\Delta/(1-\delta)^2$-viable for $TG_{\eta_0}$.
    The same family is used for every $Q,S,\delta$. No separate
    nonnegativity assumption on $\delta$ is imposed.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:canonical_isometric_spectral_data,
        thm:matrix_family_chebyshev_viability_card_bound}
    Transfer the supplied low spectral spaces to an orthonormal spectral
    basis of $TBT^{-1}$. The endpoint equality gives the closed spectral gap, and
    the norm bound gives the upper spectral endpoint. Apply the common-family
    Chebyshev estimate and transfer its target back to $TG_{\eta_0}$.
    This is the finite-dimensional composition of the spectral filter and
    error-reduction steps of~\cite[Theorem 4.2 and Lemma 2.9]{Arad2016RigorousRG};
    the boundary convention is recorded in
    \cite{gap:alvv16_spectral_projector_boundary}.
\end{proof}

\begin{lemma}[Actual powers and ordered tensor words]
    \label{thm:spectral_operator_word_powers}
    \lean{Submodule.IsSpectralProjector.pow,
        TensorProduct.prod_mapL_ofFn,
        TensorProduct.pow_sum_mapL_eq_sum_words,
        Submodule.IsSpectralProjector.viability_word_power}
    \leanok
    \uses{thm:spectral_projector_contract,thm:viable_tensor_decomposition}
    Let $H,K$ have a common orthonormal eigenbasis, with real eigenvalues
    $h_j,k_j$. Suppose every $h_j$ is at most $\eta_0$ or at least
    $\eta_1$, that $k_j\ge1$ when $h_j\le\eta_0$, and that
    $k_j^2\le\Delta$ when $h_j\ge\eta_1$, where $\Delta\ge0$.
    Put $T=\operatorname{span}\{b_j:h_j\le\eta_0\}$. Let
    $K=\sum_{i\in I}A_i\otimes B_i$ be its actual finite tensor decomposition. For every integer $a\ge0$, put
    $A_w=A_{w_1}\cdots A_{w_a}$ and $B_w=B_{w_1}\cdots B_{w_a}$.
    Then $K^a=\sum_{w\in I^a}A_w\otimes B_w$, with the order of factors
    preserved. The same $T$ is invariant under $K^a$, low spectral
    eigenvalues are still at least one, and squared suppression is
    $\Delta^a$. For every $\delta$-viable $S$ for $T$ with $\delta<1$,
    $V_a=\sum_{w\in I^a}A_wS$ satisfies
    \begin{align}
        \dim V_a & \le |I|^a\dim S,                                              &
        V_a      & \text{ is }\frac{\Delta^a}{(1-\delta)^2}\text{-viable for }T.
        \label{eq:operator_word_power_improvement}
    \end{align}
    Neither the factors nor their summands need commute. The empty word
    and empty alphabet are included. No identity action of $K$ on $T$
    is required.
\end{lemma}
\begin{proof}\leanok
    In a common eigenbasis $Kb_j=k_jb_j$, one has $K^ab_j=k_j^ab_j$.
    Distributing the noncommutative product gives
    \begin{align}
        K^a=\left(\sum_{i\in I}A_i\otimes B_i\right)^a
        =\sum_{w\in I^a}A_w\otimes B_w.
        \label{eq:operator_word_power_expansion}
    \end{align}
    The two tensor factors retain the same order. Applying error reduction
    to (\ref{eq:operator_word_power_expansion}) gives
    (\ref{eq:operator_word_power_improvement}).
\end{proof}

\begin{lemma}[One polynomial family for every filter power]
    \label{thm:actual_cluster_chebyshev_word_powers}
    \lean{Matrix.chebyshev_word_power_viability_of_actual_cluster_polynomial_family}
    \leanok
    Under the hypotheses of Theorem~\ref{thm:actual_cluster_polynomial_viability},
    define $W_w=W_{w_1}\cdots W_{w_a}$ for $w\in\{1,\ldots,N\}^a$.
    The same target $TG_{\eta_0}$ is used for every $a\ge0$ and every
    viable input $S$. The actual image sum $V_a=\sum_w W_wS$ has
    dimension at most $C^a\dim S$ and is
    $\Delta^a/(1-\delta)^2$-viable for $TG_{\eta_0}$ whenever $S$ is
    $\delta$-viable and $\delta<1$.
    The Chebyshev polynomial and its complementary tensor factors are
    chosen before $a,S,\delta$.
\end{lemma}
\begin{proof}\leanok
    \uses{thm:canonical_isometric_spectral_data,thm:chebyshev_spectral_projector,
        thm:spectral_operator_word_powers}
    Derive the canonical spectral data from the actual Hermitian matrix.
    Choose its Chebyshev polynomial once, obtain the actual complementary
    factors from the common polynomial decomposition, and use
    (\ref{eq:operator_word_power_improvement}). The actual number of
    words is $N^a\le C^a$.
\end{proof}

\begin{lemma}[A polynomial dimension scale for a powered step]
    \label{thm:viable_node_power_parameters}
    \lean{Nat.viable_node_power_bounds, Nat.viable_node_sample_ratio,
        Real.viable_node_power_tradeoff_lt,
        Real.viable_node_sample_overlap_lower,
        Real.viable_node_power_error_bound}
    \leanok
    Let $D\ge2$ and $s\ge1$ be integers, and set
    $a=\lceil\log_D(D^2s)\rceil$, interpreted as the upper integer logarithm.
    Then $a\ge2$, $D^2s\le D^a\le D^3s$ and
    $(D^a)^2s\le D^6s^3$.
    If $\Delta\ge0$, $0<b<1$ and $D^{12}\Delta\le b$, then
    $(D^a)^{12}\Delta^a<b$.
    Moreover, suppose $1\le Q\le D^{12}s^6$ and $\delta\le1/4$. Put
    $c=(1-\delta)\min(s,Q)/(8Q)$. Then
    \begin{align}
        c                    & \ge\frac{3}{32D^{12}s^5}>0,     &
        \frac{\Delta^a}{c^2} & \le\frac{1024}{9}\frac{b}{s^2}.
        \label{eq:viable_node_power_error}
    \end{align}
    These are scalar estimates; no sampling event or recursive family is
    part of the hypotheses or conclusion.
\end{lemma}
\begin{proof}\leanok
    The defining inequalities for the upper integer logarithm give the
    two bounds on $D^a$ and the cubic dimension bound. The tradeoff is
    $(D^{12}\Delta)^a\le b^a<b$.
    In both cases $Q\le s$ and $s<Q$ one has
    $Q\le D^{12}s^5\min(s,Q)$, yielding the lower bound on $c$.
    Also $D^a\ge D^2s$, so
    $\Delta^a\le b/(D^{24}s^{12})$. Divide by the squared overlap bound
    to obtain (\ref{eq:viable_node_power_error}).
\end{proof}
