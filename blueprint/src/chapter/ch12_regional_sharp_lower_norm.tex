\section{The sharp lower norm comparison in physical coordinates}

\begin{theorem}[Explicit sharp lower norm bound]
  \label{thm:regional_sharp_lower_norm}
  \lean{Matrix.exists_replicaMetric_symProj_bandProduct_sharp_lower_norm}
  Fix nonzero finite physical site spaces, nonzero auxiliary spaces $C,R$,
  and $t>0$. There are $c,C_f\geq0$, depending only on these
  spaces and $t$, with the following property.

  Let $\Omega$ be a unit physical vector, let $\gamma>0$, and suppose
  \begin{align*}
    H-E_0I\geq\gamma(I-|\Omega\rangle\langle\Omega|).
  \end{align*}
  Fix $k>0$, an isometry $Z$ onto simultaneous copy symmetry, and
  disjoint physical regions $Q_{j,g},Y_{j,g}$ such that
  $Q_{j,g}\cup Y_{j,g}\subseteq Q_{j',h}$ whenever $g<h$.
  Suppose that $K\geq0$, $20t\leq r_0$ and $20t\leq1$.
  For each of the three regions
  $Q_{j,g}$, $Y_{j,g}$ and $(Q_{j,g}\cup Y_{j,g})^c$, suppose its
  reduced density $\rho$ has eigenvalues $(p_i)$ satisfying
  \begin{align*}
    \log\sum_{i:p_i>0}p_i^{1-u}
      \leq uS(\rho)+Ku^2 \qquad (|u|\leq r_0).
  \end{align*}
  Define
  \begin{align*}
    g_{j,g}&=S_\Omega(Q_{j,g})+
      S_\Omega((Q_{j,g}\cup Y_{j,g})^c)-S_\Omega(Y_{j,g}),\\
    g_{\max}&=\max\bigl(\{0\}\cup\{g_{j,g}:j,g\}\bigr).
  \end{align*}
  Let $T$ be one finite weighted mean tree, used for all $G$ bands.
  Suppose the nonzero vector $\xi$ is simultaneously symmetric,
  has norm at most one, and has the two original auxiliary labels
  $\lambda_C,\lambda_R$, where
  \begin{align*}
    2kS-\varepsilon\leq\log\dim\lambda_C+\log\dim\lambda_R.
  \end{align*}
  In the full original site coordinates, put
  \begin{align*}
    A_{j,g}
      &=\bigl(W_{C\cup Q_{j,g}}^{-1}
          W_{R\cup(Q_{j,g}\cup Y_{j,g})^c}^{-1}W_{Y_{j,g}}\bigr)^2,\\
    B_{j,g}&=Z^*A_{j,g}Z,\qquad
    M=\prod_{g=1}^{G}T(B_{\cdot,g}).
  \end{align*}
  For $s\geq0$, define
  \begin{align*}
    w=M^{-s}Z^*\xi,\qquad \varphi=w/\|w\|,\qquad v=Z\varphi,\qquad
    E_{\rm def}=\frac{\langle v,\overline H v\rangle-E_0}{\gamma},
  \end{align*}
  where $\overline H$ is the averaged original physical Hamiltonian
  with the two auxiliary identities.
  Then $w\ne0$, $\|v\|=1$, and $E_{\rm def}\geq0$.
  For every $0<\tau\leq1/2$, set
  \begin{align*}
    d&=\log\dim C+\log\dim R,\\
    L_{g,\tau}&=2S+(1-\tau)\sum_jw_jg_{j,g}-\tau d-
      \left(3+\frac1{2t}\right)h(\tau)-50Kt,\\
    r_k&=\log(k+1)+c\log(k+2)+2t\varepsilon,\qquad
    b_k=\frac{(k+2)^{-C_f}}2.
  \end{align*}
  Here $w_j$ denotes the terminal weight of $T$. The same vector satisfies
  \begin{align*}
    2s\sum_{g=1}^{G}\left[
      2tk\left(L_{g,\tau}-\frac{E_{\rm def}}\tau\max(0,2S+g_{\max})\right)
      -\left(2|r_k|+|\log b_k|+\log2+\log(3/2)\right)
    \right]
    \leq-\log\|w\|^2.
  \end{align*}
  The constants are chosen before the copy number, vector, labels,
  tree, regions and symmetric coordinates. This is an intermediate
  form of the sharp lower comparison in
  \cite[Section~8]{OpenAI2026AreaLaw}; the selected common initial vector
  supplies its hypotheses in the physical application.
\end{theorem}
\begin{proof}
  \uses{thm:global_regional_low_defect_sharp_lower_bound,
    thm:symmetric_replica_isometric_floor,
    thm:replica_metric_lifted_region_geometry,
    thm:replica_symmetric_common_tree_product,
    thm:regional_low_defect_product_mass,
    thm:mean_tree_exponential_lower_norm}
  Choose the uniform regional and symmetric-floor constants before
  choosing any copy or state data. The original regions form a disjoint
  partition, and nesting makes the band roots commute.
  Each compressed leaf is bounded below by $2b_kI$ and by
  $e^{2tk\ell_{j,g,\tau}-r_k}P$, where
  $\ell_{j,g,\tau}=2S+(1-\tau)g_{j,g}-\tau d-
  (3+(2t)^{-1})h(\tau)-50Kt$, and $P$ is the same compressed global
  low-defect projection. The original labels and simultaneous symmetry
  are preserved by the actual inverse power and normalization.
  Applying the original gap gives projection mass at least
  $1-E_{\rm def}/\tau$; its upper bound by one also gives
  $E_{\rm def}\geq0$.
  The explicit mean-tree lower norm estimate now applies with
  $\kappa=2tk$ and upper rate $2S+g_{\max}$.
  Since $0\leq1-\tau\leq1$ and $g_{j,g}\leq g_{\max}$ with
  $g_{\max}\geq0$, one has $(1-\tau)g_{j,g}\leq g_{\max}$.
  The remaining losses are nonnegative, so
  $\ell_{j,g,\tau}\leq2S+g_{\max}$.
  Since terminal weights sum to one, their average of
  $\ell_{j,g,\tau}$ is $L_{g,\tau}$.
  Summing over the bands gives the inequality, also when $G=0$.
\end{proof}
