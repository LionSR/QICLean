\section{A common partition of ordered-word coefficients}

\begin{theorem}[A simultaneous low-frequency partition]
    \label{thm:two_window_word_coefficient_partition}
    \lean{Fintype.exists_two_window_word_coefficient_partition}
    \leanok
    Let $I$ be a finite set, let $\mathcal R$ be a semiring, and let
    $L,R\subseteq I$ be nonempty.
    There are functions $p_L\colon\N^L\to L$ and $p_R\colon\N^R\to R$
    with the following properties. For every $\nu\colon I\to\N$ and
    $k\in\N$ with $\sum_{a\in I}\nu(a)\le k$,
    \begin{align}
        \nu\bigl(p_L(\nu|_L)\bigr) & \le\lfloor k/|L|\rfloor,
                                   & \nu\bigl(p_R(\nu|_R)\bigr) & \le\lfloor k/|R|\rfloor.\notag
    \end{align}
    For any family $H_a\in\mathcal R$ and any $k\in\N$, let
    \begin{align}
        E_k             & =\left\{\alpha\colon I\to\N:\sum_{a\in I}\alpha(a)=k\right\},
        \notag                                                                                                   \\
        C_\alpha        & =\sum_{\substack{w\colon\{0,\ldots,k-1\}\to I                                          \\
                \#w^{-1}(a)=\alpha(a)\ \text{for all }a}}
        H_{w(0)}\cdots H_{w(k-1)},\notag                                                                         \\
        u_\alpha        & =p_L(\alpha|_L),
                        & v_\alpha                                                      & =p_R(\alpha|_R),\notag \\
        \lambda(\alpha) & =\bigl((u_\alpha,\alpha(u_\alpha)),
        (v_\alpha,\alpha(v_\alpha))\bigr),\notag                                                                 \\
        B_k             & =\bigl(L\times\{0,\ldots,\lfloor k/|L|\rfloor\}\bigr)
        \times\bigl(R\times\{0,\ldots,\lfloor k/|R|\rfloor\}\bigr).
        \notag
    \end{align}
    Then
    \begin{align}
        \left(\sum_{a\in I}H_a\right)^k
         & =\sum_{t\in B_k}
        \sum_{\substack{\alpha\in E_k \\\lambda(\alpha)=t}}C_\alpha.
        \notag
    \end{align}
    The products retain their displayed order. The windows may overlap,
    and $k=0$ is allowed, with the empty product equal to $1$.
    This is the finite partition used in the
    frequency argument of \cite[Lemma~4.2]{Arad2013AreaLaw} and its
    simultaneous version in \cite[Proposition~5.7]{Arad2016RigorousRG}.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:two_window_common_cutoff_partition}
    Apply Theorem~\ref{thm:two_window_common_cutoff_partition} with $r=k$.
\end{proof}


\begin{theorem}[A common frequency cutoff for all degrees]
    \label{thm:two_window_common_cutoff_partition}
    \lean{Fintype.exists_two_window_word_coefficient_partition_of_le}
    \leanok
    Let $I$ be a finite set, let $\mathcal R$ be a semiring, and let
    $L,R\subseteq I$ be nonempty. There are selector functions
    $p_L\colon\N^L\to L$ and $p_R\colon\N^R\to R$ satisfying the
    frequency bounds of Theorem~\ref{thm:two_window_low_frequency_selectors}
    and the following simultaneous identities.
    For every family $H_a\in\mathcal R$ and integers $0\le r\le k$, put
    \begin{align*}
        E_r             & =\left\{\alpha\colon I\to\N:\sum_{a\in I}\alpha(a)=r\right\},                    \\
        C_{r,\alpha}    & =\sum_{\substack{w\colon\{0,\ldots,r-1\}\to I                                    \\
                \#w^{-1}(a)=\alpha(a)\ \text{for all }a}}
        H_{w(0)}\cdots H_{w(r-1)},                                                                         \\
        u_\alpha        & =p_L(\alpha|_L),
                        & v_\alpha                                                      & =p_R(\alpha|_R), \\
        \lambda(\alpha) & =\bigl((u_\alpha,\alpha(u_\alpha)),
        (v_\alpha,\alpha(v_\alpha))\bigr),                                                                 \\
        B_k             & =\bigl(L\times\{0,\ldots,\lfloor k/|L|\rfloor\}\bigr)
        \times\bigl(R\times\{0,\ldots,\lfloor k/|R|\rfloor\}\bigr).
    \end{align*}
    Then the same selectors and the same index set $B_k$ satisfy
    \begin{align*}
        \left(\sum_{a\in I}H_a\right)^r
         & =\sum_{t\in B_k}
        \sum_{\substack{\alpha\in E_r \\\lambda(\alpha)=t}}C_{r,\alpha}.
    \end{align*}
    The selectors precede $H$, $k$, and $r$. The windows may overlap, and
    $r=0$ is allowed. This is the common-cutoff form of the finite partition
    used in \cite[Lemma~4.2]{Arad2013AreaLaw} and
    \cite[Proposition~5.7]{Arad2016RigorousRG}.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:two_window_low_frequency_selectors,thm:finite_sum_power_ordered_words}
    Choose the selectors by the local-selection theorem. Grouping ordered
    words by their multiplicities gives
    \begin{align*}
        \left(\sum_{a\in I}H_a\right)^r=\sum_{\alpha\in E_r}C_{r,\alpha}.
    \end{align*}
    For each $\alpha\in E_r$, the identity
    $\sum_a\alpha(a)=r\le k$ gives
    $\alpha(u_\alpha)\le\lfloor k/|L|\rfloor$ and
    $\alpha(v_\alpha)\le\lfloor k/|R|\rfloor$.
    Thus $\lambda(\alpha)\in B_k$, and summing by the fibers of
    $\lambda$ proves the formula. The factors in each word retain their order.
\end{proof}

\begin{theorem}[One-window partition with a common cutoff]
    \label{thm:one_window_word_coefficient_partition}
    \lean{Fintype.exists_one_window_word_coefficient_partition_of_le}
    \leanok
    Let $I$ be finite, let $S\subseteq I$ be nonempty, and let $\mathcal R$
    be a semiring. There exists $p:\mathbb N^S\to S$, chosen before the
    operator family and degree cutoff, such that
    $\nu(p(\nu|_S))\le\lfloor\ell/|S|\rfloor$ whenever
    $\sum_a\nu(a)\le\ell$.
    For every $H_a\in\mathcal R$ and $0\le d\le\ell$, define
    $u_\beta=p(\beta|_S)$ and let $C_{d,\beta}$ be the full-frequency
    word coefficient. Then
    \begin{align*}
        \left(\sum_aH_a\right)^d
        =\sum_{u\in S}\sum_{k=0}^{\lfloor\ell/|S|\rfloor}
        \sum_{\substack{\beta:I\to\mathbb N \\\sum_a\beta(a)=d\\
                u_\beta=u,\ \beta(u)=k}} C_{d,\beta}.
    \end{align*}
    Zero degrees and cancellations between coefficient operators are allowed.
    This is the one-window partition in \cite[Lemma~4.2]{Arad2013AreaLaw}.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:two_window_low_frequency_selectors,thm:finite_sum_power_ordered_words,
        def:full_word_coefficient}
    Use the first of the two local selectors with both windows equal to $S$.
    Group the ordered words by their full occurrence vector. Each resulting
    vector has total $d\le\ell$, so its selected count lies in the displayed
    range. Group these same coefficients once more by the selected index and
    its count. Every coefficient appears in exactly one part.
\end{proof}
