\chapter{Schur Labels of Tensor Powers}\label{ch:schur_labels}

This chapter develops the label calculus for permutations of the copies of a
finite-dimensional tensor power that is used for symmetric replicas in the
two-dimensional area-law argument~\cite{OpenAI2026AreaLaw}. The labels are the
irreducible representations of the symmetric group $S_k$; we read them off
from an Artin--Wedderburn decomposition of the complex group algebra, which
treats every finite group in the same way. The identification of the labels of
$S_k$ with partitions of $k$ enters only in the dimension and branching formulas.

\section{Labels of a finite group}

\begin{definition}[Irreducible labels and central idempotents]
    \label{def:irrep_labels}
    \lean{IrrepLabel, IrrepLabel.dim, IrrepLabel.wedderburnEquiv,
        IrrepLabel.centralIdem, IrrepLabel.matrixUnit}
    \leanok
    Let $G$ be a finite group. By Maschke's theorem and the Artin--Wedderburn
    theorem over $\C$ there is an algebra isomorphism
    \[
        \C[G]\;\cong\;\prod_{\lambda}M_{d_\lambda}(\C),
    \]
    and we fix one. Its factors are the labels $\lambda$ of $G$, and
    $d_\lambda\ge1$ is the dimension of the irreducible representation with
    label $\lambda$. The central idempotent $e_\lambda\in\C[G]$ and the matrix
    units $e^\lambda_{ij}$ are the preimages of the identity and of the matrix
    units of the factor $M_{d_\lambda}(\C)$.
\end{definition}

\begin{lemma}[Central idempotents]
    \label{lem:irrep_central_idempotents}
    \lean{IrrepLabel.centralIdem_mul_centralIdem, IrrepLabel.sum_centralIdem,
        IrrepLabel.centralIdem_mul_comm, IrrepLabel.centralIdem_ne_zero,
        IrrepLabel.invStar_centralIdem}
    \leanok
    \uses{def:irrep_labels}
    The central idempotents are nonzero, central, mutually orthogonal, and sum
    to the identity. Each is fixed by the conjugate-linear involution of
    $\C[G]$ sending $c\,g$ to $\bar c\,g^{-1}$.
\end{lemma}

\begin{proof}\leanok
    The first four statements hold for the identities of the factors of a
    product of matrix algebras. For the last, let $e=e_\lambda$, let $e^\ast$
    be its image under the involution, and let $P$ be the image of $e$ in the
    left regular representation, which is faithful and sends $e^\ast$ to the
    adjoint $P^\ast$. Then $e^\ast$ is central, so $e^\ast e$ is a central
    element of the factor $\lambda$ and equals $c\,e$ for a scalar $c$. Hence
    $P^\ast P=cP$. If $c=0$ then $P=0$, which is impossible. Taking adjoints
    gives $P^\ast=uP$ with $u=c/\bar c$, and idempotence of $P^\ast$ forces
    $u=1$. Thus $P^\ast=P$, and faithfulness gives $e^\ast=e$.
\end{proof}

\begin{definition}[Permutation operators and label projectors]
    \label{def:label_projectors}
    \lean{PermutationRepresentation.permOp,
        PermutationRepresentation.groupAlgebraRep,
        PermutationRepresentation.labelProj,
        PermutationRepresentation.labelObservable,
        PermutationRepresentation.labelEntropy}
    \leanok
    \uses{def:irrep_labels}
    Let $G$ act on a finite set $X$. The permutation operator $U(g)$ on $\C^X$
    sends the basis vector $e_x$ to $e_{gx}$, and extends linearly to an algebra
    homomorphism $\C[G]\to\mathrm{End}(\C^X)$. The label projector $\pi^\lambda$
    is the image of $e_\lambda$. For a real function $f$ of the labels, the
    label observable is $\sum_\lambda f(\lambda)\pi^\lambda$; for
    $f(\lambda)=\log d_\lambda$ it is the observable
    $F=\sum_\lambda(\log d_\lambda)\pi^\lambda$
    of~\cite[Section~6.1]{OpenAI2026AreaLaw}.
\end{definition}

\begin{lemma}[Label projectors are a resolution of the identity]
    \label{lem:label_projectors}
    \lean{PermutationRepresentation.labelProj_mul_labelProj,
        PermutationRepresentation.sum_labelProj,
        PermutationRepresentation.isHermitian_labelProj,
        PermutationRepresentation.commute_labelProj_of_forall_commute,
        PermutationRepresentation.isHermitian_labelObservable}
    \leanok
    \uses{def:label_projectors, lem:irrep_central_idempotents}
    The label projectors are mutually orthogonal orthogonal projections with sum
    the identity, and they commute with every operator that commutes with all
    permutation operators. Every label observable is self-adjoint.
\end{lemma}

\begin{proof}\leanok
    Apply the homomorphism to the relations of
    Lemma~\ref{lem:irrep_central_idempotents}. The image of the involution of
    $\C[G]$ is the adjoint, since $U(g)^\ast=U(g^{-1})$, so each projector is
    self-adjoint. An operator commuting with every $U(g)$ commutes with the
    image of every element of $\C[G]$.
\end{proof}

\section{Copies of a tensor power}

Let $\mathcal V=\bigotimes_{f\in F}\C^{n_f}$ be a finite-dimensional space
with a specified finite tensor factorization, and let $k\ge0$. For a subsystem
$Q\subseteq F$ and $\pi\in S_k$, the operator $U_Q(\pi)$ on
$\mathcal V^{\otimes k}$ permutes the $k$ copies of the factors in $Q$ and acts
as the identity on the other factors~\cite[Section~6]{OpenAI2026AreaLaw}.
Thus $U(\pi)=U_F(\pi)$ permutes entire copies, and the symmetric subspace
$\mathcal S_k=\operatorname{Sym}^k\mathcal V$ is the subspace fixed by every $U(\pi)$. The
Schur-label projectors $\pi^\lambda_{Q,k}$ are the label projectors of the
action $\pi\mapsto U_Q(\pi)$.

\begin{definition}[Copy permutations and the symmetric subspace]
    \label{def:copy_permutations}
    \lean{TensorPower.copyPerm, TensorPower.subsystemPerm,
        TensorPower.symmetricSubspace, PermutationRepresentation.symProj}
    \leanok
    \uses{def:label_projectors}
    The copy permutations of a subsystem, the symmetric subspace, and the
    averaging projector $\Pi_k=(k!)^{-1}\sum_{\pi\in S_k}U(\pi)$ onto it.
\end{definition}

\begin{lemma}[Central functions of $S_k$ are invariant under inversion]
    \label{lem:symmetric_central_inverse}
    \lean{IrrepLabel.coeff_conj_of_mem_center, IrrepLabel.coeff_inv_of_mem_center}
    \leanok
    The coefficients of a central element of $\C[G]$ form a class function.
    For $G=S_k$, a central element has the same coefficient at $\pi$ and at
    $\pi^{-1}$.
\end{lemma}

\begin{proof}\leanok
    Compare the coefficients of $h\,a$ and $a\,h$ at $hg$. A permutation and
    its inverse have the same cycle type, hence are conjugate.
\end{proof}

\section{The Schur-label estimates}

The following entries formalize the six parts of the Schur-label
lemma~\cite[Lemma 6.1]{OpenAI2026AreaLaw}.

\begin{theorem}[Schur-label estimates, part 2: commutation and complements]
    \label{thm:schur_label_commutation}
    \lean{TensorPower.commute_groupAlgebraRep_subsystemPerm_of_subset,
        TensorPower.commute_groupAlgebraRep_subsystemPerm_of_disjoint,
        TensorPower.commute_labelProj_subsystemPerm_of_subset,
        TensorPower.commute_labelProj_subsystemPerm_of_disjoint,
        TensorPower.groupAlgebraRep_mulVec_compl_of_mem_symmetricSubspace,
        TensorPower.labelProj_mulVec_compl_of_mem_symmetricSubspace,
        TensorPower.labelProj_mul_symProj_compl}
    \leanok
    \uses{def:copy_permutations, lem:label_projectors}
    Central label functions commute for nested or disjoint subsystems: if
    $Q\subseteq Q'$, a central function of the copy permutations of $Q$
    commutes with every operator generated by the copy permutations of $Q'$,
    and operators generated by the copy permutations of disjoint subsystems
    commute. On $\mathcal S_k$, complementary label projectors agree:
    $\pi^\lambda_{Q,k}\Pi_k=\pi^\lambda_{Q^c,k}\Pi_k$, even when $Q$ and $Q^c$
    have different dimensions.
\end{theorem}

\begin{proof}\leanok
    \uses{lem:symmetric_central_inverse}
    For $Q\subseteq Q'$ we have $U_{Q'}(\pi)=U_Q(\pi)U_{Q'\setminus Q}(\pi)$,
    where the second factor commutes with every operator generated by $Q$ and
    the first commutes with every central function of $Q$. Disjoint
    commutation is immediate. On $\mathcal S_k$,
    $U_Q(\pi)U_{Q^c}(\pi)z=z$, so $U_Q(\pi)z=U_{Q^c}(\pi^{-1})z$. Writing a
    central element as $\sum_\pi a(\pi)\pi$ with $a(\pi)=a(\pi^{-1})$ and
    reindexing by $\pi\mapsto\pi^{-1}$ shows that it acts in the same way
    through $Q$ and through $Q^c$ on $z$.
\end{proof}

\section{Isotypic multiplicities and the surprisal bound}

\begin{lemma}[Orthogonal resolutions]
    \label{lem:orthogonal_resolution}
    \lean{Matrix.IsOrthogonalResolution, Matrix.IsOrthogonalResolution.hom,
        Matrix.IsOrthogonalResolution.exp_hom, Matrix.IsOrthogonalResolution.prod,
        Matrix.IsOrthogonalResolution.posSemidef_hom_of_ne_zero,
        Matrix.IsHermitian.spectralProj, Matrix.IsHermitian.cfc_eq_hom}
    \leanok
    Let $(P_i)$ be mutually orthogonal idempotents with $\sum_iP_i=1$. Then
    $f\mapsto\sum_if(i)P_i$ is an algebra homomorphism from functions on the
    index set, and $\exp\bigl(\sum_if(i)P_i\bigr)=\sum_ie^{f(i)}P_i$. Two
    commuting resolutions give the product resolution $P_iQ_j$. If the $P_i$
    are self-adjoint and $f(i)\ge0$ whenever $P_i\ne0$, the image of $f$ is
    positive semidefinite. The spectral projections of a self-adjoint matrix
    form a resolution, and its functional calculus is the image of the
    restriction of the function to the eigenvalues.
\end{lemma}

\begin{proof}\leanok
    Multiplicativity follows from the orthogonality relations. The
    homomorphism is continuous, so it commutes with the exponential series,
    which is computed pointwise. The remaining statements are direct.
\end{proof}

\begin{lemma}[Invariant subspaces of an isotypic component]
    \label{lem:isotypic_dimension}
    \lean{PermutationRepresentation.matrixUnitOp,
        PermutationRepresentation.dim_le_finrank_of_invariant,
        PermutationRepresentation.multiplicity,
        PermutationRepresentation.trace_labelProj}
    \leanok
    \uses{lem:label_projectors}
    A nonzero subspace of the range of $\pi^\lambda$ invariant under all
    permutation operators has dimension at least $d_\lambda$. Moreover
    $\tr\pi^\lambda=d_\lambda m_\lambda$, where $m_\lambda$ is the rank of the
    image of the matrix unit $e^\lambda_{11}$, the multiplicity of $\lambda$.
\end{lemma}

\begin{proof}\leanok
    Let $E_{ij}$ be the images of the matrix units. For $w\ne0$ in the
    subspace, $w=\sum_iE_{ii}w$, so some $E_{ii}w\ne0$. The vectors $E_{ji}w$
    lie in the subspace and are linearly independent: applying $E_{ik}$ to a
    vanishing combination $\sum_jc_jE_{ji}w$ gives $c_kE_{ii}w=0$. For the
    trace, $\tr E_{ii}=\tr E_{i1}E_{1i}=\tr E_{1i}E_{i1}=\tr E_{11}$, and the
    trace of an idempotent is its rank.
\end{proof}

\begin{lemma}[Polynomial multiplicity count]
    \label{lem:multiplicity_count}
    \lean{PermutationRepresentation.commutant,
        PermutationRepresentation.sum_multiplicity_le_finrank_commutant,
        PermutationRepresentation.finrank_commutant_le_card_pairOrbits,
        TensorPower.card_pairOrbits_le, TensorPower.sum_multiplicity_le,
        TensorPower.card_labelProj_ne_zero_le}
    \leanok
    \uses{lem:isotypic_dimension}
    For any permutation action, $\sum_\lambda m_\lambda$ is at most the
    dimension of the commutant of the permutation operators, which is at most
    the number of orbits of pairs. For the copy permutations of
    $(\C^q)^{\otimes k}$ this number is at most $(k+1)^{q^2}$. Hence the total
    multiplicity, and the number of labels occurring, are at most
    $(k+1)^{q^2}$.
\end{lemma}

\begin{proof}\leanok
    For each label choose a basis $(y_a)$ of the range of $E_{11}$ and a
    nonzero row $r$ with $rE_{11}=r$. The operators
    $\sum_jE_{j1}\,y_ar^{\mathsf T}E_{1j}$ lie in the commutant, and
    compressing a vanishing combination by $E^\mu_{11}$ on both sides shows
    that they are linearly independent across all labels. A commuting operator
    has constant entries on orbits of pairs. An orbit of a pair of
    configurations is determined by the number of copies carrying each pair of
    basis labels, a function from $q^2$ pairs to $\{0,\dots,k\}$.
\end{proof}

\begin{theorem}[Schur-label estimates, part 5: surprisal]
    \label{thm:schur_label_surprisal}
    \lean{TensorPower.schurSurprisal,
        PermutationRepresentation.mul_dim_le_one_of_ne_zero,
        PermutationRepresentation.commute_labelEntropy,
        PermutationRepresentation.posSemidef_labelEntropy,
        PermutationRepresentation.supportProj_mul_labelEntropy_mul_supportProj_le,
        PermutationRepresentation.re_trace_mul_exp_le_sum_multiplicity}
    \leanok
    \uses{lem:orthogonal_resolution, lem:multiplicity_count}
    Let $\rho$ be any permutation-invariant density matrix on $Q^{\otimes k}$,
    $q=d_Q$, and $L_\rho=-\log\rho$ on its support. Then $F_Q$ commutes with
    $\rho$, $0\le F_Q\le L_\rho$ on the support of $\rho$, and for
    $0\le b\le1$,
    \[
        \Tr\rho\exp\bigl(b(L_\rho-F_Q)\bigr)\le(k+1)^{q^2}.
    \]
\end{theorem}

\begin{proof}\leanok
    The spectral projections $E_u$ of $\rho$ commute with the label
    projectors, so the products $E_u\pi^\lambda$ form an orthogonal resolution
    and all operators involved are functions of it. The range of a nonzero
    $E_u\pi^\lambda$ is an invariant subspace of the $\lambda$-isotypic
    component, so its rank $n_{u,\lambda}$ is at least $d_\lambda$; since
    $\sum_{u,\lambda}u\,n_{u,\lambda}=\tr\rho=1$, this gives $ud_\lambda\le1$,
    so $\log d_\lambda\le-\log u$ there. The trace equals
    $\sum_{u,\lambda}(ud_\lambda)^{1-b}n_{u,\lambda}/d_\lambda
    \le\sum_\lambda\tr\pi^\lambda/d_\lambda=\sum_\lambda m_\lambda$, and
    Lemma~\ref{lem:multiplicity_count} bounds the last sum.
\end{proof}

\section{Merging subsystems and grouping copies}

\begin{lemma}[Irreducible and joint blocks]
    \label{lem:joint_blocks}
    \lean{PermutationRepresentation.permOp_mul_matrixUnitOp,
        PermutationRepresentation.matrixUnitOp_mul_permOp,
        PermutationRepresentation.blockSpan, PermutationRepresentation.pairBlockSpan,
        PermutationRepresentation.mul_dim_le_finrank_of_invariant}
    \leanok
    \uses{lem:isotypic_dimension}
    For one action, the span of the vectors $E^\lambda_{j1}v$ is invariant and
    has dimension at most $d_\lambda$. For two commuting actions with labels
    $\alpha,\beta$ and matrix units $A_{ij},B_{ab}$, the span of the vectors
    $A_{p1}B_{q1}z$ is invariant under both and has dimension at most
    $d_\alpha d_\beta$, and every nonzero subspace of the joint isotypic
    component invariant under both actions has dimension at least
    $d_\alpha d_\beta$.
\end{lemma}

\begin{proof}\leanok
    A permutation operator acts on $E^\lambda_{jo}$ from the left by
    $\sum_pc_{pj}E^\lambda_{po}$ and on $E^\lambda_{oj}$ from the right by
    $\sum_qc_{jq}E^\lambda_{oq}$. The lower bound repeats the proof of
    Lemma~\ref{lem:isotypic_dimension} with the products $A_{ji}B_{ba}$.
\end{proof}

\begin{theorem}[Schur-label estimates, part 4: grouped copies]
    \label{thm:schur_label_groups}
    \lean{TensorPower.groupedCopies_dim, TensorPower.groupedCopies_commute,
        TensorPower.index_range_youngHom,
        PermutationRepresentation.mul_dim_le_dim_of_compatible,
        PermutationRepresentation.dim_le_index_mul_of_compatible}
    \leanok
    \uses{lem:joint_blocks}
    Split the $k$ copies into specified groups of sizes $k-r$ and $r$. If
    their labels $\alpha,\beta$ on a fixed subsystem are compatible with the
    full label $\lambda$, then
    $d_\alpha d_\beta\le d_\lambda\le\binom{k}{r}d_\alpha d_\beta$. The group
    label projectors commute with the full label projector.
\end{theorem}

\begin{proof}\leanok
    Compatibility gives an irreducible block $B$ of $\lambda$ whose image $Y$
    under $\pi^\alpha\pi^\beta$ is nonzero. $Y$ is invariant under both groups
    and lies in the joint isotypic component, so
    $d_\alpha d_\beta\le\dim Y\le\dim B\le d_\lambda$. For the upper bound
    choose a joint block $Z\subseteq Y$ of dimension at most
    $d_\alpha d_\beta$. The translates of $Z$ by coset representatives of the
    Young subgroup span a nonzero invariant subspace of the $\lambda$-isotypic
    component, of dimension at least $d_\lambda$ and at most
    $\binom{k}{r}d_\alpha d_\beta$. Central elements of $S_k$ commute with
    its subgroup algebras.
\end{proof}

\begin{theorem}[Schur-label estimates, part 3: merge dimensions]
    \label{thm:schur_label_merge_dimensions}
    \lean{TensorPower.merge_dim_le,
        PermutationRepresentation.dim_le_mul_dim_of_compatible,
        PermutationRepresentation.dim_le_mul_dim_of_compatible'}
    \leanok
    \uses{lem:joint_blocks, thm:schur_label_commutation}
    If $Q,E$ are disjoint and their labels $\lambda,\mu$ are compatible with
    the label $\nu$ on $QE$, then $d_\nu\le d_\lambda d_\mu$ and
    $d_\lambda\le d_\nu d_\mu$.
\end{theorem}

\begin{proof}\leanok
    A joint block of $\lambda,\mu$ meeting the $\nu$-isotypic component is
    invariant under $U_{QE}(\pi)=U_Q(\pi)U_E(\pi)$ and has dimension at most
    $d_\lambda d_\mu$; its image under $\pi^\nu$ has dimension at least
    $d_\nu$. For the second inequality, let $S_k$ act on matrices by
    $M\mapsto U_{QE}(\pi)MU_E(\pi)^{-1}$; with $P=\pi^\lambda\pi^\mu\pi^\nu$
    this sends $P$ to $PU_Q(\pi)$, so the $\lambda$-projection of $P$ is $P$.
    Writing $P=\sum_{j,b}E^\nu_{jo}NE^\mu_{ob}$ with
    $N=E^\nu_{oj}PE^\mu_{bo}$, some span of the matrices
    $E^\nu_{j'o}NE^\mu_{ob'}$ is invariant, of dimension at most
    $d_\nu d_\mu$, and has nonzero $\lambda$-projection, of dimension at least
    $d_\lambda$. This replaces the duality argument of the source.
\end{proof}

\begin{theorem}[Schur-label estimates, part 3: merge moment]
    \label{thm:schur_label_merge_moment}
    \lean{TensorPower.merge_conditional_le, TensorPower.merge_moment_le,
        PermutationRepresentation.trace_mul_trace_eq,
        PermutationRepresentation.merge_conditional_le,
        PermutationRepresentation.merge_moment_le_sum_multiplicity}
    \leanok
    \uses{thm:schur_label_merge_dimensions, lem:multiplicity_count}
    Let $\rho$ be a density matrix on $(Q\otimes E)^{\otimes k}$ invariant
    under the copy permutations of $Q$ and of $E$, for instance a tensor
    product of permutation-invariant density matrices, and $q=d_Qd_E$. The
    conditional probability of $\nu$ given $\lambda,\mu$ is at most
    $(k+1)^{q^2}d_\nu/(d_\lambda d_\mu)$, and for $0\le b\le1$,
    \[
        \E\left(\frac{d_\lambda d_\mu}{d_\nu}\right)^b\le(k+1)^{q^2}.
    \]
\end{theorem}

\begin{proof}\leanok
    With $P=\pi^\lambda\pi^\mu$, the joint matrix units give
    $\Tr(\rho Z)\tr P=\Tr(\rho P)\tr Z$ for $Z$ in their span, in particular
    for $Z=\pi^\nu P$. Since $\tr P\ge d_\lambda d_\mu$ when $P\ne0$ and
    $\tr(\pi^\nu P)\le\tr\pi^\nu=d_\nu m_\nu$, the conditional probability is
    at most $m_\nu d_\nu/(d_\lambda d_\mu)$. When it is nonzero the labels are
    compatible, so $R=d_\lambda d_\mu/d_\nu\ge1$ and $R^b\le R$. Summing,
    the moment is at most $\sum_\nu m_\nu\le(k+1)^{q^2}$.
\end{proof}

\section{The dimension formulas}

The first part of the Schur-label lemma identifies the labels of $S_k$
occurring in $(\C^q)^{\otimes k}$ with the partitions $\lambda\vdash k$ of
length at most $q$ and states the dimension formulas
\[
    \dim V^{(q)}_\lambda=\prod_{i<j}\frac{l_i-l_j}{j-i},\qquad
    d_\lambda=\frac{k!\prod_{i<j}(l_i-l_j)}{\prod_il_i!},\qquad
    l_i=\lambda_i+q-i.
\]
The identification of the labels with partitions is proved in the section on
Schur--Weyl duality below. The following lemmas concern the right-hand sides of
these formulas.

\begin{lemma}[Stirling bounds]
    \label{lem:stirling_log_factorial}
    \lean{Real.log_factorial_sub_nonneg, Real.log_factorial_sub_le,
        Real.mul_log_add_sub_bounds}
    \leanok
    For every $n\ge0$,
    $0\le\log n!-(n\log n-n)\le\log(n+2)+1$. For natural numbers $a,s$,
    $0\le(a+s)\log(a+s)-a\log a\le s(\log(a+s)+1)$.
\end{lemma}

\begin{proof}\leanok
    The lower bound is Stirling's inequality
    $n!\ge\sqrt{2\pi n}(n/e)^n$. The upper bound follows because the Stirling
    sequence $n!/(\sqrt{2n}(n/e)^n)$ decreases and equals $e/\sqrt2$ at
    $n=1$. For the second statement use $\log(1+s/a)\le s/a$.
\end{proof}

\begin{lemma}[Entropy asymptotics of the dimension formula]
    \label{lem:hook_formula_entropy}
    \lean{Partition.hookFormula, Partition.entropyTerm,
        Partition.abs_log_hookFormula_sub_le,
        Partition.abs_log_hookFormula_sub_le_mul_log,
        Partition.weylFormula, Partition.weylFormula_le}
    \leanok
    \uses{lem:stirling_log_factorial}
    Let $\lambda\vdash k$ have length at most $q$, and let $D_\lambda$ be the
    right-hand side of the formula for $d_\lambda$. Then
    \[
        |\log D_\lambda-kH(\lambda/k)|\le(q^2+q+1)\bigl(\log(k+q+2)+1\bigr),
    \]
    so $\log D_\lambda=kH(\lambda/k)+O_q(\log(k+2))$. The right-hand side of the
    formula for $\dim V^{(q)}_\lambda$ is at most $(k+q)^{q^2}$.
\end{lemma}

\begin{proof}\leanok
    Write $\log D_\lambda=\log k!+\sum_{i<j}\log(l_i-l_j)-\sum_i\log l_i!$ and
    $kH(\lambda/k)=k\log k-\sum_i\lambda_i\log\lambda_i$. The difference is the
    sum of the Stirling errors, the Vandermonde term, the shifts
    $\sum_i(l_i-\lambda_i)\le q^2$, and $\sum_i(\lambda_i\log\lambda_i-l_i\log l_i)$;
    each is bounded by Lemma~\ref{lem:stirling_log_factorial} using
    $1\le l_i-l_j\le l_i\le k+q$.
\end{proof}

\section{Schur--Weyl duality and highest weights}

For a matrix $A$ on $\C^q$ write $A^{(j)}$ for $A$ acting on the $j$-th copy of
$(\C^q)^{\otimes k}$ and $\Delta(A)=\sum_jA^{(j)}$ for the diagonal action of the
general linear Lie algebra. The matrix units give the operators
$E_{ab}=\Delta(e_{ab})$. The identification of the labels with partitions rests
on the half of Schur--Weyl duality proved below, cited in the area-law paper
from the textbook of Etingof and coauthors.

\begin{theorem}[Schur--Weyl duality]
    \label{thm:schur_weyl_commutant}
    \lean{TensorPower.tensorPow_mem_diagAlgebra, TensorPower.symTensor_eq_sum,
        TensorPower.commutant_le_diagAlgebra, TensorPower.commutant_eq_diagAlgebra}
    \leanok
    The operators on $(\C^q)^{\otimes k}$ commuting with the copy permutations
    form the algebra generated by the operators $\Delta(A)$.
\end{theorem}

\begin{proof}\leanok
    The operators $\Delta(A)$ commute with the copy permutations. Conversely,
    $A^{\otimes k}=A^{(1)}\cdots A^{(k)}$ is the top elementary symmetric
    polynomial of the commuting family $A^{(j)}$, hence by Newton's identities a
    polynomial in the power sums $\sum_j(A^{(j)})^m=\Delta(A^m)$. Polarization,
    \[
        \sum_{\sigma\in S_k}B_{\sigma(1)}\otimes\cdots\otimes B_{\sigma(k)}
        =\sum_{S\subseteq\{1,\dots,k\}}(-1)^{k-|S|}
            \Bigl(\sum_{i\in S}B_i\Bigr)^{\otimes k},
    \]
    follows by expanding the right side and using
    $\sum_{S\supseteq T}(-1)^{k-|S|}=[T=\{1,\dots,k\}]$. Finally an operator
    commuting with the copy permutations is the average of its conjugates,
    a combination of symmetrizations of tensor products of matrix units.
\end{proof}

A configuration $x$ of the $k$ copies has weight
$\mathrm{wt}(x)_a=\#\{j:x_j=a\}$, and $E_{ab}$ maps vectors of weight $\mu$ to
vectors of weight $\mu+e_a-e_b$. A highest-weight vector is a nonzero weight
vector annihilated by the raising operators $E_{ab}$, $a<b$. Weights are
compared lexicographically.

\begin{theorem}[Highest weights]
    \label{thm:highest_weight}
    \lean{TensorPower.exists_isHighestWeight,
        TensorPower.IsIrreducible.mem_span_of_isWeightVector,
        TensorPower.IsIrreducible.toLex_le,
        TensorPower.IsIrreducible.weight_eq_of_isHighestWeight,
        TensorPower.IsIrreducible.antitone_of_isHighestWeight}
    \leanok
    \uses{thm:schur_weyl_commutant}
    Every nonzero subspace of $(\C^q)^{\otimes k}$ invariant under the commutant
    of the copy permutations contains a highest-weight vector. If the subspace
    is irreducible and contains a highest-weight vector $v$ of weight $\mu$,
    then its vectors of weight $\mu$ are the multiples of $v$, all its weights
    are lexicographically at most $\mu$, every highest-weight vector in it has
    weight $\mu$, and $\mu$ is weakly decreasing.
\end{theorem}

\begin{proof}\leanok
    A weight vector of lexicographically maximal weight is annihilated by the
    raising operators. For irreducibility, the span $L$ of the vectors
    $E_{a_1b_1}\cdots E_{a_rb_r}v$ with all $a_i>b_i$ is invariant under every
    $E_{ab}$, by induction on $r$ using
    $[E_{ab},E_{cd}]=\delta_{bc}E_{ad}-\delta_{ad}E_{cb}$; by
    Theorem~\ref{thm:schur_weyl_commutant} it is invariant under the commutant,
    hence equal to the subspace. Each nonempty lowering product strictly
    lowers the weight. Permuting the basis of $\C^q$ on every copy commutes with
    the copy permutations and permutes the weights; if $\mu_a<\mu_b$ for some
    $a<b$, the transposition of $a$ and $b$ would produce a weight above $\mu$.
\end{proof}

\section{The labels of the symmetric group are partitions}

\begin{lemma}[Counting labels]
    \label{lem:count_labels}
    \lean{IrrepLabel.card_eq_card_conjClasses, Equiv.Perm.nat_card_conjClasses,
        Partition.card_padded_self}
    \leanok
    The number of labels of a finite group is its number of conjugacy classes.
    The conjugacy classes of $S_k$ correspond to the partitions of $k$, which
    correspond to the weakly decreasing vectors of $k$ nonnegative integers
    summing to $k$.
\end{lemma}

\begin{proof}\leanok
    The centre of $\C[G]$ is the algebra of tuples of scalars of the
    Artin--Wedderburn decomposition, and also the space of class functions. Two
    permutations are conjugate exactly when they have the same cycle type, and
    every partition is a cycle type. A partition $p$ gives the decreasing vector
    $i\mapsto\#\{\text{parts}>i\}$ of its conjugate, and a decreasing vector gives
    the partition formed by its nonzero entries; both maps are injective, since a
    decreasing vector takes values at least $v$ on an initial segment.
\end{proof}

\begin{lemma}[Column antisymmetrization]
    \label{lem:column_antisymmetrizer}
    \lean{TensorPower.isHighestWeight_columnVector}
    \leanok
    \uses{thm:highest_weight}
    Let $\mu$ be a partition of $k$ padded to length $q$. Place the copies of
    $(\C^q)^{\otimes k}$ in the boxes of the Young diagram of $\mu$, let $x$ assign
    to each copy the row of its box, and let $C_\mu$ be the group of permutations
    of the copies preserving every column. Then
    $\sum_{\sigma\in C_\mu}\operatorname{sgn}(\sigma)U(\sigma)e_x$ is a
    highest-weight vector of weight $\mu$.
\end{lemma}

\begin{proof}\leanok
    Every term has weight $\mu$, and the coefficient of $e_x$ is one because a
    column-preserving permutation fixing $x$ also preserves rows. For $a<b$ the
    operator $E_{ab}$ commutes with the antisymmetrization and changes the value
    of a copy in row $b$ to $a$; since $\mu_a\ge\mu_b$ the same column has a copy
    in row $a$, and the transposition of these two copies fixes the new
    configuration and reverses the sign, so the antisymmetrization vanishes.
\end{proof}

For a label $\lambda$ of $S_k$ the multiplicity space $V_\lambda$ in
$(\C^q)^{\otimes k}$ is the range of the image of a diagonal matrix unit of the
$\lambda$ factor. It is invariant under the commutant of the copy permutations
and, when nonzero, irreducible; its highest weight is the \emph{shape} of
$\lambda$.

\begin{theorem}[Labels of the symmetric group]
    \label{thm:labels_are_partitions}
    \lean{TensorPower.isIrreducible_multSpace, TensorPower.labelShape,
        TensorPower.labelShape_bijective}
    \leanok
    \uses{thm:highest_weight, lem:count_labels, lem:column_antisymmetrizer}
    The shape of a label of $S_k$ in $(\C^k)^{\otimes k}$ is a partition of $k$,
    and this is a bijection between the labels of $S_k$ and the partitions
    of $k$.
\end{theorem}

\begin{proof}\leanok
    The orbit of a configuration with distinct entries carries the regular
    representation, so every label occurs. A highest-weight vector of weight
    $\mu$ with a nonzero component of label $\lambda$ yields, through a matrix
    unit, a highest-weight vector of weight $\mu$ in $V_\lambda$, so $\mu$ is the
    shape of $\lambda$. Applied to Lemma~\ref{lem:column_antisymmetrizer}, every
    partition is a shape. Lemma~\ref{lem:count_labels} gives equal cardinalities.
\end{proof}

\begin{theorem}[Labels occurring in a tensor power]
    \label{thm:schur_weyl_labels}
    \lean{TensorPower.multSpace_ne_bot_iff, TensorPower.shape_eq_labelShape,
        TensorPower.labelProj_ne_zero_iff}
    \leanok
    \uses{thm:labels_are_partitions}
    The label projector $\pi^\lambda$ on $(\C^q)^{\otimes k}$ is nonzero exactly
    when the partition $\lambda$ has at most $q$ rows, and then the highest
    weight of $V^{(q)}_\lambda$ is $\lambda$.
\end{theorem}

\begin{proof}\leanok
    For $q\le q'$ the inclusion $\C^q\subseteq\C^{q'}$ induces an embedding of
    tensor powers commuting with the copy permutations and mapping highest-weight
    vectors of weight $\mu$ to highest-weight vectors of weight $\mu$ padded with
    zeros. Compare with $q=k$, using Lemma~\ref{lem:column_antisymmetrizer} for
    the converse.
\end{proof}
