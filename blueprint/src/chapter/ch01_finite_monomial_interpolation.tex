\section{Finite monomial interpolation}

\begin{theorem}[Evaluation recovery of monomial coefficients]
    \label{thm:finite_monomial_interpolation}
    \lean{MvPolynomial.exists_monomial_evaluation_recovery,
        MvPolynomial.monomial_evaluation_recovery_smul}\leanok
    Let $\sigma$ be a set of variables and let $S$ be a finite set of distinct
    monomials. Write $t=|S|$ and $m_\alpha(z)=\prod_i z_i^{\alpha_i}$,
    with only finitely many nonzero exponents for each $\alpha\in S$.
    There exist points $z_0,\ldots,z_{t-1}\in\mathbb C^\sigma$ and complex
    coefficients $c_{\alpha r}$, depending only on $S$, such that
    \begin{align}
        \sum_{r=0}^{t-1}c_{\alpha r}m_\beta(z_r)
         & =\begin{cases}1,&\alpha=\beta,\\0,&\alpha\ne\beta.\end{cases}\notag
    \end{align}
    For every complex vector space $V$ and family $(f_\beta)_{\beta\in S}$ in $V$,
    put
    \begin{align*}
        P_f(z)=\sum_{\beta\in S}m_\beta(z)f_\beta.
    \end{align*}
    The same points and coefficients satisfy
    \begin{align}
        f_\alpha & =\sum_{r=0}^{t-1}c_{\alpha r}P_f(z_r).\notag
    \end{align}
    Empty $S$ and dependent or zero coefficient vectors are allowed.
    This is the interpolation step in \cite[Lemma~4.2]{Arad2013AreaLaw}.
    The spanning-family interpretation and a separate correction to the
    subsequent word count are recorded in
    \cite{gap:aklv13_interpolation_and_word_count}.
\end{theorem}
\begin{proof}\leanok
    Scalar polynomial evaluation is injective over the infinite field
    $\mathbb C$. The monomial basis therefore gives linear independence of
    the functions $z\mapsto m_\beta(z)$.
    Consequently the evaluation vectors
    $E_z=(m_\beta(z))_{\beta\in S}$ span $\mathbb C^S$.
    Choose $t$ of them spanning this $t$-dimensional space.
    Expressing each coordinate vector as a linear combination of those
    evaluations gives the coefficients $c_{\alpha r}$.
    For vector coefficients, distribute scalar multiplication and interchange
    the two finite sums; the scalar recovery identity leaves precisely $f_\alpha$.
\end{proof}


\begin{theorem}[Rank of a selected coefficient sum]
    \label{thm:monomial_evaluation_selected_rank}
    \lean{Matrix.operatorSchmidtRank_sum_le_of_monomial_evaluations}
    \leanok
    Let $S$ be a finite monomial support, let $f_\alpha$ be operator coefficients
    on a tensor product of two finite-dimensional complex spaces, and put
    $P_f(z)=\sum_{\alpha\in S}m_\alpha(z)f_\alpha$.
    If $\operatorname{OSR}(P_f(z))\le r$ for every $z\in\mathbb C^\sigma$,
    where $r\ge0$ is an integer, then every $U\subseteq S$ satisfies
    \begin{align*}
        \operatorname{OSR}\!\left(\sum_{\alpha\in U}f_\alpha\right)\le |S|r.
    \end{align*}
    Empty supports, empty subsets, and dependent coefficient operators are allowed.
\end{theorem}
\begin{proof}
    \leanok
    \uses{thm:finite_monomial_interpolation, thm:operator_schmidt_rank_algebra}
    Choose the $t=|S|$ recovery points and coefficients from $S$ before
    choosing the operator family or the selected subset. Summing the recovery
    formulas over $U$ gives
    \begin{align*}
        \sum_{\alpha\in U}f_\alpha
        =\sum_{j=0}^{t-1}\left(\sum_{\alpha\in U}c_{\alpha j}\right)P_f(z_j).
    \end{align*}
    Scalar multiplication does not increase operator Schmidt rank, and the
    rank of a sum is at most the sum of the ranks. The right side therefore
    has rank at most $tr$.
\end{proof}

For a rectangular family of monomials, the evaluation points in
Theorem~\ref{thm:finite_monomial_interpolation} may be prescribed on an integer
product grid. This gives an explicit choice for the scalar interpolation in
\cite[Lemma~4.2]{Arad2013AreaLaw} and
\cite[Proposition~5.7]{Arad2016RigorousRG}.

\begin{definition}[Integer interpolation matrices]
    \label{def:integer_grid_matrices}
    \lean{FiniteScalarInterpolation.integerNode,
        FiniteScalarInterpolation.integerRecovery,
        FiniteScalarInterpolation.integerEvaluationGrid,
        FiniteScalarInterpolation.integerRecoveryGrid}\leanok
    Let $O$ be a finite set, let $K\in\mathbb N$, and put
    $S=\{0,\ldots,K\}$ and $G=S^O$. Set $x_r=r\in\mathbb C$ for $r\in S$.
    Let $V^{(0)}=(x_r^j)_{r,j\in S}$ be the Vandermonde matrix and put
    $C^{(0)}=(V^{(0)})^{-1}$. Define matrices indexed by $G$ by
    \begin{align*}
        V_{g\beta}   & =\prod_{a\in O}x_{g_a}^{\beta_a},      \\
        C_{\alpha g} & =\prod_{a\in O}C^{(0)}_{\alpha_a,g_a}.
    \end{align*}
\end{definition}

\begin{theorem}[Recovery on an integer product grid]
    \label{thm:integer_grid_recovery}
    \lean{FiniteScalarInterpolation.integerRecovery_sum,
        FiniteScalarInterpolation.integerRecoveryGrid_mul_integerEvaluationGrid,
        FiniteScalarInterpolation.integerRecoveryGrid_smul}\leanok
    \uses{def:integer_grid_matrices}
    With the notation of Definition~\ref{def:integer_grid_matrices},
    \begin{align}
        C^{(0)}V^{(0)} & =I_S,\notag                          \\
        CV             & =I_G.\label{eq:integer_grid_inverse}
    \end{align}
    Let $M$ be a complex module and let $(f_\beta)_{\beta\in G}$ be any family
    in $M$. Put $F(g)=\sum_{\beta\in G}V_{g\beta}f_\beta$.
    For every $\alpha\in G$,
    \begin{align*}
        \sum_{g\in G}C_{\alpha g}F(g) & =f_\alpha.
    \end{align*}
    The points and recovery matrices depend only on $O$ and $K$.
    These formulas include empty $O$ and $K=0$.
\end{theorem}
\begin{proof}\leanok
    \uses{def:integer_grid_matrices}
    Distinct integer nodes give a nonzero Vandermonde determinant, hence
    $C^{(0)}V^{(0)}=I_S$. Distributing products over finite sums gives
    \begin{align*}
        (CV)_{\alpha\beta}
         & =\sum_{g\in G}\prod_{a\in O}
        C^{(0)}_{\alpha_a,g_a}x_{g_a}^{\beta_a}
        =\prod_{a\in O}\sum_{r\in S}
        C^{(0)}_{\alpha_a,r}x_r^{\beta_a}           \\
         & =\prod_{a\in O}\delta_{\alpha_a,\beta_a}
        =\delta_{\alpha,\beta}.
    \end{align*}
    By (\ref{eq:integer_grid_inverse}), distributivity and interchange of the
    two finite sums yield
    \begin{align*}
        \sum_{g\in G}C_{\alpha g}F(g)
         & =\sum_{\beta\in G}(CV)_{\alpha\beta}f_\beta
        =f_\alpha.
    \end{align*}
\end{proof}
