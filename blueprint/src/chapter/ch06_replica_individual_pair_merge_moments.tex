\section{The two individual paired-copy moments}
\label{sec:replica_individual_pair_merge_moments}

\begin{theorem}[Separate moments on the common pair marginal]
\label{thm:replica_individual_pair_merge_moments}
\lean{Matrix.replicaGoodPairMarginal_exp_mergeDeficits_le}
\leanok
Let the one-copy physical vector $\Omega$ have norm one, and let $u$
be fixed by simultaneous permutations of the physical and two auxiliary
copy registers. For a set $B$ of bad copies, put $m=|B^c|$ and
$w=(R_B\otimes I)u$. Let $\rho_{\mathrm{pair}}$ be the marginal of
$|w\rangle\langle w|$ on the good $QC$ and $VR$ coordinates.
Write $d_Q,d_V,d_C,d_R$ for the one-copy dimensions and $D_{QC},D_{VR}$
for the two lifted merge deficits. For every real $a\leq1$,
\begin{align}
 \operatorname{Re}\operatorname{Tr}(\rho_{\mathrm{pair}}e^{aD_{QC}})
 &\leq (m+1)^{(d_Qd_C)^2}\|w\|^2,\\
 \operatorname{Re}\operatorname{Tr}(\rho_{\mathrm{pair}}e^{aD_{VR}})
 &\leq (m+1)^{(d_Vd_R)^2}\|w\|^2.
\end{align}
The component may vanish. These are the separate bounds before their
combination in \cite[Section~8]{OpenAI2026AreaLaw}.
\end{theorem}
\begin{proof}\leanok
\uses{thm:replica_component_merge_moment,thm:pair_merge_deficit_traces,thm:replica_good_pair_marginal_right,thm:replica_good_pair_marginal_left}
Partial trace reduces the first expectation to the actual $QC$
marginal. For the second, exchange the two exterior physical regions
and their auxiliary registers, preserving the middle region. This
exchange preserves the original ground norm, simultaneous symmetry,
and component norm, so the same marginal estimate applies.
\end{proof}
