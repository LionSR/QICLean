\section{Uniform floors for commuting band products}

\begin{theorem}[Uniform floor for mean trees of commuting band products]
    \label{thm:replica_symmetric_product_mean_floor}
    \lean{TensorPower.ReplicaTransport.exists_meanTree_symBandMetric_prod_floor}
    \leanok
    \uses{lem:replica_symmetric_metric, def:mean_tree}
    Fix a finite tensor product $\mathcal V=\bigotimes_{v\in V}\C^{n_v}$,
    with $n_v>0$, and let $t>0$. There exists $C\ge0$ such that the
    following holds for every $K,k\ge0$, every label set $J$, and every
    family of partitions $\pi_{j,g}$ indexed by $j\in J$ and $0\le g<K$.
    Put $A_{j,g}=\widetilde A_k(\pi_{j,g})$, using the identity-extended
    band metric of Theorem~\ref{thm:replica_symmetric_band_floor}.
    Suppose that, for each $j$, the matrices $A_{j,g}$ commute at distinct
    positions $g$. Then every nonempty finite rooted weighted binary tree
    $\mathcal T$ with terminal labels in $J$ satisfies
    \begin{align}
        (k+2)^{-CK}I
        \le \mathcal T\bigl(A_{j,0}\cdots A_{j,K-1}:j\bigr).
    \end{align}
    The exponent $C$ is chosen before $K,k,J$, the partitions, the tree,
    and its interpolation parameters. The label set need not be finite.
    Empty site sets, $K=0$, $k=0$, repeated labels, and interpolation
    parameters at either endpoint are included; the empty product is $I$.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:replica_symmetric_band_floor,
        thm:commuting_list_scalar_bound, thm:mean_tree_properties}
    Choose $C$ from Theorem~\ref{thm:replica_symmetric_band_floor} and
    set $b=(k+2)^{-C}>0$. The commuting-product bound gives
    $b^KI\le A_{j,0}\cdots A_{j,K-1}$ for each $j$.
    Since $b^K=(k+2)^{-CK}$, monotonicity and the constant-input identity
    for the mean tree give
    \begin{align}
        b^KI=\mathcal T(b^KI:j)
        \le\mathcal T\bigl(A_{j,0}\cdots A_{j,K-1}:j\bigr).
    \end{align}
\end{proof}

\section{Filtered-norm bounds}

\begin{theorem}[Squared filtered norm under a scalar metric floor]
    \label{thm:scalar_floor_filtered_norm}
    \lean{Matrix.Transport.filteredNormSq_le_of_scalar_floor}
    \leanok
    Let $M$ be a positive definite complex matrix on a finite-dimensional
    space, which may have dimension zero. Suppose that $b>0$ and $bI\le M$.
    For any vector $\xi$, set $N^2=\|M^{-1/4}\xi\|^2$. Then
    \begin{align}
        N^2\le b^{-1/2}\|\xi\|^2.
    \end{align}
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:scalar_floor_nonpositive_power, thm:quadratic_form_order}
    Since $M$ is positive definite, spectral functional calculus gives
    $N^2=\operatorname{Re}(\xi^*M^{-1/2}\xi)$.
    Theorem~\ref{thm:scalar_floor_nonpositive_power} gives
    $M^{-1/2}\le b^{-1/2}I$. The quadratic-form order then gives the
    stated bound, using $\operatorname{Re}(\xi^*\xi)=\|\xi\|^2$.
\end{proof}

\begin{theorem}[Logarithmic filtered norm under a polynomial metric floor]
    \label{thm:polynomial_floor_filtered_log}
    \lean{Matrix.Transport.log_filteredNormSq_le_of_rpow_floor}
    \leanok
    Let $M$ be positive definite, let $C\ge0$ and $r\ge1$, and suppose
    that $r^{-C}I\le M$. For any vector $\xi$ with $\|\xi\|\le1$, set
    $N^2=\|M^{-1/4}\xi\|^2$. Then
    $\log N^2\le (C/2)\log r$.
    Here the real logarithm is assigned the value zero at zero.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:scalar_floor_filtered_norm}
    If $\xi=0$, then $N=0$, and the right-hand side is nonnegative.
    Otherwise $N>0$, since $M^{-1/4}$ is invertible. The squared-norm
    bound gives $N^2\le(r^{-C})^{-1/2}\|\xi\|^2\le r^{C/2}$.
    Taking logarithms and using $\log(r^{C/2})=(C/2)\log r$ gives the result.
\end{proof}


\section{Floors for transport roots}

\begin{theorem}[Uniform floor for transport roots]
	\label{thm:replica_transport_root_floor}
	\lean{TensorPower.ReplicaTransport.exists_rootPath_scalar_floor}
	\leanok
	\uses{lem:replica_symmetric_metric, def:mean_tree}
	Fix $\mathcal V=\bigotimes_{v\in V}\C^{n_v}$ with $n_v>0$ and $t>0$.
	There exists $C_0\ge0$ such that the following holds for every
	admissible transport round with $K\ge0$ bands and every copy count $k\ge0$.
	The old parts form partitions, each move transfers a subsystem of its
	middle part, and the history and conditional choice weights are positive.
	Suppose that the identity-extended band metrics at distinct band positions
	commute for every pair of terminal labels. Write $\pi_{j,g}$ for the
	partition at terminal label $j$ and band $g$, and set
	$A_j=\prod_{g=0}^{K-1}\widetilde A_k(\pi_{j,g})$.
	For $q\in\R$, let $\bar q=\max(0,\min(1,q))$ and let
	$\mathcal T_{\bar q}$ be the interpolated terminal tree of the given round.
	Its root $M(q)=\mathcal T_{\bar q}(A_j:j)$ satisfies
	\begin{align}
		(k+2)^{-C_0K}I\le M(q).
	\end{align}
	The constant is chosen before the number of bands, history and choice
	sets, transport round, copy count and parameter. The history and choice
	sets are not assumed finite.
\end{theorem}

\begin{proof}
	\leanok
	\uses{thm:replica_symmetric_product_mean_floor}
	The validity of the moves ensures that every terminal triple is a
	partition. The commutation assumption, applied with the same terminal
	label on both sides, gives pairwise commutation within each product
	$A_j$. Apply Theorem~\ref{thm:replica_symmetric_product_mean_floor} to
	these partitions and the given tree $\mathcal T_{\bar q}$.
\end{proof}


\begin{theorem}[Logarithmic normalization bound for transport roots]
	\label{thm:replica_transport_root_log_norm}
	\lean{TensorPower.ReplicaTransport.exists_rootPath_log_filteredNormSq_bound}
	\leanok
	Fix the physical dimensions and $t>0$ as in
	Theorem~\ref{thm:replica_transport_root_floor}. There exists $C_0\ge0$
	such that every admissible transport round satisfying the same band
	commutation assumption obeys the following bound. For its root $M(q)$,
	with any $K,k\ge0$, $q\in\R$, and $\xi$ with $\|\xi\|\le1$,
	\begin{align}
		\log\|M(q)^{-1/4}\xi\|^2\le\frac{C_0K}{2}\log(k+2).
	\end{align}
	The constant is chosen before the history and choice sets, transport
	round, numbers of bands and copies, interpolation parameter and vector.
	The real logarithm is assigned the value zero at zero.
\end{theorem}

\begin{proof}
	\leanok
	\uses{thm:replica_transport_root_floor, thm:polynomial_floor_filtered_log,
		thm:mean_tree_properties}
	Choose $C_0$ from Theorem~\ref{thm:replica_transport_root_floor}.
	Every terminal input is positive definite, and so is its mean-tree
	root $M(q)$. Since $C_0K\ge0$ and $k+2\ge1$, the scalar floor
	$(k+2)^{-C_0K}I\le M(q)$ satisfies the hypotheses of
	Theorem~\ref{thm:polynomial_floor_filtered_log}, which gives the bound.
\end{proof}
