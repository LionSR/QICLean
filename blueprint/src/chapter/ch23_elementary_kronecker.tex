\section{Finite tensor products and tensor powers of matrices}

All matrices in this section are complex and indexed by finite sets.

\begin{definition}[Finite tensor product of matrices]
    \label{def:elementary_finite_tensor_product}
    \lean{Matrix.piKronecker}
    \leanok
    For a finite set of sites $\nu$ and matrices $K_k$ with rows indexed by
    $I_k$ and columns by $J_k$, the tensor product $\bigotimes_kK_k$ is the
    matrix with rows indexed by $\prod_kI_k$, columns by $\prod_kJ_k$, and
    entries
    \begin{align}
        \Bigl(\bigotimes_kK_k\Bigr)_{x,y}
         & =\prod_k(K_k)_{x_k,y_k}.
        \notag
    \end{align}
    For a matrix $V$ with rows indexed by $I$ and columns by $J$, the
    $N$-fold tensor power $V^{\otimes N}$ is the tensor product of the
    constant family $V$ over $N$ sites, indexed by $I^N$ and $J^N$. For a
    system of sites with on-site matrices $m_v$, the product operator
    $\bigotimes_vm_v$ is the tensor product over the sites.
\end{definition}

\begin{lemma}[Algebra of finite tensor products]
    \label{lem:elementary_finite_tensor_product_algebra}
    \lean{Matrix.piKronecker_apply, Matrix.piKronecker_mul, Matrix.piKronecker_one,
        Matrix.piKronecker_smul, Matrix.sum_piKronecker,
        Matrix.conjTranspose_piKronecker, Matrix.transpose_piKronecker,
        Matrix.trace_piKronecker, Matrix.piKronecker_diagonal,
        Matrix.piKronecker_eq_zero_iff, Matrix.piKronecker_ne_zero}
    \leanok
    \uses{def:elementary_finite_tensor_product}
    Finite tensor products of rectangular matrices are multiplicative,
    $\bigl(\bigotimes_kK_k\bigr)\bigl(\bigotimes_kL_k\bigr)
        =\bigotimes_kK_kL_k$, and the tensor product of identities is the
    identity. Multiplying every factor by $c$ multiplies the product by
    $c^{\lvert\nu\rvert}$, so $(cX)^{\otimes N}=c^NX^{\otimes N}$. Transposes
    and adjoints are taken factorwise, and the trace of a tensor product of
    square matrices is the product of the traces. Summing tensor products over
    all choice functions gives the tensor product of the sums, and the tensor
    product of diagonal matrices $\operatorname{diag}(f_k)$ is diagonal with
    entries $\prod_kf_k(\sigma_k)$. A tensor product vanishes exactly when one
    of its factors does.
\end{lemma}

\begin{proof}
    \leanok
    Distributing finite products over sums gives
    \begin{align}
        \sum_{r}\prod_k(K_k)_{x_k,r_k}(L_k)_{r_k,y_k}
         & =\prod_k\sum_{j}(K_k)_{x_k,j}(L_k)_{j,y_k}.
        \notag
    \end{align}
    The remaining identities are entrywise computations with
    $\prod_k\delta_{x_k,y_k}=\delta_{x,y}$. If every factor is nonzero, the
    entry at a choice of nonzero entries of the factors is a product of
    nonzero numbers.
\end{proof}

\begin{lemma}[A scalar at one site]
    \label{lem:elementary_finite_tensor_scalar_site}
    \lean{Matrix.piKronecker_update_smul, Matrix.piKronecker_ite_smul_one}
    \leanok
    \uses{lem:elementary_finite_tensor_product_algebra}
    Multiplying the factor at one site by $c$ multiplies the tensor product
    by $c$. In particular, a finite tensor product of identities in which the
    factor at one site is replaced by $c\Id$ equals $c\Id$.
\end{lemma}

\begin{proof}
    \leanok
    The scalar factors out of the product of the entries.
\end{proof}

\begin{lemma}[Tensor products of maps on labels]
    \label{lem:elementary_tensor_power_label_map}
    \lean{Matrix.piKronecker_mulVec_prod, Matrix.piKronecker_permMatrix_mulVec,
        Matrix.piKronecker_basisInclusion,
        Matrix.piKronecker_mul_apply_of_eq_ite,
        Matrix.mul_piKronecker_apply_of_eq_ite}
    \leanok
    \uses{def:elementary_finite_tensor_product}
    A tensor product applied to a product vector $\bigotimes_kw_k$ gives the
    product vector $\bigotimes_kK_kw_k$. The tensor product of the permutation
    matrices of permutations $e_k$ of the labels acts on vectors by
    $w\mapsto w\circ e$, where $e$ applies $e_k$ at every site, and the tensor
    product of the inclusions of basis vectors along injections $e_k$ is the
    inclusion along the product injection. If $u_{ij}=\delta_{j,f(i)}$ for a
    map $f$ on labels, then left multiplication by $u^{\otimes N}$ sends the
    row configuration $\sigma$ of a matrix to $f\circ\sigma$. If
    $u_{ij}=\delta_{i,g(j)}$, right multiplication by $u^{\otimes N}$ sends
    the column configuration $\tau$ to $g\circ\tau$.
\end{lemma}

\begin{proof}
    \leanok
    The entries of these tensor products are products of Kronecker deltas, so
    only one intermediate configuration contributes.
\end{proof}

\begin{lemma}[Trace of an ordered product of linear combinations]
    \label{lem:elementary_trace_product_sums}
    \lean{Matrix.trace_prod_ofFn_sum_smul}
    \leanok
    \uses{lem:elementary_ordered_product_sum}
    For scalars $f_l(j)$ and square matrices $B_l(j)$,
    \begin{align}
        \operatorname{tr}\prod_{l=0}^{L-1}\sum_jf_l(j)B_l(j)
         & =\sum_{c}\Bigl(\prod_lf_l(c(l))\Bigr)
        \operatorname{tr}\prod_{l=0}^{L-1}B_l(c(l)),
        \notag
    \end{align}
    the sum running over all choice functions $c$.
\end{lemma}

\begin{proof}
    \leanok
    Distribute the ordered product over the sums, extract the scalars, and
    use linearity of the trace.
\end{proof}

\begin{lemma}[Tensor products of isometries and partial isometries]
    \label{lem:elementary_tensor_power_partial_isometry}
    \lean{Matrix.IsIsometry.piKronecker, Matrix.IsCoisometry.piKronecker,
        Matrix.IsUnitaryBetween.piKronecker,
        Matrix.piKronecker_mem_unitaryGroup,
        Matrix.sum_star_piKronecker_mulVec_mul,
        Matrix.sum_norm_sq_piKronecker_mulVec_le}
    \leanok
    \uses{lem:elementary_finite_tensor_product_algebra}
    A tensor product of isometries is an isometry, a tensor product of
    coisometries is a coisometry, and a tensor product of unitary matrices is
    unitary; in particular, if $E^\dagger E=\Id$, then $T_L=E^{\otimes L}$
    satisfies $T_L^\dagger T_L=\Id$ for every $L\ge0$, as for the inclusions
    in \cite[Section~II.F.2]{Schuch2011Phases}. For matrices $W_k,W'_k$ with
    the same rows and vectors $\psi,\varphi$,
    $\braket{(\bigotimes_kW_k)\psi}{(\bigotimes_kW'_k)\varphi}
        =\braket{\psi}{(\bigotimes_kW_k^\dagger W'_k)\varphi}$. If every
    $W_k^\dagger W_k$ is idempotent, then
    $\lVert(\bigotimes_kW_k)\psi\rVert\leq\lVert\psi\rVert$. This is the step
    used for the isometry $W^{\otimes M}$ in
    \cite[eq.~(19)]{Malz2024Preparation}.
\end{lemma}

\begin{proof}
    \leanok
    Multiplication and adjoints are taken factorwise, so
    $(\bigotimes_kE_k)^\dagger\bigotimes_kE_k=\bigotimes_kE_k^\dagger E_k=\Id$
    for isometries, and the inner product identity is
    $(\bigotimes_kW_k)^\dagger\bigotimes_kW'_k=\bigotimes_kW_k^\dagger W'_k$.
    If every $W_k^\dagger W_k$ is idempotent, then
    $\Pi=\bigotimes_kW_k^\dagger W_k$ is a Hermitian idempotent, so
    $\Id-\Pi\geq0$ and
    $\lVert\psi\rVert^2-\lVert(\bigotimes_kW_k)\psi\rVert^2
        =\braket{\psi}{(\Id-\Pi)\psi}\geq0$.
\end{proof}

\begin{lemma}[Products and powers of finite tensor products]
    \label{lem:elementary_finite_tensor_product_powers}
    \lean{Matrix.piKroneckerMonoidHom, Matrix.list_prod_map_piKronecker,
        Matrix.piKronecker_smul_one, Matrix.IsHermitian.piKronecker,
        Matrix.transpose_piKronecker_map_star, Matrix.piKronecker_const_isometry,
        Matrix.piKronecker_const_mul_conjTranspose, Matrix.reindex_piKronecker_two,
        Matrix.reindex_piKronecker_three}
    \leanok
    \uses{lem:elementary_finite_tensor_product_algebra}
    On square matrices, the finite tensor product is a homomorphism of monoids
    from families to matrices; hence the ordered product of the tensor products
    of finitely many families is the tensor product of the sitewise ordered
    products. The tensor product of the scalar matrices $f_k\Id$ is
    $\bigl(\prod_kf_k\bigr)\Id$, a tensor product of Hermitian matrices is
    Hermitian, and the transpose of the tensor product of the entrywise
    conjugates $\overline{K_k}$ is the adjoint of $\bigotimes_kK_k$. For one
    matrix $V$ and $N\ge0$, $V^\dagger V=\Id$ implies
    $(V^{\otimes N})^\dagger V^{\otimes N}=\Id$, and
    $V^{\otimes N}(V^{\otimes N})^\dagger=(VV^\dagger)^{\otimes N}$. On two
    and three sites, $V^{\otimes2}$ and $V^{\otimes3}$ are $V\otimes V$ and
    $V\otimes(V\otimes V)$ after the configurations are relabelled as pairs
    and right-associated triples.
\end{lemma}

\begin{proof}
    \leanok
    Multiplicativity and the identity are
    Lemma~\ref{lem:elementary_finite_tensor_product_algebra}; induct over the
    list. The remaining identities follow from the factorwise adjoint and
    transpose and from the entries $\prod_k(K_k)_{x_k,y_k}$.
\end{proof}

\begin{lemma}[Inner products after matrices on the sites, in coordinates]
    \label{lem:elementary_tensor_power_coordinates}
    \lean{Matrix.sum_star_mul_prod_eq_sum_conjTranspose_mul,
        Matrix.sum_star_mul_prod_of_support, Matrix.IsIsometry.sum_star_mul_prod,
        Matrix.IsIsometry.sum_star_mul_prod_const}
    \leanok
    \uses{lem:elementary_tensor_power_partial_isometry}
    Let $W_k$ be matrices with columns indexed by a finite set $\kappa$ and rows
    by $n_k$, one for each of $M$ sites, and let $\psi,\varphi$ be vectors
    indexed by $\kappa^M$. Then
    \begin{align}
        \Bigl\langle\Bigl(\bigotimes_kW_k\Bigr)\psi,\Bigl(\bigotimes_kW_k\Bigr)\varphi\Bigr\rangle
         & =\sum_{\tau,\tau'}\Bigl(\prod_k(W_k^\dagger W_k)_{\tau_k\tau'_k}\Bigr)
        \overline{\psi(\tau)}\varphi(\tau').
        \notag
    \end{align}
    If $W_k^\dagger W_k$ is the projection onto the coordinates of a set
    $S_k\subseteq\kappa$ and $\overline{\psi(\tau)}\varphi(\tau)=0$ whenever
    some $\tau_k\notin S_k$, the inner product is
    $\langle\psi,\varphi\rangle$; in particular isometries $W_k$, and the power
    $W^{\otimes M}$ of one isometry, preserve inner products. These are the
    isometries and partial isometries on the blocks
    in~\cite[Supplemental Material, proof of Lemma~1]{Malz2024Preparation}.
\end{lemma}

\begin{proof}
    \leanok
    Expand both sums, exchange the order of summation, and sum the product
    $\prod_k\overline{(W_k)_{s_k\tau_k}}(W_k)_{s_k\tau'_k}$ over the
    configurations $s$; it factorizes into the entries of the Gram matrices
    $W_k^\dagger W_k$. For a projection onto the coordinates of $S_k$, only
    $\tau=\tau'$ with every $\tau_k\in S_k$ contributes.
\end{proof}
