\section{Positive regulators and spectral tails}
\label{sec:patch_regulator}

Let $I$ be a finite index set. Matrix inequalities refer to the order defined
by positive semidefiniteness. The following estimates give the operator and
trace consequences used in \cite[Section~4, regulator estimate]{OpenAI2026PolynomialPEPS}.

% The first three estimates take the displayed order, rank, and tail
% hypotheses. The stationarity theorem below derives the commutation and the
% order from minimality of the ordered patch objective. Proposition 4.1 of the
% source manuscript is not asserted.

\begin{theorem}[Projection trace estimate]
\label{thm:patch_projection_trace}
\lean{Matrix.trace_le_smul_rank_add_complement_of_le}
\leanok
Let $Q\in\C^{I\times I}$ be an orthogonal projection, $b\in\R$, and
$T,\rho\in\C^{I\times I}$. If $T\le b\Id$ and $T\le\rho$, then
\begin{align}
 \tr T\le b\rank Q+\tr((\Id-Q)\rho).
\end{align}
No commutation with $Q$ is required.
\end{theorem}
\begin{proof}
\leanok
Positivity of the two trace pairings gives
\begin{align}
 0&\le\tr(Q(b\Id-T))=b\rank Q-\tr(QT),\notag\\
 0&\le\tr((\Id-Q)(\rho-T))
   =\tr((\Id-Q)\rho)-\tr T+\tr(QT).
\end{align}
Adding these inequalities proves the claim.
\end{proof}

\begin{theorem}[Bounds on a shifted regulator]
\label{thm:patch_regulator_order}
\lean{Matrix.PosSemidef.shiftedRegulator_bounds}
\leanok
Suppose $x,\rho\in\C^{I\times I}$ are positive semidefinite and commute.
Let $b>0$ and assume $\rho\le x+b\Id$. Then the matrix
$T=b\rho(x+b\Id)^{-1}$ satisfies $0\le T\le b\Id$ and $T\le\rho$.
\end{theorem}
\begin{proof}
\leanok
\uses{thm:positive_identity_shift}
Set $K=x+b\Id$. The matrices $K$ and $K^{-1}$ are positive definite,
and $\rho$ and $x$ commute with $K^{-1}$. Thus $T\ge0$, and
\begin{align}
 b\Id-T&=b(K-\rho)K^{-1}\ge0,\notag\\
 \rho-T&=\rho xK^{-1}\ge0.
\end{align}
Each product on the right consists of commuting positive matrices.
\end{proof}

\begin{theorem}[A uniform small-regulator bound]
\label{thm:patch_regulator_quarter}
\lean{Matrix.PosSemidef.trace_shiftedRegulator_le_quarter}
\leanok
Let $x,\rho\in\C^{I\times I}$ be commuting positive semidefinite matrices,
$Q$ an orthogonal projection, and $C,u,R\in\R$. Assume $u\ge0$,
$R\ge(C+\log8)(1+u)$, $\rho\le x+e^{-R}\Id$,
$\rank Q\le e^{C(1+u)}$, and $\tr((\Id-Q)\rho)\le1/8$.
Then $T=e^{-R}\rho(x+e^{-R}\Id)^{-1}$ satisfies $\tr T\le1/4$.
\end{theorem}
\begin{proof}
\leanok
\uses{thm:patch_projection_trace,thm:patch_regulator_order}
The preceding two theorems give
\begin{align}
 \tr T\le e^{-R}\rank Q+\tr((\Id-Q)\rho)
         \le e^{-R+C(1+u)}+\frac18.
\end{align}
Since $\log8\ge0$ and $u\ge0$, the radius assumption implies
$-R+C(1+u)\le-\log8$. The exponential term is therefore at most $1/8$.
\end{proof}

\begin{theorem}[Stationarity of the ordered patch objective]
\label{thm:patch_stationarity}
\lean{Entropy.patch_stationarity, Entropy.patch_shiftedRegulator_bounds}
\leanok
\uses{lem:filter_stationarity,lem:commuting_hermitian,thm:patch_regulator_order}
Let $X_1\subseteq\cdots\subseteq X_m$ be regions of a finite system, let
$a_1,\dots,a_m>0$ and $b>0$, and let $\Omega$ be a vector. For density
matrices $x_j$ on $X_j$ put $L_j=(x_j+b\Id)^{-a_j/2}$, extended by the
identity, and $M=L_m\cdots L_1$. Suppose $(x_1,\dots,x_m)$ minimizes
$\lVert M\Omega\rVert$ among all such tuples of density matrices. Let
$\rho_j$ be the state on $X_j$ of the normalized vector $\phi=M\Omega/\lVert
M\Omega\rVert$, with $\phi=0$ when $M\Omega=0$. Then for every $j$
\begin{align}
 x_j\rho_j=\rho_jx_j,\qquad \rho_j\le x_j+b\Id .
\end{align}
Consequently the regulator $b\rho_j(x_j+b\Id)^{-1}$ satisfies the bounds of
Theorem~\ref{thm:patch_regulator_order}. This is the descending stationarity
step in the proof of \cite[Proposition~4.1]{OpenAI2026PolynomialPEPS}.
\end{theorem}
\begin{proof}
\leanok
Proceed from the largest region inward. When the filters $L_k$ with $k>j$
already commute with $\rho_k$, conjugating $x_j$ by a unitary keeps it a
density matrix, and Lemma~\ref{lem:filter_stationarity} applied to the minimum
shows that $L_j$, and hence $x_j$, commutes with $\rho_j$.

For the order, set $G=\rho_j(x_j+b\Id)^{-1}$, which is positive and commutes
with $x_j$. For a density matrix $y$ commuting with $x_j$, the segment
$x_j+s(y-x_j)$ stays feasible for $0\le s\le1$. Its one-sided derivative at
$s=0$, after the outer filters are removed by the descending trace argument, is
$-\tfrac{a_j}{2}\operatorname{Re}\Tr(G(y-x_j))$, so minimality gives
$\operatorname{Re}\Tr(Gy)\le\lambda:=\operatorname{Re}\Tr(Gx_j)$. Testing
$y$ on rank-one projections of a common eigenbasis of $G$ and $x_j$
(Lemma~\ref{lem:commuting_hermitian}) gives $G\le\lambda\Id$. Since
$\lambda=\Tr\rho_j-b\Tr G\le1$, we get $\rho_j\le\lambda(x_j+b\Id)\le x_j+b\Id$.
\end{proof}

\begin{theorem}[First-order inequalities for normalized patch outputs]
\label{thm:patch_normalized_first_variation}
\lean{Entropy.re_trace_variation_le_of_isPatchMinimizer}
\leanok
Let $X_0\subseteq\cdots\subseteq X_{m-1}$ be regions of a finite tensor-product
system, let $a_0,\dots,a_{m-1}>0$ and $b>0$, and let $\Omega$ be a vector.
For density matrices $x_j$ on $X_j$, put $L_j=(x_j+b\Id)^{-a_j/2}$,
extended by the identity outside $X_j$, and $M=L_{m-1}\cdots L_0$.
Suppose that $(x_j)_{j=0}^{m-1}$ minimizes $\lVert M\Omega\rVert$ over all
such tuples of density matrices, with the regions and exponents fixed.
Set $\phi=M\Omega/\lVert M\Omega\rVert$, with $\phi=0$ when $M\Omega=0$,
and let $\rho_j$ be its reduced matrix on $X_j$.
Then, for every $j$ and every density matrix $y$ on $X_j$ commuting with $x_j$,
\begin{align}
 \operatorname{Re}\tr\bigl(\rho_j(x_j+b\Id)^{-1}y\bigr)
 &\le \operatorname{Re}\tr\bigl(\rho_j(x_j+b\Id)^{-1}x_j\bigr).
 \label{eq:patch_normalized_variation}
\end{align}
Repeated regions and the empty product are allowed.
This is the first-order inequality used in
\cite[Section~4, first-order patch inequalities]{OpenAI2026PolynomialPEPS}.
\end{theorem}
\begin{proof}
\leanok
\uses{lem:filter_stationarity}
Write $\Psi=M\Omega$, and let $\sigma_j$ be the reduced matrix of
$\ketbra{\Psi}{\Psi}$ on $X_j$. The descending stationarity argument gives
commutation of each outer filter with the corresponding reduced matrix.
The segment $x_j+t(y-x_j)$ is feasible for $0\le t\le1$.
Differentiating at $t=0$ and removing the outer filters gives
\begin{align}
 \operatorname{Re}\tr\bigl(\sigma_j(x_j+b\Id)^{-1}y\bigr)
 &\le \operatorname{Re}\tr\bigl(\sigma_j(x_j+b\Id)^{-1}x_j\bigr).
 \label{eq:patch_unnormalized_variation}
\end{align}
Set $c=\lVert\Psi\rVert^{-2}$ when $\Psi\ne0$, and $c=0$ otherwise.
Then $c\ge0$ and $\rho_j=c\sigma_j$. Multiplying
(\ref{eq:patch_unnormalized_variation}) by $c$ proves
(\ref{eq:patch_normalized_variation}), including the case $\Psi=0$.
\end{proof}

\begin{theorem}[Clipped joint eigenvalues]
\label{thm:shifted_density_clipping}
\lean{Matrix.PosSemidef.exists_unitary_diagonal_clipped_of_forall_density}
\leanok
Let $x,\rho\in\C^{I\times I}$ be commuting density matrices and let $b>0$.
Suppose that every density matrix $y$ commuting with $x$ satisfies
\begin{align}
 \operatorname{Re}\tr\bigl(\rho(x+b\Id)^{-1}y\bigr)
 &\le \operatorname{Re}\tr\bigl(\rho(x+b\Id)^{-1}x\bigr).
 \label{eq:shifted_density_variation}
\end{align}
Then there are a unitary $U$, real families $(s_i)_{i\in I}$ and
$(p_i)_{i\in I}$, and $0<\lambda\le1$ such that
$x=U\operatorname{diag}(s_i)U^*$, $\rho=U\operatorname{diag}(p_i)U^*$, and
$s_i+b=\max\{p_i/\lambda,b\}$ for every $i\in I$.
This is the consequence of the first-order inequalities used in
\cite[Section~4, clipped eigenvalues]{OpenAI2026PolynomialPEPS}.
\end{theorem}
\begin{proof}
\uses{lem:commuting_hermitian,thm:rpow_conj_unitary}
\leanok
Simultaneous diagonalization gives $s_i,p_i\ge0$ and
$\sum_i s_i=\sum_i p_i=1$. Set $g_i=p_i/(s_i+b)$ and
$\lambda=\sum_i s_i g_i$. Testing (\ref{eq:shifted_density_variation})
on each rank-one projection in this basis gives $g_i\le\lambda$.
Since $\sum_i s_i g_i=\sum_i s_i\lambda$, equality holds in each
inequality $s_i g_i\le s_i\lambda$. Thus $g_i=\lambda$ when $s_i>0$.
Some $p_i$ is positive, so $\lambda>0$. Also
$s_i g_i\le(s_i+b)g_i=p_i$, and summation gives $\lambda\le1$.
If $s_i=0$, then $p_i=g_i b\le\lambda b$; if $s_i>0$, then
$p_i=\lambda(s_i+b)$. These two cases give the stated maximum.
\end{proof}

\begin{theorem}[Inverse-power filters with clipped eigenvalues]
    \label{thm:shifted_filter_clipping}
    \lean{Matrix.exists_clipped_rpow_add_smul_one_of_diagonal_max}
    Let $I$ be finite, let $x,\rho,U\in\C^{I\times I}$, and let
    $s,p\in\R^I$. Suppose that $U$ is unitary and
    \begin{align}
        x    & =U\operatorname{diag}(s_i)U^*, \\
        \rho & =U\operatorname{diag}(p_i)U^*.
    \end{align}
    Let $b,\lambda>0$ and $a\ge0$, and suppose that
    $s_i+b=\max\{p_i/\lambda,b\}$ for every $i\in I$.
    Set $L=(x+b\Id)^{-a/2}$. Then $L\rho=\rho L$, and there are a unitary
    $W$ and real families $\ell,q\in\R^I$ such that $\ell_i>0$ for all $i$,
    \begin{align}
        L    & =W\operatorname{diag}(\ell_i)W^*, \\
        \rho & =W\operatorname{diag}(q_i)W^*.
    \end{align}
    For every $i,k\in I$ with $q_i,q_k>0$,
    \begin{align}
        |\log\ell_i-\log\ell_k|
         & \le \frac a2|\log q_i-\log q_k|.
    \end{align}
    This gives the logarithmic bound for inverse-power filters in
    \cite[Section~4, clipped eigenvalues]{OpenAI2026PolynomialPEPS}.
\end{theorem}
\begin{proof}
    \uses{thm:rpow_conj_unitary,lem:clipped_filter_cost}
    Put $g_i=s_i+b=\max\{p_i/\lambda,b\}>0$. Functional calculus in the
    supplied unitary basis gives $L=U\operatorname{diag}(g_i^{-a/2})U^*$.
    The two diagonal matrices commute, so their conjugates $L$ and $\rho$
    commute. Choose $W=U$, $\ell_i=g_i^{-a/2}$ and $q_i=p_i$.
    Lemma~\ref{lem:clipped_filter_cost} bounds the logarithmic differences
    of $g_i^{a/2}$ by $(a/2)|\log p_i-\log p_k|$ whenever $p_i,p_k>0$.
    Since $\log(g_i^{-a/2})=-\log(g_i^{a/2})$, the absolute difference is
    unchanged. This proves the estimate, including $a=0$; an empty index set
    requires no separate argument.
\end{proof}
