\section{The original regions of the grouped metric}
\label{sec:regional_replica_metric_regions}

\begin{theorem}[The left, far and middle regions]
\label{thm:regional_replica_metric_regions}
\lean{TensorPower.regionalOriginalRegion_metric_regions}
\leanok
Let $V$ be a finite set, let $Q,Y\subseteq V$ be disjoint, and adjoin the original auxiliary
sites $c,r$. Group the sites into the five factors
$Q,Y,(Q\cup Y)^c,\{c\},\{r\}$, in that order.
The inverse images of the grouped regions $\{0,3\}$, $\{2,4\}$,
and $\{1\}$ are respectively
\begin{align*}
 Q\cup\{c\},\qquad (Q\cup Y)^c\cup\{r\},\qquad Y.
\end{align*}
These are the original metric regions of
\cite[Section~8]{OpenAI2026AreaLaw}.
\end{theorem}
\begin{proof}
\leanok
\uses{def:regional_original_region}
Evaluate the grouping map at each physical site and at the two
auxiliary sites. Disjointness identifies the second physical block
with all of $Y$.
\end{proof}

\begin{theorem}[Nesting after adjoining the auxiliary sites]
\label{thm:replica_metric_lifted_region_geometry}
\lean{TensorPower.replicaMetric_lifted_region_geometry}
\leanok
Let $Q_{jg},Y_{jg}$ be physical regions, indexed by arbitrary terminal
labels $j$ and bands $0\leq g<G$. Assume that every pair is disjoint
and that $Q_{jg}\cup Y_{jg}\subseteq Q_{j'h}$ for all $j,j'$ whenever
$g<h$. Adjoin distinct auxiliary sites $c,r$, and write
\begin{align*}
 L_{jg}&=\{c\}\cup Q_{jg},
 &E_{jg}&=Y_{jg},
 &F_{jg}&=\{r\}\cup(Q_{jg}\cup Y_{jg})^c,
\end{align*}
where the complement on the right is taken among physical sites.
Then $L_{jg}$ and $E_{jg}$ are disjoint, $F_{jg}=(L_{jg}\cup E_{jg})^c$
in the enlarged site set, and $L_{jg}\cup E_{jg}\subseteq L_{j'h}$
whenever $g<h$. These are the lifted regions used in
\cite[Section~8, lines~80--110 and~421--443]{OpenAI2026AreaLaw}.
\end{theorem}
\begin{proof}
\leanok
\uses{thm:regional_replica_metric_regions,thm:regional_replica_metric_partition}
The two auxiliary sites are distinct and lie outside the physical sites.
The induced partition gives disjointness. The complement identity follows
by separating the two auxiliary sites from a physical site. Finally,
$L_{jg}\cup E_{jg}=\{c\}\cup(Q_{jg}\cup Y_{jg})$; adjoining the same
site to the given physical inclusion proves the nesting assertion.
\end{proof}

\begin{theorem}[The induced partition]
\label{thm:regional_replica_metric_partition}
\lean{TensorPower.regionalOriginalRegion_metric_partition}
\leanok
For arbitrary subsets $Q,Y\subseteq V$, assign a physical site to
factor $0$ if it belongs to $Q$, otherwise to factor $1$ if it belongs
to $Y$, and otherwise to factor $2$. Assign $c$ to factor $3$ and
$r$ to factor $4$. The inverse images of $\{0,3\}$, $\{1\}$ and
$\{2,4\}$ are pairwise disjoint and cover every original physical
and auxiliary site. This assertion includes empty physical factors.
\end{theorem}
\begin{proof}
\leanok
\uses{def:regional_original_region}
The three displayed subsets partition the five factor indices.
Their inverse images therefore form a partition of the original sites.
\end{proof}
