\section{Quadratic forms under compression and relabelling}

The following elementary facts pass from operator inequalities to the
expectations of fixed vectors, as in the metric comparisons of
\cite[Section~8]{OpenAI2026AreaLaw}.
For complex matrices, $A\leq B$ means that $B-A$ is positive semidefinite.
All index sets below are finite and may be empty.

\begin{theorem}[Order of real quadratic forms]
  \label{thm:quadratic_form_order}
  \lean{Matrix.re_dotProduct_mulVec_le_of_le}
  \leanok
  Let $A,B$ be complex matrices indexed by the same finite set. If $A\leq B$,
  then every complex vector $v$ on that set satisfies
  $\operatorname{Re}(v^*Av)\leq\operatorname{Re}(v^*Bv)$.
\end{theorem}

\begin{proof}
  \leanok
  Positivity of $B-A$ gives
  $0\leq\operatorname{Re}(v^*(B-A)v)$; expanding this expression gives
  the conclusion.
\end{proof}

\begin{theorem}[Quadratic bound on a fixed vector]
  \label{thm:hermitian_compression_quadratic}
  \lean{Matrix.IsHermitian.re_dotProduct_mulVec_le_of_compression_le}
  \leanok
  Let $P,M$ be complex matrices indexed by the same finite set, with
  $P=P^*$. Suppose that $c\in\mathbb R$, $PMP\leq cP$, and $Pv=v$.
  Then $\operatorname{Re}(v^*Mv)\leq c\operatorname{Re}(v^*v)$.
\end{theorem}

\begin{proof}
  \leanok
  \uses{thm:quadratic_form_order}
  Since $Pv=v$ and $P=P^*$, one has $v^*P=v^*$. Consequently,
  $v^*PMPv=v^*Mv$ and
  $\operatorname{Re}(v^*(cP)v)=c\operatorname{Re}(v^*v)$.
  Apply Theorem~\ref{thm:quadratic_form_order} to $PMP\leq cP$.
\end{proof}

\begin{theorem}[Relabelling a quadratic form]
  \label{thm:quadratic_form_reindex}
  \lean{Matrix.star_dotProduct_submatrix_equiv}
  \leanok
  Let $e\colon I\to J$ be a bijection of finite sets, let $H$ be a complex
  matrix indexed by $I$, and let $w\in\mathbb C^J$. Define the matrix
  $H^e$ by $H^e_{a,b}=H_{e^{-1}(a),e^{-1}(b)}$. Then
  $w^*H^e w=(w\circ e)^*H(w\circ e)$.
\end{theorem}

\begin{proof}
  \leanok
  In the finite sum defining the left-hand side, substitute $a=e(i)$ and
  $b=e(j)$. This gives the right-hand side term by term.
\end{proof}

\section{Scalar lower bounds for commuting products}

For complex matrices, $A\le B$ means that $B-A$ is positive semidefinite.
The following estimates combine scalar lower bounds for commuting band
metrics in the replica construction of~\cite{OpenAI2026AreaLaw}.
All matrices are indexed by the same finite set $I$, which may be empty.

\begin{theorem}[Multiplication of scalar lower bounds]
    \label{thm:commuting_product_scalar_bound}
    \lean{Matrix.smul_one_le_mul_of_commute}
    \leanok
    Let $A,B$ be complex matrices indexed by $I$, and let $a,b\in\mathbb R$.
    Suppose that $a\ge0$, $b\ge0$, $aI\le A$, $bI\le B$, and $AB=BA$.
    Then $(ab)I\le AB$.
\end{theorem}

\begin{proof}
    \leanok
    The bound $B\ge bI\ge0$ makes $B$ positive semidefinite.
    The positive semidefinite matrices $A-aI$ and $B$ commute, so their
    product is positive semidefinite. Since $a\ge0$, the identity
    \begin{align}
        AB-abI=(A-aI)B+a(B-bI)\ge0
        \label{eq:commuting_product_floor_decomposition}
    \end{align}
    gives the conclusion.
\end{proof}

\begin{theorem}[A common scalar lower bound for an ordered product]
    \label{thm:commuting_list_scalar_bound}
    \lean{Matrix.pow_smul_one_le_list_prod}
    \leanok
    Let $(A_0,\ldots,A_{m-1})$ be any finite list of complex matrices
    indexed by $I$, and let $b\ge0$. Suppose that $bI\le A_j$ for every
    position $j$ and that the matrices at distinct positions commute.
    Then $b^m I\le A_0\cdots A_{m-1}$.
    Repeated matrices are allowed. For the empty list, the product is $I$
    and $b^0=1$, including when $b=0$.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:commuting_product_scalar_bound}
    Induct on the length of the list. The empty product gives equality.
    Write a nonempty list as $A$ followed by a list of length $m$ whose
    ordered product is $B$. The induction hypothesis gives $b^m I\le B$.
    Since $A$ commutes with every factor of $B$, it commutes with $B$.
    Theorem~\ref{thm:commuting_product_scalar_bound}, applied to the scalar
    bounds $bI\le A$ and $b^m I\le B$, gives $b^{m+1}I\le AB$.
\end{proof}
