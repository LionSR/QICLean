\section{Bell insertion in the original physical coordinates}

\begin{definition}[Separating the added copies]
    \label{def:added_site_copy_separation}
    \lean{TensorPower.addedSiteConfigEquiv}
    \uses{def:added_site_tensor_insertion}
    Let $F$ be a finite set with finite coordinate sets $I_f$, let $A$
    be another finite coordinate set, and let $k\geq0$.
    Write $\mathcal C_k=(\prod_{f\in F}I_f)^k$.
    Separating the added coordinate in each copy gives the bijection
    \begin{align*}
        s_k:\left(A\times\prod_{f\in F}I_f\right)^k
        &\longrightarrow A^k\times\mathcal C_k,
        &
        s_k((a_j,x_j)_j)&=((a_j)_j,(x_j)_j).
    \end{align*}
\end{definition}

\begin{theorem}[Gram matrix of a tensor-power insertion]
    \label{thm:added_site_insertion_gram}
    \lean{TensorPower.addedSiteInsertion_conjTranspose_mul,TensorPower.isIsometry_addedSiteInsertion}
    \uses{def:added_site_tensor_insertion,def:added_site_copy_separation}
    For every $b:A\to\C$, the insertion matrix $J_b$ satisfies
    \begin{align*}
        J_b^*J_b
        &=\left(\sum_{a\in A}|b(a)|^2\right)^k I_{\mathcal C_k}.
    \end{align*}
    In particular, if $\|b\|_2=1$, then $J_b$ is an isometry.
    The identity includes $k=0$ and empty coordinate sets.
\end{theorem}
\begin{proof}
    Reindex the sum defining the Gram matrix by $s_k$.
    The original-coordinate indicators give $\mathbf1_{\{x=y\}}$.
    The remaining sum is
    $\sum_{a\in A^k}\prod_j|b(a_j)|^2
      =(\sum_{a\in A}|b(a)|^2)^k$.
\end{proof}

\begin{theorem}[Copy symmetry under insertion]
    \label{thm:added_site_insertion_copy_symmetry}
    \lean{TensorPower.permOp_copyPerm_mul_addedSiteInsertion,TensorPower.symProj_mul_addedSiteInsertion,TensorPower.addedSiteInsertion_mulVec_mem_symmetricSubspace}
    \uses{thm:added_site_tensor_insertion_permutations}
    Let $U(\sigma)$ and $\widetilde U(\sigma)$ denote simultaneous copy
    permutations before and after insertion. Let $P_{\rm sym}$ and
    $\widetilde P_{\rm sym}$ be their averaging projections.
    For every $b$ and $\sigma\in\mathfrak S_k$,
    \begin{align*}
        \widetilde U(\sigma)J_b&=J_bU(\sigma),\\
        \widetilde P_{\rm sym}J_b&=J_bP_{\rm sym}.
    \end{align*}
    Consequently, $J_bv$ is symmetric whenever $v$ is symmetric.
    Neither vector is required to be normalized.
\end{theorem}
\begin{proof}
    Specialize the subsystem identity to all original sites.
    Averaging over $\mathfrak S_k$ gives the projection identity.
    Applying the permutation identity to a fixed vector proves the last
    assertion.
\end{proof}

Let $V$ be finite with finite coordinate sets $D_v$, fix $X\subseteq V$,
and let $C$ and $R$ be independent finite auxiliary coordinate sets.
Put $A_X=\prod_{v\in X}D_v$ and
$B_X=\prod_{v\notin X}D_v$.
The original site family has configuration set
$\mathcal H=D_V\times C\times R$, where
$D_V=\prod_{v\in V}D_v$.
The exterior family has configuration set
$\mathcal H_e=B_X\times R$, with $R$ retained as an individual site.

\begin{definition}[Grouping the inserted Bell coordinates]
    \label{def:bell_insertion_coordinates}
    \lean{TensorPower.bellInsertionSiteEquiv,TensorPower.bellInsertionCopyEquiv}
    \uses{def:compressed_site_family}
    Group the coordinates of $X$ with $C$, and leave $R$ and all
    complementary physical coordinates separate. This gives
    \begin{align*}
        e:\mathcal H&\longrightarrow(A_X\times C)\times\mathcal H_e,
        &
        e(x,c,r)&=((x|_X,c),(x|_{X^c},r)).
    \end{align*}
    Applying $e$ independently to the $k$ copies defines $e_k$.
    These are the coordinate changes in
    \cite[Section~8, lines~283--308]{OpenAI2026AreaLaw}.
\end{definition}

\begin{theorem}[Subsystem actions in the Bell coordinates]
    \label{thm:bell_insertion_subsystem_coordinates}
    \lean{TensorPower.bellInsertionCopyEquiv_subsystemPerm_withPin,TensorPower.bellInsertionCopyEquiv_subsystemPerm_exterior,TensorPower.labelObservable_bellInsertionCopyEquiv_withPin,TensorPower.labelObservable_bellInsertionCopyEquiv_exterior}
    \uses{def:bell_insertion_coordinates,def:copy_permutations,def:label_projectors}
    Fix any $U\subseteq V$ and write $\widehat U$ for its exterior part
    $U\setminus X$. Denote the new bundled site $A_X\times C$ by $*$.
    For every $\sigma\in\mathfrak S_k$ and original configuration $x$,
    \begin{align*}
        e_k\bigl(U_{C\cup X\cup U}(\sigma)x\bigr)
            &=\widetilde U_{\{*\}\cup\widehat U}(\sigma)e_k(x),\\
        e_k\bigl(U_{U\setminus X}(\sigma)x\bigr)
            &=\widetilde U_{\widehat U}(\sigma)e_k(x).
    \end{align*}
    For a real function $f$ on the irreducible labels of $\mathfrak S_k$,
    let $L_Q(f)=\sum_\lambda f(\lambda)\Pi_{Q,\lambda}$.
    If $W_kv=v\circ e_k$, then
    \begin{align*}
        W_k\widetilde L_{\{*\}\cup\widehat U}(f)W_k^*
            &=L_{C\cup X\cup U}(f),\\
        W_k\widetilde L_{\widehat U}(f)W_k^*
            &=L_{U\setminus X}(f).
    \end{align*}
    No disjointness condition on $X$ and $U$ is required.
\end{theorem}
\begin{proof}
    On the bundled site, the first action permutes all coordinates of
    $X$ and $C$ together; the second action fixes them. Both actions fix
    $R$ and agree on each complementary physical site.
    Transporting the same central group-algebra element
    $\sum_\lambda f(\lambda)e_\lambda$ gives the observable identities.
\end{proof}

For $b:A_X\times C\to\C$, let
$J=W_kJ_b:\ell^2(\mathcal H_e^k)\to\ell^2(\mathcal H^k)$
be the insertion expressed in the original physical coordinates.

\begin{theorem}[Label observables through the physical insertion]
    \label{thm:bell_insertion_label_intertwining}
    \lean{TensorPower.labelObservable_mul_bellInsertion_withPin,TensorPower.labelObservable_mul_bellInsertion_exterior,TensorPower.labelObservable_mul_bellInsertion_auxiliary}
    \uses{thm:bell_insertion_subsystem_coordinates,thm:added_site_tensor_insertion_label_observables}
    For every $b$, $U\subseteq V$, and real label function $f$,
    \begin{align*}
        L_{C\cup X\cup U}(f)J&=J L^e_{\widehat U}(f),\\
        L_{U\setminus X}(f)J&=J L^e_{\widehat U}(f),\\
        L_R(f)J&=J L^e_R(f).
    \end{align*}
    The same function $f$ occurs on both sides of each equality.
    The registers $C$ and $R$ need not have the same dimension.
\end{theorem}
\begin{proof}
    Use the subsystem coordinate identities and the two insertion
    identities for label observables. For the last equality, the
    original register $R$ becomes the retained exterior register $R$.
\end{proof}

\begin{theorem}[Isometry and copy symmetry of the physical insertion]
    \label{thm:bell_insertion_isometry_symmetry}
    \lean{TensorPower.isIsometry_bellInsertion,TensorPower.permOp_copyPerm_mul_bellInsertion,TensorPower.symProj_mul_bellInsertion}
    \uses{thm:added_site_insertion_gram,thm:added_site_insertion_copy_symmetry,def:bell_insertion_coordinates}
    The coordinate change $e_k$ commutes with simultaneous copy
    permutations. Hence, for every $b$ and $\sigma\in\mathfrak S_k$,
    \begin{align*}
        U_{\mathcal H}(\sigma)J&=J U_{\mathcal H_e}(\sigma),\\
        P_{\mathcal H,{\rm sym}}J&=J P_{\mathcal H_e,{\rm sym}}.
    \end{align*}
    If $\|b\|_2=1$, then also $J^*J=I$.
    These identities include zero copies and empty coordinate sets.
\end{theorem}
\begin{proof}
    The map $e_k$ acts separately in each copy, so it commutes with
    their simultaneous permutation. Transport the corresponding
    rectangular identity and average over $\mathfrak S_k$.
    Finally, $W_k$ is unitary, and the Gram identity gives
    $J^*J=J_b^*J_b=I$ when $\|b\|_2=1$.
\end{proof}
