\section{Error reduction by a spectral projector}

\begin{theorem}[Error reduction from a finite tensor decomposition]
    \label{thm:viable_tensor_decomposition}
    \lean{Submodule.IsViable.map_of_tensor_decomposition}
    \leanok
    \uses{thm:subspace_close_contract, thm:subspace_close_mono, def:subspace_viable}
    Let $\mathcal H_M$ and $\mathcal H_R$ be finite-dimensional real or complex
    inner product spaces, let $I$ be finite, and let
    $K=\sum_{i\in I}A_i\otimes R_i$ act on
    $\mathcal H_M\otimes\mathcal H_R$. Let $T$ be a subspace of this tensor
    product. Suppose that $K$ preserves $T$ and $T^\perp$, that $\Delta\ge0$,
    and that $\|Kt\|\ge\|t\|$ for $t\in T$ and
    $\|Kz\|^2\le\Delta\|z\|^2$ for $z\in T^\perp$.
    If $S\subseteq\mathcal H_M$ is $\delta$-viable for $T$ and $\delta<1$,
    then $V=\sum_{i\in I}A_iS$ has dimension at most $|I|\dim S$ and is
    $\Delta/(1-\delta)^2$-viable for $T$.
    This is the two-factor tensor-containment argument supporting
    Lemma~2.9 of~\cite[lines 606--622]{Arad2016RigorousRG}.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:finite_image_span_dimension}
    Each image $A_iS$ has dimension at most $\dim S$, so their sum has
    dimension at most $|I|\dim S$. Error reduction by a contraction gives
    closeness of $K(S\otimes\mathcal H_R)$ to $T$ with parameter
    $\Delta/(1-\delta)^2$. For $s\in S$ and $r\in\mathcal H_R$,
    $K(s\otimes r)=\sum_{i\in I}A_is\otimes R_ir$ belongs to
    $V\otimes\mathcal H_R$. Closeness passes to this larger subspace.
\end{proof}

Let $\mathcal H=\mathcal H_L\otimes\mathcal H_M\otimes\mathcal H_R$ and let
$b_j$ be an orthonormal basis of common eigenvectors of a Hamiltonian $H$ and
an operator $K$, with $Hb_j=h_jb_j$ and $Kb_j=k_jb_j$. The low-energy space is
$T=\operatorname{span}\{b_j:h_j\le\eta_0\}$.

\begin{definition}[Spectral projector]
    \label{def:spectral_projector}
    \lean{Submodule.IsSpectralProjector}
    \leanok
    The operator $K$ is a spectral projector for $(H,\eta_0,\eta_1)$ with
    parameter $\Delta$ if $h_j\le\eta_0$ implies $k_j\ge1$ and $h_j\ge\eta_1$
    implies $k_j^2\le\Delta$. This is Definition~def:spectral-AGSP
    of~\cite[lines 570--581]{Arad2016RigorousRG} with closed spectral
    conditions and without positivity of $K$; with the printed open
    conditions the lemma below fails at the eigenvalue
    $\eta_0$~\cite{gap:alvv16_spectral_projector_boundary}.
\end{definition}

\begin{theorem}[Spectral projectors contract the complement]
    \label{thm:spectral_projector_contract}
    \lean{Submodule.IsSpectralProjector.contract}
    \leanok
    \uses{def:spectral_projector}
    If $K$ is a spectral projector and $H$ has no eigenvalue in
    $(\eta_0,\eta_1)$, then $K$ maps $T$ and $T^\perp$ into themselves,
    $\lVert Kt\rVert\ge\lVert t\rVert$ for $t\in T$, and
    $\lVert Kq\rVert^2\le\Delta\lVert q\rVert^2$ for $q\in T^\perp$.
\end{theorem}

\begin{proof}
    \leanok
    In the basis $b_j$, the vectors of $T$ have vanishing coefficients
    outside $\{j:h_j\le\eta_0\}$ and those of $T^\perp$ inside it, and $K$
    multiplies the $j$-th coefficient by $k_j$. Parseval's identity gives
    the two norm bounds.
\end{proof}

\begin{theorem}[Error reduction]
    \label{thm:viable_spectral_projector}
    \lean{Submodule.IsViable.spectralProjector}
    \leanok
    \uses{thm:spectral_projector_contract, thm:subspace_close_contract, thm:subspace_close_mono, def:subspace_viable}
    Let $K=\sum_{i=1}^{D^2}L_i\otimes A_i\otimes R_i$ be a spectral
    projector for $(H,\eta_0,\eta_1)$ with parameter $\Delta\ge0$, let $H$
    have no eigenvalue in $(\eta_0,\eta_1)$, and let $S\subseteq\mathcal H_M$
    be $\delta$-viable for $T$ with $\delta<1$. Then
    $V=\operatorname{span}\{A_iS\}$ has dimension at most $D^2\dim S$ and
    is $\Delta/(1-\delta)^2$-viable for $T$. This is Lemma~l:er
    of~\cite[lines 597--622]{Arad2016RigorousRG}, with the closed spectral
    conditions and the hypothesis $\delta<1$ that its proof uses.
\end{theorem}

\begin{proof}
    \uses{thm:finite_image_span_dimension}
    \leanok
    The dimension bound holds because $V$ is the sum of the $D^2$ images
    $A_iS$. By the previous theorem, $K$ satisfies the hypotheses of the
    error reduction by a contraction, so
    $K(\mathcal H_L\otimes S\otimes\mathcal H_R)$ is
    $\Delta/(1-\delta)^2$-close to $T$. Each
    $K((l\otimes s)\otimes r)=\sum_i(L_il\otimes A_is)\otimes R_ir$ lies in
    $\mathcal H_L\otimes V\otimes\mathcal H_R$, and closeness passes to the
    larger subspace.
\end{proof}

\begin{theorem}[Dimension of a finite sum of linear images]
    \label{thm:finite_image_span_dimension}
    \lean{Submodule.finrank_iSup_map_le_card_mul}
    \leanok
    Let $\mathbb K$ be $\R$ or $\C$, let $E,F$ be inner product spaces over
    $\mathbb K$, and let $S\subseteq E$ be a finite-dimensional subspace.
    For a finite set $I$ and linear maps $A_i:E\to F$, set
    $V=\sum_{i\in I}A_iS$. Then
    \begin{align}
        \dim_{\mathbb K}V\le |I|\dim_{\mathbb K}S.\notag
    \end{align}
    The spaces $E,F$ need not be finite-dimensional or complete, and $I$ may be empty.
    This is the dimension estimate used in
    \cite[Lemma~2.9 and Proposition~4.10]{Arad2016RigorousRG}.
\end{theorem}
\begin{proof}
    \leanok
    Each restriction $A_i|_S$ has image of dimension at most
    $\dim_{\mathbb K}S$. The dimension of a finite sum of subspaces is at most
    the sum of their dimensions, so
    \begin{align}
        \dim_{\mathbb K}V
        \le\sum_{i\in I}\dim_{\mathbb K}(A_iS)
        \le |I|\dim_{\mathbb K}S.\notag
    \end{align}
\end{proof}


\section{Distinct lower and upper targets}

\begin{theorem}[Tensor error reduction between nested targets]
    \label{thm:viable_nested_tensor_decomposition}
    \lean{Submodule.IsViable.map_of_nested_tensor_decomposition}
    \leanok
    \uses{def:subspace_viable}
    Let $T\subseteq T_+\subseteq\mathcal H_M\otimes\mathcal H_R$,
    and let $K=\sum_{i\in I}A_i\otimes R_i$, with $I$ finite.
    Suppose that $K$ satisfies the hypotheses of
    Theorem~\ref{thm:subspace_close_nested_contract} for $T,T_+$ and
    $\Delta\ge0$. If $S$ is $\delta$-viable for $T_+$ and $\delta<1$,
    then $V=\sum_i A_iS$ has dimension at most $|I|\dim S$ and is
    $\Delta/(1-\delta)^2$-viable for $T$.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:subspace_close_nested_contract, thm:subspace_close_mono,
        thm:finite_image_span_dimension}
    Apply nested-target error reduction to $S\otimes\mathcal H_R$.
    The identity $K(s\otimes r)=\sum_i A_is\otimes R_ir$ places its
    image in $V\otimes\mathcal H_R$. Closeness passes to this larger
    space, and the dimension bound follows by summing the image dimensions.
\end{proof}

\begin{theorem}[Contraction above a separate cutoff]
    \label{thm:spectral_projector_separate_cutoff}
    \lean{Submodule.IsSpectralProjector.contract_between_cutoffs}
    \leanok
    \uses{def:spectral_projector}
    Let $K$ be a spectral projector for $(H,\eta_0,\eta_1)$ with
    parameter $\Delta$. Let $c\in\R$ satisfy
    $h_j>c\Longrightarrow h_j\ge\eta_1$ for every eigenvalue $h_j$.
    Set $T=\operatorname{span}\{b_j:h_j\le\eta_0\}$ and
    $T_c=\operatorname{span}\{b_j:h_j\le c\}$.
    Then $KT\subseteq T$, $K(T_c^\perp)\subseteq T_c^\perp$,
    $\|Kt\|\ge\|t\|$ on $T$, and
    $\|Kz\|^2\le\Delta\|z\|^2$ on $T_c^\perp$.
\end{theorem}
\begin{proof}\leanok
    The coordinates of $T_c^\perp$ vanish on $h_j\le c$.
    Multiplication of the remaining coordinates by $k_j$ preserves this
    support and contracts their squared norms by $\Delta$.
    Parseval's identity gives the contraction estimate. The same
    coordinate argument on $T$ uses $k_j\ge1$.
\end{proof}

\begin{theorem}[Spectral error reduction between two cutoffs]
    \label{thm:viable_spectral_two_cutoffs}
    \lean{Submodule.IsViable.spectralProjector_between_cutoffs}
    \leanok
    \uses{def:spectral_projector, def:subspace_viable}
    Let $\eta_0\le\eta_1$ and let
    $K=\sum_{i\in I}A_i\otimes R_i$ be a spectral projector for
    $(H,\eta_0,\eta_1)$ with parameter $\Delta\ge0$.
    Write $T_a=\operatorname{span}\{b_j:h_j\le a\}$.
    If $S$ is $\delta$-viable for $T_{\eta_1}$ and $\delta<1$, then
    $V=\sum_i A_iS$ has dimension at most $|I|\dim S$ and is
    $\Delta/(1-\delta)^2$-viable for $T_{\eta_0}$.
    No restriction is imposed on eigenvalues between the cutoffs.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:spectral_projector_separate_cutoff,
        thm:viable_nested_tensor_decomposition}
    Take $c=\eta_1$ in the preceding contraction estimate.
    Since $T_{\eta_0}\subseteq T_{\eta_1}$, the image of
    $T_{\eta_1}^\perp$ also lies in $T_{\eta_0}^\perp$.
    Apply tensor error reduction between nested targets.
\end{proof}

\begin{theorem}[A Chebyshev filter from a common polynomial family]
    \label{thm:chebyshev_nested_polynomial_family}
    \lean{Matrix.exists_chebyshev_nested_viability_of_polynomial_family}
    \leanok
    \uses{def:subspace_viable}
    Let $B$ be a finite Hermitian matrix with a complete orthonormal
    eigenbasis, let $a<b\le u$, and suppose that $\|B\|\le u$.
    Let $\mathcal T$ identify its vector space isometrically with
    $\mathcal H_M\otimes\mathcal H_R$. Suppose there are fixed matrices
    $W_1,\ldots,W_N$ such that every complex polynomial $Q$ of degree
    at most $n$ admits a decomposition
    $\mathcal TQ(B)\mathcal T^{-1}=\sum_{i=1}^N W_i\otimes M_i(Q)$.
    There is a real polynomial $p$ of degree exactly $n$, with $p(a)=1$,
    and such a decomposition of $p(B)$, chosen before every subspace $S$,
    with the following property. Write $T_c$ for the image under
    $\mathcal T$ of the spectral space of $B$ below the closed cutoff $c$,
    and set $V=\sum_i W_iS$. If $S$ is $\delta$-viable for $T_b$ and
    $\delta<1$, then $\dim V\le N\dim S$ and $V$ is
    $\Delta/(1-\delta)^2$-viable for $T_a$, where
    $\Delta=4\exp(-4n\sqrt{(b-a)/(u-a)})$.
    The polynomial decompositions are explicit assumptions of this
    auxiliary statement; no spectral gap between $a$ and $b$ is assumed.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:chebyshev_spectral_projector, thm:viable_spectral_two_cutoffs}
    Conjugation by $\mathcal T$ preserves self-adjointness and identifies
    all closed spectral spaces. The Chebyshev filter expands below $a$
    and contracts squared norms above $b$ by $\Delta$.
    Apply the assumed decomposition to this one polynomial and then
    spectral error reduction between the two cutoffs.
\end{proof}

\begin{lemma}[The image of a finite tensor family]
    \label{lem:tensor_family_image_containment}
    \lean{Submodule.tmul_mem_tensorProd, Submodule.tensorProd_le_iff,
        Submodule.map_tensorProd_top_le_iSup_map}
    \leanok
    Let $I$ be finite and let $K=\sum_{i\in I}A_i\otimes B_i$ act on
    $E\otimes F$. For any subspace $S\subseteq E$, put
    $V=\sum_{i\in I}A_iS$. Then
    $K(S\otimes F)\subseteq V\otimes F$.
    This is the containment argument of
    \cite[Lemma~l:er, lines 606--622]{Arad2016RigorousRG}.
\end{lemma}
\begin{proof}\leanok
    A tensor product of subspaces lies in a subspace precisely when all
    its elementary tensors do. For $s\in S$ and $f\in F$,
    $K(s\otimes f)=\sum_i A_is\otimes B_if$, and every summand belongs
    to $V\otimes F$.
\end{proof}

\begin{lemma}[Normalized reduction by a finite tensor family]
    \label{lem:viable_tensor_decomposition_sharp}
    \lean{Submodule.IsViable.map_of_nested_tensor_decomposition_sharp,
        Submodule.IsViable.map_of_tensor_decomposition_sharp}
    \leanok
    Let $T\subseteq T_+\subseteq E\otimes F$ and let
    $K=\sum_{i\in I}A_i\otimes B_i$, with $I$ finite. Suppose
    $KT\subseteq T$, $K(T_+^\perp)\subseteq T^\perp$,
    $\|Kt\|\ge\|t\|$ for $t\in T$, and
    $\|Kz\|^2\le\Delta\|z\|^2$ for $z\in T_+^\perp$, where
    $\Delta\ge0$. If $S\subseteq E$ is $\delta$-viable for $T_+$ and
    $0\le\delta<1$, then $V=\sum_iA_iS$ has dimension at most
    $|I|\dim S$ and is $\Delta\delta/(1-\delta+\Delta\delta)$-viable
    for $T$. In particular this holds with $T_+=T$.
    The low-band operator need not be the identity or have a constant
    eigenvalue on $T$.
\end{lemma}
\begin{proof}\leanok
    \uses{lem:subspace_nested_error_reduction_sharp,
        lem:tensor_family_image_containment,thm:finite_image_span_dimension}
    Apply normalized error reduction to $S\otimes F$ and use
    $K(S\otimes F)\subseteq V\otimes F$. Closeness passes to the larger
    space. The sum of the image dimensions is at most $|I|\dim S$.
\end{proof}
