\section{Error reduction by a spectral projector}

\begin{theorem}[Error reduction from a finite tensor decomposition]
    \label{thm:viable_tensor_decomposition}
    \lean{Submodule.IsViable.map_of_tensor_decomposition}
    \leanok
    Let $\mathcal H_M$ and $\mathcal H_R$ be finite-dimensional real or complex
    inner product spaces, let $I$ be finite, and let
    $K=\sum_{i\in I}A_i\otimes R_i$ act on
    $\mathcal H_M\otimes\mathcal H_R$. Let $T$ be a subspace of this tensor
    product. Suppose that $K$ preserves $T$ and $T^\perp$, that $\Delta\ge0$,
    and that $\|Kt\|\ge\|t\|$ for $t\in T$ and
    $\|Kz\|^2\le\Delta\|z\|^2$ for $z\in T^\perp$.
    If $S\subseteq\mathcal H_M$ is $\delta$-viable for $T$ and $\delta<1$,
    then $V=\sum_{i\in I}A_iS$ has dimension at most $|I|\dim S$ and is
    $\Delta/(1-\delta)^2$-viable for $T$.
    This is the two-factor tensor-containment argument supporting
    Lemma~2.9 of~\cite[lines 606--622]{Arad2016RigorousRG}.
\end{theorem}

\begin{proof}\leanok
    \uses{thm:viable_nested_tensor_decomposition}
    Apply Theorem~\ref{thm:viable_nested_tensor_decomposition} with
    $T_-=T_+=T$.
\end{proof}

Let $\mathcal H=\mathcal H_L\otimes\mathcal H_M\otimes\mathcal H_R$ and let
$b_j$ be an orthonormal basis of common eigenvectors of a Hamiltonian $H$ and
an operator $K$, with $Hb_j=h_jb_j$ and $Kb_j=k_jb_j$. The low-energy space is
$T=\operatorname{span}\{b_j:h_j\le\eta_0\}$.

\begin{definition}[Spectral projector]
    \label{def:spectral_projector}
    \lean{Submodule.IsSpectralProjector}
    \leanok
    The operator $K$ is a spectral projector for $(H,\eta_0,\eta_1)$ with
    parameter $\Delta$ if $h_j\le\eta_0$ implies $k_j\ge1$ and $h_j\ge\eta_1$
    implies $k_j^2\le\Delta$. This is Definition~def:spectral-AGSP
    of~\cite[lines 570--581]{Arad2016RigorousRG} with closed spectral
    conditions and without positivity of $K$; with the printed open
    conditions the lemma below fails at the eigenvalue
    $\eta_0$~\cite{gap:alvv16_spectral_projector_boundary}.
\end{definition}

\begin{theorem}[Spectral projectors contract the complement]
    \label{thm:spectral_projector_contract}
    \lean{Submodule.IsSpectralProjector.contract}
    \leanok
    \uses{def:spectral_projector}
    If $K$ is a spectral projector and $H$ has no eigenvalue in
    $(\eta_0,\eta_1)$, then $K$ maps $T$ and $T^\perp$ into themselves,
    $\lVert Kt\rVert\ge\lVert t\rVert$ for $t\in T$, and
    $\lVert Kq\rVert^2\le\Delta\lVert q\rVert^2$ for $q\in T^\perp$.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:spectral_projector_distinct_cutoffs}
    Apply Theorem~\ref{thm:spectral_projector_distinct_cutoffs} with
    $c=\eta_0$. The spectral gap implies that every eigenvalue greater
    than $c$ is at least $\eta_1$.
\end{proof}

\begin{theorem}[Error reduction]
    \label{thm:viable_spectral_projector}
    \lean{Submodule.IsViable.spectralProjector}
    \leanok
    \uses{thm:spectral_projector_contract, thm:subspace_close_contract, thm:subspace_close_mono, def:subspace_viable}
    Let $K=\sum_{i=1}^{D^2}L_i\otimes A_i\otimes R_i$ be a spectral
    projector for $(H,\eta_0,\eta_1)$ with parameter $\Delta\ge0$, let $H$
    have no eigenvalue in $(\eta_0,\eta_1)$, and let $S\subseteq\mathcal H_M$
    be $\delta$-viable for $T$ with $\delta<1$. Then
    $V=\operatorname{span}\{A_iS\}$ has dimension at most $D^2\dim S$ and
    is $\Delta/(1-\delta)^2$-viable for $T$. This is Lemma~l:er
    of~\cite[lines 597--622]{Arad2016RigorousRG}, with the closed spectral
    conditions and the hypothesis $\delta<1$ that its proof uses.
\end{theorem}

\begin{proof}
    \uses{thm:finite_image_span_dimension}
    \leanok
    The dimension bound holds because $V$ is the sum of the $D^2$ images
    $A_iS$. By the previous theorem, $K$ satisfies the hypotheses of the
    error reduction by a contraction, so
    $K(\mathcal H_L\otimes S\otimes\mathcal H_R)$ is
    $\Delta/(1-\delta)^2$-close to $T$. Each
    $K((l\otimes s)\otimes r)=\sum_i(L_il\otimes A_is)\otimes R_ir$ lies in
    $\mathcal H_L\otimes V\otimes\mathcal H_R$, and closeness passes to the
    larger subspace.
\end{proof}

\begin{theorem}[Dimension of a finite sum of linear images]
    \label{thm:finite_image_span_dimension}
    \lean{Submodule.finrank_iSup_map_le_card_mul}
    \leanok
    Let $\mathbb K$ be $\R$ or $\C$, let $E,F$ be inner product spaces over
    $\mathbb K$, and let $S\subseteq E$ be a finite-dimensional subspace.
    For a finite set $I$ and linear maps $A_i:E\to F$, set
    $V=\sum_{i\in I}A_iS$. Then
    \begin{align}
        \dim_{\mathbb K}V\le |I|\dim_{\mathbb K}S.\notag
    \end{align}
    The spaces $E,F$ need not be finite-dimensional or complete, and $I$ may be empty.
    This is the dimension estimate used in
    \cite[Lemma~2.9 and Proposition~4.10]{Arad2016RigorousRG}.
\end{theorem}
\begin{proof}
    \leanok
    Each restriction $A_i|_S$ has image of dimension at most
    $\dim_{\mathbb K}S$. The dimension of a finite sum of subspaces is at most
    the sum of their dimensions, so
    \begin{align}
        \dim_{\mathbb K}V
        \le\sum_{i\in I}\dim_{\mathbb K}(A_iS)
        \le |I|\dim_{\mathbb K}S.\notag
    \end{align}
\end{proof}

\begin{theorem}[Tensor error reduction between nested targets]
    \label{thm:viable_nested_tensor_decomposition}
    \lean{Submodule.IsViable.map_of_nested_tensor_decomposition}
    \leanok
    \uses{def:subspace_viable}
    Let $K=\sum_{i\in I}A_i\otimes R_i$ and $T_-\subseteq T_+$ be
    subspaces of $\mathcal H_M\otimes\mathcal H_R$. Suppose $K$ preserves
    $T_-$ and $T_+^\perp$, expands norms on $T_-$ and contracts squared
    norms by $\Delta\ge0$ on $T_+^\perp$.
    If $S\subseteq\mathcal H_M$ is $\delta$-viable for $T_+$ with
    $\delta<1$, then $V=\sum_{i\in I}A_iS$ has dimension at most
    $|I|\dim S$ and is $\Delta/(1-\delta)^2$-viable for $T_-$.
    This is the tensor argument in
    \cite[Theorem~4.8, lines 1037--1041]{Arad2016RigorousRG}.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:subspace_close_nested_contract,thm:subspace_close_mono,
        thm:finite_image_span_dimension}
    The nested-target estimate applies to $S\otimes\mathcal H_R$.
    For $s\in S$ and $r\in\mathcal H_R$,
    $K(s\otimes r)=\sum_{i\in I}A_is\otimes R_ir\in V\otimes\mathcal H_R$.
    Each $A_iS$ has dimension at most $\dim S$.
\end{proof}

\begin{theorem}[Contraction at distinct spectral cutoffs]
    \label{thm:spectral_projector_distinct_cutoffs}
    \lean{Submodule.IsSpectralProjector.contract_between_cutoffs}
    \leanok
    \uses{def:spectral_projector}
    Let $K$ be a spectral projector as in
    Definition~\ref{def:spectral_projector} for $(H,\eta_0,\eta_1)$, and let $c$
    satisfy $h_j>c\Longrightarrow h_j\ge\eta_1$ for every eigenvalue
    $h_j$. Then $K$ preserves and expands norms on
    $T_-=\operatorname{span}\{b_j:h_j\le\eta_0\}$, and preserves
    and contracts squared norms by $\Delta$ on
    $T_+^\perp$, where $T_+=\operatorname{span}\{b_j:h_j\le c\}$.
    In particular, $c=\eta_1$ is admissible without a spectral gap.
\end{theorem}

\begin{proof}
    \leanok
    \uses{def:spectral_projector}
    Write $\hat x_j=\langle b_j,x\rangle$. Since $Kb_j=k_jb_j$ with $k_j$
    real, Parseval's identity gives
    \begin{align}
        \widehat{(Kx)}_j=k_j\hat x_j,\qquad
        \lVert x\rVert^2=\sum_j\lvert\hat x_j\rvert^2,\qquad
        \lVert Kx\rVert^2=\sum_jk_j^2\lvert\hat x_j\rvert^2.\notag
    \end{align}
    A vector $x$ lies in $T_-$ exactly when $\hat x_j=0$ whenever
    $h_j>\eta_0$, and in $T_+^\perp$ exactly when $\hat x_j=0$ whenever
    $h_j\le c$. Multiplication by $k_j$ preserves both vanishing conditions,
    so $K$ preserves $T_-$ and $T_+^\perp$. For $t\in T_-$, every nonzero
    coefficient has $h_j\le\eta_0$, hence $k_j^2\ge1$ and
    $\lVert Kt\rVert^2\ge\lVert t\rVert^2$. For $q\in T_+^\perp$, every
    nonzero coefficient has $h_j>c$, hence $h_j\ge\eta_1$, $k_j^2\le\Delta$
    and $\lVert Kq\rVert^2\le\Delta\lVert q\rVert^2$.
\end{proof}

\begin{theorem}[Viability reduction between spectral cutoffs]
    \label{thm:viable_spectral_distinct_cutoffs}
    \lean{Submodule.IsViable.spectralProjector_between_cutoffs}
    \leanok
    \uses{def:subspace_viable}
    Let $\eta_0\le\eta_1$ and let $K=\sum_{i\in I}A_i\otimes R_i$
    be a spectral projector as in
    Definition~\ref{def:spectral_projector} for these endpoints, with $\Delta\ge0$.
    If $S$ is $\delta$-viable for
    $T_+=\operatorname{span}\{b_j:h_j\le\eta_1\}$ and $\delta<1$,
    then $V=\sum_{i\in I}A_iS$ has dimension at most $|I|\dim S$ and
    is $\Delta/(1-\delta)^2$-viable for
    $T_-=\operatorname{span}\{b_j:h_j\le\eta_0\}$.
    No absence of spectrum in $(\eta_0,\eta_1)$ is assumed.
    This gives the error-reduction step of
    \cite[Theorem~4.8, lines 1027--1041]{Arad2016RigorousRG}
    for a supplied finite tensor decomposition.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:viable_nested_tensor_decomposition,
        thm:spectral_projector_distinct_cutoffs}
    Use $c=\eta_1$ in the preceding contraction statement and apply the
    nested-target tensor estimate.
\end{proof}
