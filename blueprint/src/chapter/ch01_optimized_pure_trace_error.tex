\section{Optimized pure-state error and tensor powers}

The scalar error below is the expression used in the discussion of optimized
trace-norm error in \cite{Schuch2008Entropy}
(arXiv:0705.0292v2, lines 160--173). The trigonometric expression there is
restricted to angles between zero and $\pi/4$.

\begin{definition}[Scalar optimized error]
    \label{def:optimized_pure_trace_error}
    \lean{Real.optimizedPureTraceError, Real.optimizedPureTraceError_nonneg,
        Real.optimizedPureTraceError_sq,
        Real.half_le_of_optimizedPureTraceError_lt_one}
    \leanok
    For $f\in[0,1]$, define
    \begin{align}
        T(f) & =\begin{cases}
                    2\sqrt{f(1-f)}, & f\ge\tfrac12, \\
                    1,              & f<\tfrac12.
                \end{cases}
        \label{eq:optimized_error_scalar}
    \end{align}
    Thus $T(f)\ge0$, and $T(f)<1$ implies $f\ge\tfrac12$.
    On the latter interval, $T(f)^2=4f(1-f)$.
\end{definition}

\begin{lemma}[Monotonicity of the optimized error]
    \label{lem:optimized_pure_error_monotone}
    \lean{Real.optimizedPureTraceError_antitoneOn}
    \leanok
    \uses{def:optimized_pure_trace_error}
    If $0\le f\le g\le1$, then $T(g)\le T(f)$.
\end{lemma}
\begin{proof}\leanok
    \uses{def:optimized_pure_trace_error}
    The function is constant on $[0,1/2]$. On $[1/2,1]$, the identity
    $f(1-f)-g(1-g)=(g-f)(f+g-1)\ge0$ gives the result.
    The two expressions agree at $1/2$.
\end{proof}

\begin{theorem}[Amplification below saturation]
    \label{thm:optimized_pure_error_power_small}
    \lean{Real.sqrt_mul_optimizedPureTraceError_le_pow_of_lt_one}
    \leanok
    \uses{def:optimized_pure_trace_error}
    Let $f\in[0,1]$ and let $K\ge1$ be an integer. If $T(f^K)<1$, then
    $\sqrt{K/4}\,T(f)\le T(f^K)$.
\end{theorem}
\begin{proof}\leanok
    \uses{def:optimized_pure_trace_error}
    Put $g=f^K$. The error hypothesis gives $g\ge\tfrac12$, and
    $g\le f\le1$. Since $f^i\ge f^K$ for $0\le i<K$, the geometric sum gives
    \begin{align*}
        Kg(1-f)\le (1-f)\sum_{i=0}^{K-1}f^i=1-g.
    \end{align*}
    Moreover $4g^2\ge1\ge f$. Multiplying the preceding inequality by
    $4g$ therefore gives $Kf(1-f)\le4g(1-g)$.
    Substituting the expressions for $T(f)^2$ and $T(g)^2$, then taking
    nonnegative square roots, proves the assertion.
\end{proof}

\begin{theorem}[Scalar tensor-power estimate]
    \label{thm:optimized_pure_error_power}
    \lean{Real.sqrt_mul_optimizedPureTraceError_le_pow_of_sq_le}
    \leanok
    \uses{def:optimized_pure_trace_error}
    Let $f\in[0,1]$ and let $K\ge1$ be an integer. If $T(f)^2\le2/K$, then
    $\sqrt{K/8}\,T(f)\le T(f^K)$.
    This is the scalar inequality underlying
    \cite{Schuch2008Entropy}, arXiv:0705.0292v2, lines 472--474.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:optimized_pure_error_power_small,def:optimized_pure_trace_error}
    If $T(f^K)<1$, apply Theorem~\ref{thm:optimized_pure_error_power_small}
    and use $K/8\le K/4$.
    Otherwise $T(f^K)\ge1$, whereas the hypothesis gives
    $(\sqrt{K/8}\,T(f))^2\le1/4\le1$.
    Nonnegativity completes the proof.
\end{proof}

\begin{corollary}[Repeating a factor of maximal fidelity]
    \label{cor:optimized_pure_error_max_factor}
    \lean{Real.exists_optimizedPureTraceError_pow_le_prod}
    \leanok
    \uses{lem:optimized_pure_error_monotone}
    Let $K\ge1$ and $f_1,\ldots,f_K\in[0,1]$. There is an index $i$ such that
    $T(f_i^K)\le T(\prod_{j=1}^K f_j)$.
\end{corollary}
\begin{proof}\leanok
    \uses{lem:optimized_pure_error_monotone}
    Choose $i$ with $f_i$ maximal. Then $\prod_j f_j\le f_i^K$,
    and both quantities belong to $[0,1]$. Apply monotonicity.
\end{proof}

\section{Product replacement in subspaces}

The following argument gives the Hilbert-space step in the product replacement
proof of \cite{Schuch2008Entropy}, arXiv:0705.0292v2, lines 436--470.

\begin{theorem}[Product replacement in two subspaces]
    \label{thm:product_overlap_subspaces}
    \lean{TensorProduct.exists_unit_product_overlap_ge_of_subspaces}
    \leanok
    Let $E,F$ be complex inner product spaces and let $U\subseteq E$ and
    $V\subseteq F$ admit orthogonal projections. Given a unit vector
    $\Phi\in U\otimes V$ and arbitrary $a\in E$, $b\in F$, there are
    unit vectors $u\in U$, $v\in V$ such that
    $|\langle a\otimes b,\Phi\rangle|
        \le |\langle a,u\rangle|\,|\langle b,v\rangle|$.
\end{theorem}
\begin{proof}\leanok
    Let $p=P_Ua$ and $q=P_Vb$. Orthogonality and Cauchy--Schwarz give
    $|\langle a\otimes b,\Phi\rangle|
        =|\langle p\otimes q,\Phi\rangle|\le\|p\|\,\|q\|$.
    When $p,q$ are nonzero, take $u=p/\|p\|$ and $v=q/\|q\|$.
    If a projection vanishes, choose any unit vector in the corresponding
    subspace. Such a vector exists because the unit vector $\Phi$ excludes
    either subspace being zero. These choices attain the upper bound.
\end{proof}

\begin{theorem}[Product replacement in column and row spaces]
    \label{thm:product_overlap_matrix_ranges}
    \lean{Matrix.exists_unit_product_overlap_ge}
    \leanok
    \uses{thm:product_overlap_subspaces}
    Let $C=(C_{ij})$ be a finite complex matrix with
    $\sum_{i,j}|C_{ij}|^2=1$. For arbitrary vectors $a,b$ of the corresponding
    dimensions, there are unit vectors $u\in\operatorname{range} C$ and
    $v\in\operatorname{range} C^{\mathsf T}$ satisfying
    \begin{align}
        \label{eq:product_overlap_matrix_ranges}
        \left|\sum_{i,j}\overline{a_i b_j}C_{ij}\right|
        \le |\langle a,u\rangle|\,|\langle b,v\rangle|.
    \end{align}
\end{theorem}
\begin{proof}\leanok
    \uses{thm:product_overlap_subspaces}
    Write $U=\operatorname{range} C$ and $V=\operatorname{range} C^{\mathsf T}$.
    If $e_i$ are the coordinate vectors and $r_i=C^{\mathsf T}e_i$, then
    $\sum_iP_Ue_i\otimes r_i$ has coefficient matrix $C$: its $j$th column
    is $P_U(Ce_j)=Ce_j$. The canonical tensor coordinates preserve norms
    and inner products. Apply Theorem~\ref{thm:product_overlap_subspaces}.
\end{proof}

\begin{theorem}[Trace norm of a pure density]
    \label{thm:pure_density_trace_norm}
    \lean{Matrix.rectangularTraceNorm_euclideanOuterProduct_self}
    \leanok
    \uses{def:trace_norm}
    For every vector $\psi$ in a finite-dimensional complex Hilbert space,
    $\lVert\ketbra{\psi}{\psi}\rVert_{\tr}=\lVert\psi\rVert^2$.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:trace_norm_abs}
    The pure density is positive semidefinite, so its absolute value is
    itself. Its trace is $\langle\psi,\psi\rangle=\lVert\psi\rVert^2$.
\end{proof}

\begin{theorem}[Weighted pure-projection trace norm]
    \label{thm:weighted_pure_trace_norm}
    \lean{Matrix.re_trace_abs_vecMulVec_sub_smul_eq,
        Matrix.rectangularTraceNorm_vecMulVec_sub_smul_eq}
    \leanok
    \uses{def:trace_norm}
    Let $\psi,\phi$ be unit vectors in a finite-dimensional complex Hilbert
    space, and put $P=\ketbra{\psi}{\psi}$,
    $Q=\ketbra{\phi}{\phi}$ and $f=|\braket{\psi}{\phi}|^2$.
    For every real $t\ge0$,
    \begin{align}
        \lVert P-tQ\rVert_{\tr}
         & =\sqrt{(1-t)^2+4t(1-f)}.
        \label{eq:optimized_error_weighted}
    \end{align}
    This is an auxiliary identity for the optimized pure-state error
    in \cite{Schuch2008Entropy}, arXiv:0705.0292v2, lines 139 and 160--173.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:trace_norm_abs}
    Put $X=P-tQ$, $b=1-t$, $c=t(1-f)$,
    $\Delta=b^2+4c$ and $N=2X^2-bX$.
    The projection identities $P^2=P$, $Q^2=Q$, $PQP=fP$ and $QPQ=fQ$ give
    \begin{align}
        X^3 & =bX^2+cX, \label{eq:optimized_error_cubic} \\
        N^2 & =\Delta X^2.
        \label{eq:optimized_error_square}
    \end{align}
    If $Y=(1+t)P-2tQ$, then
    $(1+t)N=Y^2+t(1-t)^2Q\ge0$.
    Thus $N\ge0$, since $Y$ is Hermitian, $Q\ge0$ and $1+t>0$.
    Also
    \begin{align}
        \tr(X^2) & =1+t^2-2tf, \notag \\
        \tr N    & =\Delta.
        \label{eq:optimized_error_traces}
    \end{align}
    If $\Delta>0$, (\ref{eq:optimized_error_square}) and positivity show
    that the positive square root of $X^2=X^\dagger X$ is $N/\sqrt\Delta$.
    Theorem~\ref{thm:trace_norm_abs} and (\ref{eq:optimized_error_traces}) give
    $\lVert X\rVert_{\tr}=\tr N/\sqrt\Delta=\sqrt\Delta$.
    If $\Delta=0$, the inequalities $0\le f\le1$ and $t\ge0$ give
    $t=f=1$, and (\ref{eq:optimized_error_cubic}) gives $X^3=0$.
    Hermiticity implies $(X^2)^\dagger X^2=X^4=0$, so $X^2=0$;
    applying the same argument to $X^\dagger X$ gives $X=0$.
    The asserted formula follows in both cases.
\end{proof}

\begin{theorem}[Attained optimized pure-state error]
    \label{thm:optimized_pure_trace_minimum}
    \lean{Matrix.isLeast_optimizedPureTraceError}
    \leanok
    \uses{def:trace_norm,def:optimized_pure_trace_error}
    Let $\psi,\phi$ be nonzero vectors in a finite-dimensional complex
    Hilbert space. Put $P=\ketbra{\psi}{\psi}$,
    $Q=\ketbra{\phi}{\phi}$,
    $a=\lVert\psi\rVert^2$, $b=\lVert\phi\rVert^2$ and
    $f=|\braket{\psi}{\phi}|^2/(ab)$.
    Then
    \begin{align}
        \inf_{\alpha\in\C}
        \frac{\lVert P-\alpha Q\rVert_{\tr}}{\lVert P\rVert_{\tr}}
        =T(f)
         & =\begin{cases}
                2\sqrt{f(1-f)}, & f\ge\tfrac12, \\
                1,              & f<\tfrac12.
            \end{cases}
        \label{eq:optimized_error_minimum}
    \end{align}
    The infimum is attained.
    For $f=\cos^2\theta$ with $0\le\theta\le\pi/4$, this is the
    optimized trace-norm error $\sin(2\theta)$ of
    \cite{Schuch2008Entropy}, arXiv:0705.0292v2, lines 139 and 160--173.
\end{theorem}
\begin{proof}\leanok
    \uses{thm:weighted_pure_trace_norm,thm:pure_density_trace_norm,thm:source_branch_sum,
        thm:random_trace_duality}
    Since $a,b>0$, the vectors $u=\psi/\sqrt a$ and
    $v=\phi/\sqrt b$ are unit vectors. Write $P_u=\ketbra{u}{u}$ and
    $Q_v=\ketbra{v}{v}$. Then $P=aP_u$, $Q=bQ_v$,
    $\lVert P\rVert_{\tr}=a$, and $f=|\braket{u}{v}|^2\in[0,1]$.
    Homogeneity reduces the normalized error to
    $\lVert P_u-\gamma Q_v\rVert_{\tr}$, where $\gamma=\alpha b/a$.
    For a nonnegative real coefficient $t$, (\ref{eq:optimized_error_weighted})
    and the identity
    \begin{align}
        (1-t)^2+4t(1-f)
        =(t-(2f-1))^2+4f(1-f)
        \label{eq:optimized_error_completion}
    \end{align}
    show that the minimum is $T(f)$ and is attained.
    By (\ref{eq:optimized_error_completion}), take $t=2f-1$ when
    $f\ge\tfrac12$, and $t=0$ when $f<\tfrac12$.
    In particular, $T(f)\le1$, by testing $t=0$.

    It remains to allow arbitrary complex $\gamma$.
    Set $A=P_u-\gamma Q_v$ and $x=\re\gamma$.
    The triangle inequality, homogeneity and invariance under adjoints give
    \begin{align}
        \lVert P_u-xQ_v\rVert_{\tr}
        =\left\lVert\frac{A+A^\dagger}{2}\right\rVert_{\tr}
        \le\frac{\lVert A\rVert_{\tr}+\lVert A^\dagger\rVert_{\tr}}2
        =\lVert A\rVert_{\tr}.
        \label{eq:optimized_error_hermitian}
    \end{align}
    If $x\ge0$, (\ref{eq:optimized_error_hermitian}) and the real-coefficient minimum give
    $T(f)\le\lVert A\rVert_{\tr}$.
    If $x<0$, trace-norm duality with the identity contraction gives
    \begin{align}
        \lVert A\rVert_{\tr}
        \ge|\tr A|\ge\re(\tr A)=1-x\ge1\ge T(f).
        \label{eq:optimized_error_trace}
    \end{align}
    By (\ref{eq:optimized_error_hermitian}) and (\ref{eq:optimized_error_trace}),
    complex coefficients cannot lower the real-coefficient minimum.
    Positive rescaling carries an attaining nonnegative real coefficient
    to one for the original vectors. This is also an allowed complex
    coefficient, so the stated infimum is attained.
    Here $\alpha$ multiplies the density operator $Q$.
    Rescaling the vector by $\phi\mapsto\lambda\phi$ instead produces
    density coefficient $|\lambda|^2$. Zero is among the allowed coefficients.
\end{proof}
