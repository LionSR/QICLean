\section{Entropy bounds for canonical spectral tails}
\label{sec:entropy_spectral_tail}

Let $\sigma$ be a positive semidefinite matrix of trace one on a finite
index set $I$, and write $S(\sigma)=-\sum_i\lambda_i\log\lambda_i$ with
$0\log0=0$. Let $Q_t=\mathbf 1_{[t,\infty)}(\sigma)$.
These estimates give the finite-dimensional entropy step in
\cite[Section~4, constant-tail ranks]{OpenAI2026PolynomialPEPS}.
% The global-gap area law giving the entropy bound and the actual filtered-state
% closeness estimate remain separate obligations.

\begin{theorem}[Entropy Markov bound]
\label{thm:entropy_spectral_tail}
\lean{Matrix.IsHermitian.trace_complement_spectralProjectionGE,
Matrix.PosSemidef.trace_complement_spectralProjectionGE_le_entropy,
Matrix.PosSemidef.spectralProjectionGE_exp_entropy_bounds}
\leanok
For $0<t<1$ the complementary mass satisfies
\begin{align}
 \tr((\Id-Q_t)\sigma)
 &=\sum_{\lambda_i<t}\lambda_i
 \le\frac{S(\sigma)}{-\log t}.
\end{align}
If $S(\sigma)\le M$, $M>0$ and $c>0$, then at $t=e^{-cM}$,
$\rank Q_t\le e^{cM}$ and $\tr((\Id-Q_t)\sigma)\le1/c$.
\end{theorem}
\begin{proof}
\leanok
\uses{thm:closed_threshold_rank_trace}
Every eigenvalue lies in $[0,1]$. If $0<\lambda_i<t$, then
$\lambda_i(-\log t)\le-\lambda_i\log\lambda_i$; at zero both sides vanish.
The other entropy summands are nonnegative. The rank estimate follows from
$t\rank Q_t\le\tr\sigma=1$. Eigenvalues equal to $t$ belong to the head.
\end{proof}

\begin{theorem}[Zero entropy]
\label{thm:entropy_spectral_tail_zero}
\lean{Matrix.PosSemidef.spectralProjectionGE_one_of_entropy_nonpos}
\leanok
If $S(\sigma)\le0$, then $\rank Q_1\le1$ and
$\tr((\Id-Q_1)\sigma)=0$.
\end{theorem}
\begin{proof}
\leanok
Each entropy summand is nonnegative, so each vanishes. Since
$-s\log s>0$ for $0<s<1$, every eigenvalue is zero or one.
\end{proof}

\begin{theorem}[Transfer with a fixed projection]
\label{thm:entropy_spectral_tail_transfer}
\lean{Matrix.re_trace_mul_sub_le_traceDistance,
Matrix.PosSemidef.trace_complement_spectralProjectionGE_le_entropy_add_distance}
\leanok
Let $\rho,\sigma$ be Hermitian matrices of equal trace and $0\le E\le\Id$.
Then $\Re\tr(E\rho)-\Re\tr(E\sigma)\le\delta(\rho,\sigma)$, where
$\delta(\rho,\sigma)=\frac12\|\rho-\sigma\|_1$.
In particular, for density $\sigma$, Hermitian trace-one $\rho$ and $0<t<1$,
\begin{align}
 \Re\tr((\Id-Q_t)\rho)
 \le\frac{S(\sigma)}{-\log t}+\delta(\rho,\sigma).
\end{align}
No commutation with $\rho$ is required.
\end{theorem}
\begin{proof}
\leanok
\uses{lem:trace_distance_jordan,thm:entropy_spectral_tail}
Write $\rho-\sigma=H_+-H_-$. Positivity gives
$\tr(EH_-)\ge0$ and $\tr(EH_+)\le\tr H_+=\delta(\rho,\sigma)$.
Apply this estimate to $E=\Id-Q_t$ and use the entropy bound.
\end{proof}

\begin{theorem}[Constant tail and regulator consequence]
\label{thm:entropy_spectral_tail_regulator}
\lean{Matrix.PosSemidef.spectralProjectionGE_constant_tail,
Matrix.PosSemidef.trace_shiftedRegulator_le_quarter_of_entropy}
\leanok
Suppose $S(\sigma)\le M$, $M>0$, $\rho$ is Hermitian of trace one and
$\|\rho-\sigma\|_1\le1/16$. Then $Q=Q_{e^{-16M}}$ satisfies
$\rank Q\le e^{16M}$ and $\Re\tr((\Id-Q)\rho)\le1/8$.
If moreover $\rho,x\ge0$ commute, $C>0$, $u\ge0$,
$M=C(1+u)$, $R\ge(16C+\log8)(1+u)$, and $\rho\le x+e^{-R}\Id$, then
\begin{align}
 \tr\bigl(e^{-R}\rho(x+e^{-R}\Id)^{-1}\bigr)\le\frac14.
\end{align}
\end{theorem}
\begin{proof}
\leanok
\uses{thm:entropy_spectral_tail,thm:entropy_spectral_tail_transfer,
thm:patch_regulator_quarter}
The original-state tail is at most $1/16$. Transfer contributes at most
$1/32$, hence the weaker bound $1/8$ follows. Apply the positive-regulator
estimate with the same $Q$ and rank exponent $16C$.
\end{proof}

\subsection{Rank-constrained approximation}

\begin{theorem}[Spectral tail lower bound for trace-norm approximation]
  \label{thm:spectral_tail_trace_norm}
  \lean{Matrix.PosSemidef.sum_descEigenvalue_tail_le_traceNorm,
    Matrix.PosSemidef.sum_descEigenvalue_tail_le_half_traceNorm}
  \leanok
  \uses{def:ky_fan_norm,def:trace_norm}
  Let $\rho\ge0$ act on an $n$-dimensional Hilbert space, with ordered
  eigenvalues $\lambda_1\ge\cdots\ge\lambda_n$. Let $\sigma$ be Hermitian
  with $\operatorname{rank}\sigma\le D$, where $D\ge0$ is an integer.
  Set $\epsilon(D)=\sum_{i=D+1}^n\lambda_i$, with an empty sum interpreted
  as zero. Then
  \begin{align}
    \epsilon(D)&\le\lVert\rho-\sigma\rVert_1.
  \end{align}
  If $\tr\rho=\tr\sigma$, the sharper estimate
  $\epsilon(D)\le\frac12\lVert\rho-\sigma\rVert_1$ holds.
  No normalization or strict positivity is required. For density matrices,
  the first bound is the spectral estimate used in the approximation lower
  bound of Schuch, Wolf, Verstraete, and Cirac~\cite{Schuch2008EntropyScaling}.
\end{theorem}

\begin{proof}\leanok
  \uses{thm:ky_fan_upper_bound,lem:trace_distance_jordan,
    lem:trace_norm_jordan,lem:psd_trace_pairing_bounds}
  If $D\ge n$, the tail vanishes. Otherwise let $P$ be the support
  projection of $\sigma$ and put $Q=1-P$. Ky Fan's maximum principle and
  positivity of the eigenvalues give
  \begin{align}
    \tr(P\rho)&\le\sum_{i=1}^{\operatorname{rank}\sigma}\lambda_i
      \le\sum_{i=1}^D\lambda_i, &
    \epsilon(D)&\le\tr(Q\rho)=\tr\bigl(Q(\rho-\sigma)\bigr).
  \end{align}
  Writing $H=\rho-\sigma=H_+-H_-$ and using $0\le Q\le1$, we obtain
  $\tr(QH)\le\tr H_+\le\tr H_++\tr H_-=\lVert H\rVert_1$.
  When $\tr H=0$, the two Jordan parts have equal trace, so
  $\tr H_+=\frac12\lVert H\rVert_1$.
\end{proof}

\begin{theorem}[Stability of ordered spectral tails]
  \label{thm:spectral_tail_stability}
  \lean{Matrix.IsHermitian.sum_descEigenvalue_tail_le_add_half_traceNorm,
    Matrix.IsHermitian.sum_descEigenvalue_tail_le_add_traceDistance,
    Matrix.traceDistance_eq_rectangularTraceNorm}
  \leanok
  \uses{def:ky_fan_norm,def:trace_norm}
  For Hermitian matrices $\rho,\sigma$ of equal trace on the same finite
  Hilbert space, let $\epsilon_\rho(D)$ and $\epsilon_\sigma(D)$ denote the
  sums of the eigenvalues after the first $D$, in decreasing order. Then
  \begin{align}
    \epsilon_\rho(D)&\le\epsilon_\sigma(D)
      +\frac12\lVert\rho-\sigma\rVert_1.
  \end{align}
  For density matrices, this estimate transfers a truncation bound from a
  nearby state, as needed in the smooth R\'enyi entropy argument
  of~\cite{Schuch2008EntropyScaling}.
\end{theorem}

\begin{proof}\leanok
  \uses{thm:ky_fan_achievability,thm:ky_fan_upper_bound,
    lem:trace_distance_jordan,lem:psd_trace_pairing_bounds}
  The assertion is immediate when $D$ is at least the dimension.
  Otherwise choose a rank-$D$ spectral projection $P$ onto the largest
  eigenvalues of $\sigma$ and set $Q=1-P$. Ky Fan's principle gives
  $\epsilon_\rho(D)\le\tr(Q\rho)$, whereas
  $\epsilon_\sigma(D)=\tr(Q\sigma)$. Since $0\le Q\le1$, the Jordan
  decomposition and equality of traces imply
  $\tr\bigl(Q(\rho-\sigma)\bigr)\le\frac12\lVert\rho-\sigma\rVert_1$.
\end{proof}
