\chapter{Replica Entropy Metrics}\label{ch:replica_metrics}

The symmetric-replica argument of the two-dimensional area
law~\cite{OpenAI2026AreaLaw} compares the label observable of a subsystem with
a positive operator built from one label function shared by every subsystem.
Fix the dimension $d\ge1$ of the whole one-copy system and $0<t<1/4$. Pad each
label $\lambda\vdash k$ to length $d$, put $l_i=\lambda_i+d-i$ for
$1\le i\le d$, and set
\[
    c=d+(1+t)\binom d2,\qquad
    w_k(\lambda)=\frac{\Gamma(c)}{\Gamma(c+tk)}
        \prod_{i=1}^d\frac{\Gamma(1+tl_i)}{\Gamma(1+t(d-i))}.
\]
The metric of a subsystem $Q$ is $W_{Q,k}=\sum_\lambda w_k(\lambda)\pi^\lambda_{Q,k}$.
This chapter proves the scalar properties of $w_k$, and from them the operator
statements of the paper's Lemma~6.2 other than the integral representation of
$W_{Q,k}$.

\section{Ratios of Gamma values}

\begin{lemma}[Wendel's inequalities]
    \label{lem:wendel_gamma_ratio}
    \lean{Real.Gamma_add_le_rpow_mul_Gamma, Real.mul_Gamma_le_rpow_mul_Gamma_add,
        Real.Gamma_add_div_Gamma_le_rpow, Real.div_rpow_le_Gamma_add_div_Gamma,
        Real.rpow_sub_le_Gamma_add_div_Gamma}
    \leanok
    For $x>0$ and $0\le s\le1$,
    \[
        x(x+s)^{s-1}\le\frac{\Gamma(x+s)}{\Gamma(x)}\le x^s .
    \]
    If moreover $x\ge1$, then $x^s-s\le\Gamma(x+s)/\Gamma(x)$.
\end{lemma}

\begin{proof}\leanok
    Write $x+s=(1-s)x+s(x+1)$ and $x+1=s(x+s)+(1-s)(x+s+1)$, apply
    log-convexity of $\Gamma$ to each, and use $\Gamma(y+1)=y\Gamma(y)$. For
    the last claim, $(x+s)^{s-1}\ge x^s/(x+s)$, so the lower bound is at least
    $x^s-sx^s/(x+s)\ge x^s-s$ when $x\ge1$.
\end{proof}

\begin{lemma}[Stirling estimate for the Gamma function]
    \label{lem:log_gamma_stirling}
    \lean{Real.abs_log_Gamma_one_add_sub_le}
    \leanok
    \uses{lem:wendel_gamma_ratio, lem:stirling_log_factorial}
    For $u\ge0$,
    $|\log\Gamma(1+u)-(u\log u-u)|\le3\log(u+2)+3$.
\end{lemma}

\begin{proof}\leanok
    Let $n=\lfloor u\rfloor$ and $s=u-n$. Wendel's inequalities at $x=n+1$
    give $|\log\Gamma(1+u)-\log n!|\le\log(u+2)$. The Stirling bounds control
    $\log n!-(n\log n-n)$, and $u\log u-u$ differs from $n\log n-n$ by at most
    $\log(u+2)+2$.
\end{proof}

\section{The common label function}

\begin{lemma}[Positivity, normalization and marked ratio]
    \label{lem:replica_weight_ratio}
    \lean{Partition.replicaConst, Partition.replicaWeight, Partition.replicaRatio,
        Partition.replicaWeight_pos, Partition.replicaWeight_zero,
        Partition.replicaWeight_update_succ}
    \leanok
    The label function is positive, and $w_0=1$. If $\lambda\vdash k$ is
    obtained from $\nu\vdash k-1$ by adding a box in row $i$, then
    $w_k(\lambda)=r_{\lambda,i}\,w_{k-1}(\nu)$ with
    \[
        r_{\lambda,i}=\frac{\Gamma(c+t(k-1))}{\Gamma(c+tk)}
            \frac{\Gamma(1+tl_i)}{\Gamma(1+t(l_i-1))},
    \]
    where $l_i$ is the shifted part of $\lambda$.
\end{lemma}

\begin{proof}\leanok
    Every argument of $\Gamma$ is positive. Adding the box raises the number of
    copies by one and the $i$-th shifted part by one, leaving the other shifted
    parts unchanged.
\end{proof}

\begin{lemma}[Entropy asymptotics of the label function]
    \label{lem:replica_weight_entropy}
    \lean{Partition.exists_abs_log_replicaWeight_add_le}
    \leanok
    \uses{lem:log_gamma_stirling, lem:replica_weight_ratio}
    There is a constant $C_{d,t}$ such that every $\lambda\vdash k$ of length at
    most $d$ satisfies
    $|\log w_k(\lambda)+tkH(\lambda/k)|\le C_{d,t}\log(k+2)$.
\end{lemma}

\begin{proof}\leanok
    Apply the Stirling estimate to each of the $d+2$ Gamma factors. The terms
    containing $\log t$ cancel up to a constant because
    $\sum_il_i=k+\binom d2$. The shifts $0\le l_i-\lambda_i\le d-1$ change
    $\sum_il_i\log l_i$ by $O_d(\log(k+2))$, and
    $kH(\lambda/k)=k\log k-\sum_i\lambda_i\log\lambda_i$.
\end{proof}

Together with the entropy asymptotics of the dimension formula, this gives
$w_k(\lambda)=d_\lambda^{-t}$ up to factors polynomial in $k$, which is the
label-wise form of the comparison of $W_{Q,k}$ with $e^{-tF_{Q,k}}$.

\section{The marked ratio}

\begin{lemma}[Size of the marked ratio]
    \label{lem:replica_ratio_bounds}
    \lean{Partition.replicaRatio_eq_div, Partition.replicaRatio_wendel,
        Partition.exists_replicaRatio_bounds}
    \leanok
    \uses{lem:wendel_gamma_ratio, lem:replica_weight_ratio}
    Let $0<t<1$. There is $C\ge1$ such that, for $k\ge1$ and $1\le l\le k+d$,
    the ratio $r(k,l)$ of the previous lemma satisfies
    $C^{-1}(l/k)^t\le r(k,l)\le C$.
\end{lemma}

\begin{proof}\leanok
    The ratio is a quotient of two Wendel ratios, with $x=1+t(l-1)$ and
    $y=c+t(k-1)$, both at least one. Wendel's inequalities give
    $(1-t)(x/y)^t\le r(k,l)\le(x/y)^t/(1-t)$. On the stated range,
    $tl\le x\le(1+d)y$ and $y\le(c+t)k$.
\end{proof}

\begin{lemma}[Limit of the marked ratio]
    \label{lem:replica_ratio_limit}
    \lean{Real.abs_rpow_sub_rpow_le, Partition.exists_abs_replicaRatio_sub_le}
    \leanok
    \uses{lem:replica_ratio_bounds}
    Let $0<t<1$. There is a constant $C$ such that, for $k\ge1$ and
    $1\le l\le k+d$,
    \[
        \bigl|r(k,l)-\max\bigl((l-d)/k,0\bigr)^t\bigr|\le Ck^{-t}.
    \]
\end{lemma}

\begin{proof}\leanok
    Wendel's inequalities give $|r(k,l)-(x/y)^t|\le Ck^{-t}$, since $y\ge tk$
    and $(x/y)^t\le1+d$. Moreover $|x/y-l/k|=O(1/k)$ and
    $|l/k-\max((l-d)/k,0)|\le d/k$, and $|a^t-b^t|\le|a-b|^t$ for $a,b\ge0$.
\end{proof}

On a removable branch the star operator has eigenvalue $(\lambda_i-i)/k=(l_i-d)/k$,
so this lemma is the scalar content of the convergence of the marked ratio of
the metrics to the positive part of the star operator raised to the power $t$.

\section{Functional calculus on an orthogonal resolution}

\begin{lemma}[Functions of an orthogonal resolution]
    \label{lem:orthogonal_resolution_cfc}
    \lean{Matrix.IsOrthogonalResolution.aeval_hom,
        Matrix.IsOrthogonalResolution.spectrum_hom_subset,
        Matrix.IsOrthogonalResolution.cfc_hom,
        Matrix.IsOrthogonalResolution.l2_opNorm_hom_le,
        Matrix.IsOrthogonalResolution.re_star_dotProduct_hom_mulVec_self,
        Matrix.IsOrthogonalResolution.sum_weight}
    \leanok
    \uses{lem:orthogonal_resolution}
    Let $P_i$ be Hermitian idempotents with $P_iP_j=\delta_{ij}P_i$ and
    $\sum_iP_i=I$, and let $g$ be real. Then $f(\sum_ig_iP_i)=\sum_if(g_i)P_i$ for
    every real function $f$. If $|c_i|\le M$ whenever $P_i\ne0$, then
    $\|\sum_ic_iP_i\|\le M$ in operator norm. For every vector $z$,
    $\|\sum_ic_iP_iz\|^2=\sum_i|c_i|^2\|P_iz\|^2$ and $\sum_i\|P_iz\|^2=\|z\|^2$.
\end{lemma}

\begin{proof}\leanok
    Polynomials act pointwise, and the spectrum lies in the range of $g$, so
    interpolating $f$ on that finite set by a polynomial proves the first claim.
    For the second, $A^\ast A=\sum_i|c_i|^2P_i\le M^2I$ and
    $\|A\|^2=\|A^\ast A\|$.
\end{proof}

\section{The metrics}

Let $\mathcal V=\bigotimes_f\C^{\iota_f}$ have dimension $d$, and let a subsystem
$Q$ be a set of factors. A label occurring on $Q^{\otimes k}$ has at most $d_Q\le d$
rows.

\begin{lemma}[Labels of a subsystem]
    \label{lem:subsystem_label_transport}
    \lean{PermutationRepresentation.labelProj_of_intertwine,
        PermutationRepresentation.labelProj_prodLeft,
        TensorPower.labelProj_subsystemPerm_eq,
        TensorPower.labelPart_eq_zero_of_subsystemPerm,
        TensorPower.card_subConfig_le}
    \leanok
    Under $\mathcal V^{\otimes k}\cong Q^{\otimes k}\otimes(Q^c)^{\otimes k}$, the
    label projector $\pi^\lambda_{Q,k}$ is the label projector of $Q^{\otimes k}$
    tensored with the identity. Consequently a label with $\pi^\lambda_{Q,k}\ne0$
    has at most $d_Q$ rows, and $d_Q\le d$.
\end{lemma}

\begin{proof}\leanok
    Intertwined permutation actions have the same group-algebra image up to
    relabelling the basis, and an action on the first factor of a product gives
    the image tensored with the identity. The row bound is the corresponding
    statement for $(\C^{d_Q})^{\otimes k}$.
\end{proof}

\begin{theorem}[Positivity, centrality and complements]
    \label{thm:replica_metric_basic}
    \lean{TensorPower.replicaMetric, TensorPower.posDef_replicaMetric,
        TensorPower.replicaMetric_zero, TensorPower.commute_replicaMetric_permOp,
        TensorPower.commute_replicaMetric_of_subset,
        TensorPower.commute_replicaMetric_of_disjoint,
        TensorPower.replicaMetric_mul_symProj_compl}
    \leanok
    \uses{lem:replica_weight_ratio, thm:schur_label_commutation}
    Let $t\ge0$ and assume every factor is nonzero. The metric
    $W_{Q,k}=\sum_\lambda w_k(\lambda)\pi^\lambda_{Q,k}$ is positive definite,
    commutes with the copy permutations of $Q$ and with the metrics of nested and
    disjoint subsystems, and $W_{Q,0}=I$. On the symmetric subspace,
    $W_{Q,k}=W_{Q^c,k}$.
\end{theorem}

\begin{proof}\leanok
    The weights are positive and the label projectors form an orthogonal
    resolution of the identity. The other claims hold for every central label
    function, by the commutation and complement properties of the label
    projectors.
\end{proof}

\begin{theorem}[Comparison with the label observable]
    \label{thm:replica_metric_comparison}
    \lean{TensorPower.exists_abs_log_replicaLabelWeight_add_le,
        TensorPower.cfc_exp_labelEntropy, TensorPower.exists_replicaMetric_comparison}
    \leanok
    \uses{lem:replica_weight_entropy, lem:hook_formula_entropy,
        lem:subsystem_label_transport, lem:orthogonal_resolution_cfc}
    Let $t>0$. There is $C$, depending only on $d$ and $t$, such that for every
    subsystem $Q$ and every $k$,
    \[
        (k+2)^{-C}e^{-tF_{Q,k}}\le W_{Q,k}\le(k+2)^Ce^{-tF_{Q,k}}.
    \]
\end{theorem}

\begin{proof}\leanok
    Both sides are functions of the label projectors, so it suffices to compare
    the coefficients of the labels that occur. Such a label has at most $d$ rows,
    $d_\lambda$ equals the dimension formula, and the entropy asymptotics of the
    dimension formula and of the label function give
    $|\log w_k(\lambda)+t\log d_\lambda|\le C\log(k+2)$.
\end{proof}

\section{The marked ratio of the metrics}

Let $k=m+1$. Extend $W_{Q,k-1}$ by the identity on the last copy, put
$R_{Q,k}=W_{Q,k}W_{Q,k-1}^{-1}$, and let $J_{Q,k}$ be the normalized star
operator of $Q$.

\begin{lemma}[Branch decomposition of the marked ratio]
    \label{lem:marked_ratio_branches}
    \lean{TensorPower.snocEquiv, TensorPower.replicaMetricFirst,
        TensorPower.replicaMetricFirst_eq, TensorPower.markedRatio,
        TensorPower.starOp_subsystemPerm_eq_hom, TensorPower.markedRatio_eq_hom,
        TensorPower.exists_branch_scalars}
    \leanok
    \uses{thm:replica_metric_basic, lem:replica_weight_ratio}
    In the joint resolution by the label $\lambda$ of all copies and the label
    $\nu$ of the first $k-1$ copies, $R_{Q,k}$ and $J_{Q,k}$ are diagonal. On a
    nonzero block, $\nu=\lambda-e_i$, the ratio acts by $r(k,l_i)$ and the star
    operator by $(l_i-d)/k$, where $1\le l_i\le k+d$ is the shifted part of
    $\lambda$ padded to length $d$.
\end{lemma}

\begin{proof}\leanok
    The extended metric is the central label function of the copy permutations
    fixing the last copy. On a nonzero block the labels are compatible, so
    $\nu=\lambda-e_i$; the star eigenvalue is the one of the Schur-label
    estimates, and the ratio of label functions is the exact marked ratio.
\end{proof}

\begin{theorem}[Forward marked ratio]
    \label{thm:marked_ratio_forward}
    \lean{TensorPower.exists_l2_opNorm_markedRatio_sub_le,
        TensorPower.exists_l2_opNorm_markedRatio_le}
    \leanok
    \uses{lem:marked_ratio_branches, lem:replica_ratio_limit, lem:replica_ratio_bounds,
        lem:orthogonal_resolution_cfc}
    Let $0<t<1$. There are constants such that, on the entire tensor space and
    for every subsystem,
    $\|R_{Q,k}-(J_{Q,k})_+^t\|\le Ck^{-t}$ and $\|R_{Q,k}\|\le C$. In particular
    $\|R_{Q,k}-(J_{Q,k})_+^t\|\to0$.
\end{theorem}

\begin{proof}\leanok
    Both operators are diagonal in the branch decomposition; apply the limit and
    the bounds of the marked ratio blockwise and the operator-norm bound on an
    orthogonal resolution.
\end{proof}

\begin{lemma}[Branch weights of a symmetric vector]
    \label{lem:symmetric_branch_weights}
    \lean{TensorPower.complAverage, TensorPower.trace_complAverage_mul,
        TensorPower.commute_complAverage, TensorPower.weight_branch,
        TensorPower.card_branch_le, TensorPower.exists_branch_data}
    \leanok
    \uses{lem:marked_ratio_branches, lem:subsystem_label_transport}
    Let $z\in\mathcal S_k$. On a branch $\nu=\lambda-e_i$,
    $d_\lambda\|\pi^\lambda\pi^\nu z\|^2=d_\nu\|\pi^\lambda z\|^2$; hence
    $\|\pi^\lambda\pi^\nu z\|^2\le2^d(l_i/k)\|\pi^\lambda z\|^2$. Each label has at
    most $d$ branches with nonzero projector.
\end{lemma}

\begin{proof}\leanok
    The average of $|z\rangle\langle z|$ over the copy permutations of $Q^c$ commutes
    with the copy permutations of $Q$, because $z$ is symmetric, and pairs with every
    operator generated by them as $k!\,|z\rangle\langle z|$; this is the matrix form
    of the permutation invariance of the $Q$ marginal. The branch-probability
    identity of the Schur-label estimates applies to it. Distinct branches of one
    label differ in their removed row.
\end{proof}

\begin{theorem}[Inverse marked ratio on symmetric vectors]
    \label{thm:marked_ratio_inverse}
    \lean{TensorPower.markedRatio_inv_eq_hom,
        TensorPower.exists_norm_markedRatio_inv_mulVec_le,
        TensorPower.exists_markedRatio_inv_sub_clipped_sq_le,
        TensorPower.exists_norm_markedRatio_inv_sub_clipped_le}
    \leanok
    \uses{lem:symmetric_branch_weights, lem:replica_ratio_bounds, lem:replica_ratio_limit}
    Let $0<t<1/2$. There is $C$ such that $\|R_{Q,k}^{-1}z\|\le C\|z\|$ for every
    $z\in\mathcal S_k$, and, for every $0<\delta<1$,
    \[
        \limsup_{k\to\infty}\sup_{z\in\mathcal S_k,\ \|z\|=1}
        \bigl\|\bigl(R_{Q,k}^{-1}-\max(J_{Q,k},\delta)^{-t}\bigr)z\bigr\|
        \le C\delta^{(1-2t)/2}.
    \]
\end{theorem}

\begin{proof}\leanok
    Both operators are diagonal in the branch decomposition. On a branch,
    $r^{-2}\|\pi^\lambda\pi^\nu z\|^2\le C(l_i/k)^{1-2t}\|\pi^\lambda z\|^2$, and summing
    over at most $d$ branches and over labels gives the first bound. For the second,
    on $J\ge\delta$ the clipped inverse differs from $r^{-1}$ by $O(k^{-t}\delta^{-2t})$.
    On $J<\delta$, $l_i/k<\delta+d/k$, and both inverses contribute at most
    $C\bigl((\delta+d/k)^{1-2t}+\delta^{-2t}(\delta+d/k)\bigr)\|\pi^\lambda z\|^2$.
    Let $k\to\infty$ and take square roots.
\end{proof}

\section{Similarity transforms of copy means}

Let $\mathcal V=\bigotimes_v\C^{n_v}$ with every $n_v\ge1$, let $P,Y,F$ be disjoint
sets of factors, and let $A_k=(W_{P,k}^{-1}W_{F,k}^{-1}W_{Y,k})^2$ be the partition
metric. For a one-copy operator $h$ supported on $P\cup Y$, write $h^{(j)}$ for $h$
acting on the $j$-th copy, $\bar h=k^{-1}\sum_jh^{(j)}$, and
$O_k=A_k^{-1/2}\bar hA_k^{1/2}$.

\begin{lemma}[One-copy operators on replicas]
    \label{lem:replica_site_operators}
    \lean{PermutationRepresentation.commute_permOp_of_apply_eq,
        TensorPower.siteOp_apply_of_isSupportedOn,
        TensorPower.commute_permOp_subsystemPerm_siteOp_of_disjoint,
        TensorPower.commute_permOp_subsystemPerm_siteOp_of_fix,
        TensorPower.permOp_copyPerm_mul_siteOp, TensorPower.siteOp_eq_reindex,
        TensorPower.l2_opNorm_siteOp_le, Entropy.IsSupportedOn.exists_sum_mul}
    \leanok
    The operator $h^{(j)}$ commutes with the copy permutations of every subsystem
    disjoint from the support of $h$, and with every copy permutation of a subsystem
    that fixes the copy $j$. A permutation $\tau$ of entire copies conjugates $h^{(j)}$
    to $h^{(\tau j)}$, and $\|h^{(j)}\|\le\|h\|$. An operator supported on a disjoint
    union $X\cup T$ is a finite sum of products $a_ib_i$ with $a_i$ supported on $X$
    and $b_i$ supported on $T$.
\end{lemma}

\begin{proof}\leanok
    The first two statements compare matrix entries: $h^{(j)}$ is nonzero only
    between configurations that agree outside the support factors of copy $j$, and
    the permutations in question preserve this relation and these factors. The
    operator $h^{(j)}$ is $h\otimes I$ after splitting off copy $j$. For the
    decomposition, expand a local operator on $X\cup T$ in matrix units.
\end{proof}

\begin{theorem}[Uniform boundedness of the similarity transform]
    \label{thm:marked_similarity_bounded}
    \lean{TensorPower.leafRoot, TensorPower.leafMetric, TensorPower.copyMean,
        TensorPower.markedSimilarity, TensorPower.sqrt_leafMetric,
        TensorPower.star_dotProduct_copyMean_eq, TensorPower.replicaMetric_conj_siteOp_mul,
        TensorPower.exists_norm_markedSimilarity_mulVec_le}
    \leanok
    \uses{lem:replica_site_operators, thm:marked_ratio_forward, thm:marked_ratio_inverse,
        thm:replica_metric_basic}
    Let $0<t<1/2$. For fixed $P,Y,F$ and $h$ there is $C$, depending on $\mathcal V$, $t$,
    the partition and $h$, such that
    $\|O_kz\|\le C\|z\|$ for every $k$ and every $z\in\mathcal S_k$.
\end{theorem}

\begin{proof}\leanok
    The square root of $A_k$ is $W_{P,k}^{-1}W_{F,k}^{-1}W_{Y,k}$, the $F$ factors
    commute with $\bar h$ and cancel, and $O_k$ preserves $\mathcal S_k$. Between
    symmetric vectors, $h^{(k)}$ may replace $\bar h$. Write $h=\sum_ib_ia_i$ with $b_i$
    supported on $Y$ and $a_i$ on $P$. Since the metric of the first $k-1$ copies
    commutes with operators on the last copy,
    \[
        W_Y^{-1}W_P(b_ia_i)^{(k)}W_P^{-1}W_Y
        =\bigl(R_Y^{-1}b_i^{(k)}R_Y\bigr)\bigl(R_Pa_i^{(k)}R_P^{-1}\bigr).
    \]
    Hence $\langle w,O_kz\rangle=\sum_i\langle R_Yb_i^{(k)\ast}R_Y^{-1}w,
    R_Pa_i^{(k)}R_P^{-1}z\rangle$ for symmetric $w,z$, in which both inverses act on
    symmetric vectors. The forward and inverse bounds on the marked ratio bound each
    term by $C_i\|w\|\|z\|$; take $w=O_kz$.
\end{proof}

\section{Unitary twirls}

The integral representation of the metrics averages over a Haar-distributed unitary
eigenbasis. The following statements show that such averages are central label functions.

\begin{lemma}[Central label functions]
    \label{lem:central_label_function}
    \lean{Matrix.exists_eq_smul_of_commute_corner,
        PermutationRepresentation.exists_eq_sum_labelProj_of_commute}
    \leanok
    \uses{lem:label_projectors}
    For a permutation action of a finite group, an operator commuting with the
    permutation operators and with every operator of their commutant is
    $\sum_\lambda c_\lambda\pi^\lambda$.
\end{lemma}

\begin{proof}\leanok
    On the corner of the matrix unit $E^\lambda_{00}$ the operator commutes with every
    compressed matrix, because the averages $\sum_jE^\lambda_{j0}BE^\lambda_{0j}$ lie in
    the commutant; hence it is a multiple of the corner projection there. The matrix
    units $E^\lambda_{i0}$ carry this multiple to the whole block.
\end{proof}

\begin{lemma}[The differentiated unitary action]
    \label{lem:unitary_commutant}
    \lean{TensorPower.rotation, TensorPower.rotation_mem_unitaryGroup,
        TensorPower.commute_diagOp_of_commute_tensorPow_rotation,
        TensorPower.commute_diagOp_of_forall_unitary,
        TensorPower.exists_eq_sum_labelProj_of_forall_unitary}
    \leanok
    \uses{thm:schur_weyl_commutant, lem:central_label_function}
    An operator on $(\C^q)^{\otimes k}$ commuting with $U^{\otimes k}$ for every unitary
    $U$ commutes with $\sum_jA^{(j)}$ for every matrix $A$. If it also commutes with the
    copy permutations, it is a central label function.
\end{lemma}

\begin{proof}\leanok
    Differentiate along the one-parameter unitary groups
    $1+(\cos\theta-1)P+(\sin\theta)J$ with
    $(P,J)=(e_{aa}+e_{bb},e_{ab}-e_{ba})$, $(e_{aa}+e_{bb},i(e_{ab}+e_{ba}))$ and
    $(e_{aa},ie_{aa})$, whose derivatives at $\theta=0$ span all matrices. By
    Schur--Weyl duality the diagonal actions generate the commutant of the copy
    permutations.
\end{proof}

\begin{theorem}[The unitary twirl]
    \label{thm:unitary_twirl}
    \lean{Matrix.isCompact_unitaryGroup, Matrix.unitaryHaar, TensorPower.unitaryTwirl,
        TensorPower.tensorPow_mul_unitaryTwirl, TensorPower.commute_permOp_unitaryTwirl,
        TensorPower.trace_labelProj_mul_unitaryTwirl,
        TensorPower.unitaryTwirl_eq_sum_labelProj}
    \leanok
    \uses{lem:unitary_commutant}
    Let $A$ commute with the copy permutations, and let
    $T(A)=\int U^{\otimes k}A(U^{\otimes k})^\ast\,dU$ over the Haar probability measure
    of the compact unitary group. Then $T(A)=\sum_\lambda c_\lambda\pi^\lambda$ with
    $c_\lambda\Tr\pi^\lambda=\Tr(\pi^\lambda A)$.
\end{theorem}

\begin{proof}\leanok
    Left invariance of Haar measure makes $T(A)$ commute with every unitary tensor
    power, and $T(A)$ commutes with the copy permutations because tensor powers and $A$
    do. The previous lemma gives the central form. The label projectors commute with
    unitary tensor powers, so $\Tr(\pi^\lambda T(A))=\Tr(\pi^\lambda A)$.
\end{proof}

\begin{lemma}[The trivial label]
    \label{lem:trivial_label}
    \lean{IrrepLabel.averagingElement, IrrepLabel.augmentation,
        IrrepLabel.exists_centralIdem_eq_averagingElement,
        PermutationRepresentation.exists_labelProj_eq_symProj}
    \leanok
    \uses{def:irrep_labels}
    The averaging element $|G|^{-1}\sum_gg$ of $\C[G]$ is one of the central idempotents
    $e_\lambda$. Hence, for every permutation action, the projector onto the invariant
    vectors is a label projector.
\end{lemma}

\begin{proof}\leanok
    The averaging element $\varepsilon$ satisfies $\varepsilon a=\operatorname{aug}(a)\varepsilon$.
    The augmentations of the $e_\lambda$ are idempotent, sum to one and have vanishing
    pairwise products, so exactly one $e_{\lambda_0}$ has $\varepsilon e_{\lambda_0}=\varepsilon$.
    In the simple block of $\lambda_0$ the central idempotent $\varepsilon e_{\lambda_0}$ is
    nonzero, hence equal to $e_{\lambda_0}$.
\end{proof}

\begin{theorem}[Coherent-state resolution]
    \label{thm:coherent_resolution}
    \lean{TensorPower.tensorVec, TensorPower.tensorPow_mulVec_tensorVec,
        TensorPower.tensorVec_mem_symmetricSubspace, TensorPower.coherentProj,
        TensorPower.unitaryTwirl_coherentProj}
    \leanok
    \uses{thm:unitary_twirl, lem:trivial_label}
    For a unit basis vector $e$, $\int P_{Ue,k}\,dU=D_k^{-1}\Pi_k$, where
    $P_{\theta,k}=|\theta^{\otimes k}\rangle\langle\theta^{\otimes k}|$, $\Pi_k$ is the
    projector onto the symmetric subspace and $D_k=\Tr\Pi_k$.
\end{theorem}

\begin{proof}\leanok
    The integral is the twirl of $P_{e,k}$, a central label function with coefficients
    fixed by $\langle e^{\otimes k},\pi^\lambda e^{\otimes k}\rangle$. The symmetric vector
    $e^{\otimes k}$ lies in the range of the trivial label projector $\Pi_k$ and is
    orthogonal to the other label projectors.
\end{proof}

\begin{theorem}[Finite-copy coherent identity]
    \label{thm:finite_husimi_identity}
    \lean{TensorPower.splitCopies, TensorPower.coherentProj_split,
        TensorPower.trace_mul_integral, TensorPower.integral_coherent_pairing}
    \leanok
    \uses{thm:coherent_resolution}
    For $\sigma$ on $k$ copies and $G$ on $m$ copies,
    $\int\Tr(\sigma P_{Ue,k})\Tr(GP_{Ue,m})\,dU=D_{k+m}^{-1}\Tr[(\sigma\otimes G)\Pi_{k+m}]$.
\end{theorem}

\begin{proof}\leanok
    $P_{\theta,k+m}=P_{\theta,k}\otimes P_{\theta,m}$; pair the coherent-state resolution
    for $k+m$ copies with $\sigma\otimes G$.
\end{proof}

\begin{definition}[The coherent measure]
    \label{def:coherent_measure}
    \lean{TensorPower.coherentIntegral, TensorPower.coherentIntegral_one,
        TensorPower.coherentIntegral_trace_coherentProj,
        TensorPower.norm_coherentIntegral_trace_coherentProj_le}
    \leanok
    \uses{thm:coherent_resolution, thm:finite_husimi_identity}
    For a density matrix $\sigma$ on the symmetric subspace of $k$ copies, the coherent
    measure is $d\mu_\sigma(\theta)=D_k\Tr(\sigma P_{\theta,k})\,d\theta$, where $\theta=Ue$
    for Haar-distributed $U$. It is a probability measure, and
    $\bigl|\int\langle\theta^{\otimes m},G\theta^{\otimes m}\rangle\,d\mu_\sigma\bigr|\le\|G\|$
    for every operator $G$ on $m$ copies.
\end{definition}

\begin{proof}\leanok
    The coherent-state resolution gives total mass $\Tr(\sigma\Pi_k)=\Tr\sigma=1$. The
    density $D_k\Tr(\sigma P_{\theta,k})$ is nonnegative, and
    $|\langle\theta^{\otimes m},G\theta^{\otimes m}\rangle|\le\|G\|$ for a unit vector
    $\theta$.
\end{proof}

\begin{definition}[Injection averages]
    \label{def:injection_average}
    \lean{TensorPower.placeSplit, TensorPower.placeOp, TensorPower.injectionAverage,
        TensorPower.l2_opNorm_injectionAverage_le}
    \leanok
    For an operator $G$ on $m$ copies and an ordered injection $\iota$ of $m$ copies into
    $k$ copies, $G_\iota$ is $G$ on the copies $\iota(1),\dots,\iota(m)$ and the identity
    elsewhere. The injection average $\mathcal T_{k,m}(G)$ is the average of $G_\iota$ over
    all ordered injections; $\|\mathcal T_{k,m}(G)\|\le\|G\|$.
\end{definition}

\begin{lemma}[Partial trace of matched swaps]
    \label{lem:matched_swaps}
    \lean{TensorPower.injectionSwap, TensorPower.trace_kronecker_mul_injectionSwap}
    \leanok
    \uses{def:injection_average}
    Let $\pi_\iota$ exchange the copy $\iota(j)$ with the copy $k+j$ for every $j$. Then
    $\Tr[(\sigma\otimes G)U(\pi_\iota)]=\Tr(\sigma G_\iota)$.
\end{lemma}

\begin{proof}\leanok
    Write configurations of $k+m$ copies as a configuration $u$ of the first $k$ copies and
    $w$ of the last $m$. The permutation $\pi_\iota$ sends $(u,w)$ to the configuration whose
    first part is $u$ with $w$ inserted at $\iota$ and whose last part is $u\circ\iota$; the
    trace is the resulting partial trace.
\end{proof}

\begin{theorem}[Uniform coherent approximation]
    \label{thm:uniform_husimi}
    \lean{Matrix.PosSemidef.norm_trace_mul_le, Equiv.Perm.card_filter_apply_eq_mul,
        TensorPower.sum_goodPerms, TensorPower.card_filter_not_goodPerms_mul_le,
        TensorPower.permOp_mul_copyKronecker, TensorPower.norm_exchangeTrace_le,
        TensorPower.factorial_mul_trace_copyKronecker_mul_symProj,
        TensorPower.norm_trace_mul_injectionAverage_sub_coherentIntegral_le}
    \leanok
    \uses{def:coherent_measure, lem:matched_swaps}
    For a density matrix $\sigma$ on the symmetric subspace of $k\ge1$ copies and an
    operator $G$ on $m$ copies,
    \[
        \left|\Tr\sigma\mathcal T_{k,m}(G)
            -\int\langle\theta^{\otimes m},G\theta^{\otimes m}\rangle\,d\mu_\sigma\right|
        \le\frac{4m^2(1+d^m)^2\|G\|}{k},
    \]
    where $d$ is the one-copy dimension.
\end{theorem}

\begin{proof}\leanok
    Expand $\Pi_{k+m}$ into permutations in the finite-copy identity. A permutation sending
    each of the last $m$ copies into the first $k$ is uniquely $\pi_\iota\gamma$ with
    $\gamma$ fixing the last $m$ copies; symmetry of $\sigma$ absorbs $\gamma$, and the
    previous lemma evaluates $\pi_\iota$. These permutations therefore contribute their
    proportion times $\Tr\sigma\mathcal T_{k,m}(G)$. The other permutations send a
    prescribed one of the last copies to a prescribed one of them for some pair, so they
    form a fraction at most $m^2/(k+m)$, and each term is at most $d^m\|G\|$ in absolute
    value because $|\Tr(\rho M)|\le\Tr\rho\,\|M\|$ for $\rho\ge0$. The normalization
    $D_k/D_{k+m}$ is eliminated with the case $G=1$, where the coherent measure has total
    mass one. For small $k$ both terms are bounded by $\|G\|$.
\end{proof}

\begin{theorem}[Products of injection averages]
    \label{thm:injection_product}
    \lean{TensorPower.placeOp_castAdd_mul_placeOp_natAdd,
        Function.Embedding.card_filter_apply_eq_mul, TensorPower.sum_filter_not_overlapPairs,
        TensorPower.card_overlapPairs_mul_le, TensorPower.norm_injectionAverage_mul_sub_le}
    \leanok
    \uses{def:injection_average}
    For operators $G$ on $m$ copies and $H$ on $n$ copies and every $k\ge1$,
    \[
        \|\mathcal T_{k,m}(G)\mathcal T_{k,n}(H)-\mathcal T_{k,m+n}(G\otimes H)\|
        \le\frac{2mn\|G\|\|H\|}{k}.
    \]
\end{theorem}

\begin{proof}\leanok
    If $\iota,\kappa$ are the restrictions of an injection $\lambda$ of $m+n$ copies, then
    $G_\iota H_\kappa=(G\otimes H)_\lambda$, and pairs of injections with disjoint images
    correspond to injections of $m+n$ copies. A pair has $\iota(a)=\kappa(b)$ for prescribed
    $a,b$ with probability $1/k$, so pairs with intersecting images form a fraction at most
    $mn/k$. These pairs and the change of normalization each cost at most that fraction
    times $\|G\|\|H\|$.
\end{proof}

\begin{lemma}[Sums of injection averages]
    \label{lem:injection_poly}
    \lean{TensorPower.placeOp_trans_perm, TensorPower.commute_symProj_injectionAverage,
        TensorPower.conjTranspose_injectionAverage, TensorPower.coherentExpect,
        TensorPower.coherentExpect_conjTranspose, TensorPower.coherentExpect_copyKronecker,
        TensorPower.norm_coherentIntegral_le, TensorPower.IsInjectionPoly,
        TensorPower.IsInjectionPoly.exists_mul,
        TensorPower.IsInjectionPoly.exists_norm_trace_sub_coherentIntegral_le}
    \leanok
    \uses{thm:uniform_husimi, thm:injection_product}
    Let $Y_k=\sum_i\mathcal T_{k,r_i}(G_i)$ for finitely many fixed operators $G_i$, with
    symbol $\psi(\theta)=\sum_i\langle\theta^{\otimes r_i},G_i\theta^{\otimes r_i}\rangle$.
    Then $Y_k$ commutes with the copy permutations, $Y_k^\ast$ has symbol
    $\overline\psi$, the product of two such sums is, up to $O(k^{-1})$ in operator norm, a
    third one whose symbol is the product of the symbols, and
    $|\Tr\sigma Y_k-\int\psi\,d\mu_\sigma|=O(k^{-1})$ uniformly in symmetric density matrices
    $\sigma$.
\end{lemma}

\begin{proof}\leanok
    Relocating a placement by a copy permutation conjugates it by that permutation, and
    relocation permutes the injections. Product vectors factor:
    $\langle\theta^{\otimes(m+n)},(G\otimes H)\theta^{\otimes(m+n)}\rangle
    =\langle\theta^{\otimes m},G\theta^{\otimes m}\rangle
    \langle\theta^{\otimes n},H\theta^{\otimes n}\rangle$. The product estimate and the trace
    estimate follow termwise from the two previous theorems.
\end{proof}

\begin{theorem}[Coherent symbol calculus]
    \label{thm:coherent_symbol}
    \lean{TensorPower.unitSphere, TensorPower.isCompact_unitSphere,
        TensorPower.HasCoherentSymbol, TensorPower.HasCoherentSymbol.add,
        TensorPower.HasCoherentSymbol.smul, TensorPower.HasCoherentSymbol.conjTranspose,
        TensorPower.HasCoherentSymbol.mul,
        TensorPower.HasCoherentSymbol.eventually_norm_trace_sub_le,
        TensorPower.HasCoherentSymbol.freeAlgebra}
    \leanok
    \uses{lem:injection_poly}
    Say that operators $X_k$ have coherent symbol $f$ if they commute with $\Pi_k$, are
    uniformly bounded on $\mathcal S_k$, $f$ is continuous on unit vectors, and for every
    $\varepsilon>0$ some sum of injection averages $Y_k$ with symbol $\psi$ satisfies
    $|f-\psi|\le\varepsilon$ on unit vectors and $\|(X_k-Y_k)\Pi_k\|\le\varepsilon$ for large
    $k$. This property is preserved by sums, scalar multiples, adjoints with the conjugate
    symbol, and products with the product symbol. Consequently, for every noncommutative
    polynomial $q$ in two variables, $q(X_k,X_k^\ast)$ has coherent symbol
    $q(f,\overline f)$, and
    \[
        \sup_\sigma\left|\Tr\sigma X_k-\int f\,d\mu_\sigma\right|\longrightarrow0
    \]
    over symmetric density matrices $\sigma$.
\end{theorem}

\begin{proof}\leanok
    For products, write
    $XX'\Pi=X\Pi(X'-Y')\Pi+(X-Y)\Pi Y'\Pi+YY'\Pi$, using that all factors commute with
    $\Pi$, and approximate $YY'$ by the previous lemma. Symbols are bounded on the compact
    unit sphere, so the symbol errors combine in the same way. For the trace limit,
    $|\Tr\sigma(X_k-Y_k)|\le\|(X_k-Y_k)\Pi_k\|$, the coherent measure is a probability measure,
    and the previous lemma treats $Y_k$. Polynomials follow by induction.
\end{proof}

\section{The gamma integral}

The integral representation of the metrics draws the eigenvalues $s$ of a density matrix with
a density proportional to $\Delta(s)\Delta(s^t)$. The following statements compute the
resulting averages of characters. Write $\delta_j=d-j$, $a_\alpha(y)=\det[y_i^{\alpha_j}]$ for
the alternant, $l$ for the shifted parts of a label, $N=\binom d2$, and
$D=d+N+t\sum_jl_j$. In place of the simplex we integrate over the positive orthant with the
weight $(\sum_ix_i)^{-a}e^{-\sum_ix_i}$; the quotients of gamma values are the same.

\begin{lemma}[The Beta integral on the half-line]
    \label{lem:beta_half_line}
    \lean{Real.integrableOn_rpow_mul_one_add_rpow, Real.integral_rpow_mul_one_add_rpow}
    \leanok
    For $a,b>0$,
    $\int_0^\infty u^{a-1}(1+u)^{-(a+b)}\,du=\Gamma(a)\Gamma(b)/\Gamma(a+b)$.
\end{lemma}

\begin{proof}\leanok
    Substitute $x=u/(1+u)$ in the Beta integral $\int_0^1x^{a-1}(1-x)^{b-1}\,dx$.
\end{proof}

\begin{lemma}[Monomial integrals]
    \label{lem:orthant_monomials}
    \lean{Partition.orthant, Partition.orthMono, Partition.integral_monomial,
        Partition.integrable_monomial, Partition.integral_monomial_mul_rpow_sum,
        Partition.integrable_monomial_mul_rpow_sum, Partition.integral_monomial_mul_rpow_sum'}
    \leanok
    \uses{lem:beta_half_line}
    Let $u_i>-1$ and $E=d+\sum_iu_i$. For $b>0$,
    $\int_{(0,\infty)^d}\prod_ix_i^{u_i}e^{-bx_i}\,dx=\prod_i\Gamma(1+u_i)b^{-1-u_i}$, and for
    $0\le a<E$,
    \[
        \int_{(0,\infty)^d}\prod_ix_i^{u_i}\Bigl(\sum_ix_i\Bigr)^{-a}e^{-\sum_ix_i}\,dx
        =\prod_i\Gamma(1+u_i)\,\frac{\Gamma(E-a)}{\Gamma(E)}.
    \]
\end{lemma}

\begin{proof}\leanok
    The first integral factorizes into one-dimensional gamma integrals. For the second,
    insert $\Gamma(a)r^{-a}=\int_0^\infty v^{a-1}e^{-rv}\,dv$ with $r=\sum_ix_i$, exchange the
    integrals, evaluate the inner one by the first formula with $b=1+v$, and apply the Beta
    integral.
\end{proof}

\begin{lemma}[A determinant of gamma values]
    \label{lem:gamma_determinant}
    \lean{Partition.risingPoly, Partition.Gamma_one_add_add_nat, Partition.revPerm,
        Partition.alternant_delta_eq, Partition.det_risingPoly, Partition.det_Gamma,
        Partition.alternant_mul_left}
    \leanok
    For $t\ge0$,
    $\det[\Gamma(1+\delta_i+tl_j)]_{i,j}=\prod_j\Gamma(1+tl_j)\cdot a_\delta(tl)$, and
    $a_\delta(tl)=t^Na_\delta(l)$.
\end{lemma}

\begin{proof}\leanok
    $\Gamma(1+tl+n)=\Gamma(1+tl)(1+tl)(2+tl)\cdots(n+tl)$, a monic polynomial of degree $n$ in
    $tl$. A determinant of monic polynomials of degrees $\delta_i$ evaluated at the points
    $tl_j$ equals the Vandermonde determinant with the same ordering of degrees, which is
    $a_\delta(tl)$.
\end{proof}

\begin{theorem}[The gamma integral]
    \label{thm:gamma_integral}
    \lean{Partition.gammaIntegral, Partition.alternant_mul_alternant_rpow,
        Partition.gammaIntegral_eq_sum, Partition.integrable_gammaIntegrand,
        Partition.sum_sign_prod_Gamma, Partition.gammaIntegral_eq,
        Partition.alternant_delta_mul_rpow_nonneg}
    \leanok
    \uses{lem:orthant_monomials, lem:gamma_determinant, lem:alternant_pieri}
    For $t\ge0$ and $0\le a<D$,
    \[
        \int_{(0,\infty)^d}a_\delta(x)a_l(x^t)\Bigl(\sum_ix_i\Bigr)^{-a}e^{-\sum_ix_i}\,dx
        =d!\,t^N\prod_j\Gamma(1+tl_j)\,a_\delta(l)\,\frac{\Gamma(D-a)}{\Gamma(D)},
    \]
    the integrand is integrable, and $a_\delta(x)a_\delta(x^t)\ge0$ on the orthant.
\end{theorem}

\begin{proof}\leanok
    Expand both alternants over permutations; each product is a monomial with exponents
    $\delta_{\sigma^{-1}i}+tl_{\tau^{-1}i}$, whose sum is $N+t\sum_jl_j$. The previous lemma
    integrates each term, and the double sum over $\sigma,\tau$ collapses to $d!$ copies of
    $\det[\Gamma(1+\delta_i+tl_j)]$. Finally
    $a_\delta(x)a_\delta(x^t)=\prod_{i<j}(x_j-x_i)(x_j^t-x_i^t)$.
\end{proof}

\begin{theorem}[The label function as a normalized gamma integral]
    \label{thm:replica_weight_gamma}
    \lean{Partition.alternant_delta_eq_vand, Partition.weylFormula_eq_alternant_div,
        Partition.replicaWeight_eq_gammaIntegral}
    \leanok
    \uses{thm:gamma_integral, lem:replica_weight_ratio}
    Let $J_a(l)$ denote the integral of the previous theorem and $W_\lambda$ the Weyl dimension
    formula. For $t>0$ and a partition $\lambda\vdash k$ with at most $d$ rows,
    $w_k(\lambda)=J_{tk}(l)/\bigl(J_0(\delta)W_\lambda\bigr)$.
\end{theorem}

\begin{proof}\leanok
    Here $D=c+tk$ and $D-tk=c$ for $l$, while $D=c$ for $\delta$. The factors $d!\,t^N$ cancel,
    and $a_\delta(l)/a_\delta(\delta)=W_\lambda$ because $a_\delta$ is a fixed sign times the
    Vandermonde product and $a_\delta(\delta)$ is the product of the differences $j-i$.
\end{proof}
