\chapter{Replica Entropy Metrics}\label{ch:replica_metrics}

The symmetric-replica argument of the two-dimensional area
law~\cite{OpenAI2026AreaLaw} compares the label observable of a subsystem with
a positive operator built from one label function shared by every subsystem.
Fix the dimension $d\ge1$ of the whole one-copy system and $0<t<1/4$. Pad each
label $\lambda\vdash k$ to length $d$, put $l_i=\lambda_i+d-i$ for
$1\le i\le d$, and set
\[
    c=d+(1+t)\binom d2,\qquad
    w_k(\lambda)=\frac{\Gamma(c)}{\Gamma(c+tk)}
        \prod_{i=1}^d\frac{\Gamma(1+tl_i)}{\Gamma(1+t(d-i))}.
\]
The metric of a subsystem $Q$ is $W_{Q,k}=\sum_\lambda w_k(\lambda)\pi^\lambda_{Q,k}$.
This chapter proves the scalar properties of $w_k$, and from them the operator
statements of the paper's Lemma~6.2 other than the integral representation of
$W_{Q,k}$.

\section{Ratios of Gamma values}

\begin{lemma}[Wendel's inequalities]
    \label{lem:wendel_gamma_ratio}
    \lean{Real.Gamma_add_le_rpow_mul_Gamma, Real.mul_Gamma_le_rpow_mul_Gamma_add,
        Real.Gamma_add_div_Gamma_le_rpow, Real.div_rpow_le_Gamma_add_div_Gamma,
        Real.rpow_sub_le_Gamma_add_div_Gamma}
    \leanok
    For $x>0$ and $0\le s\le1$,
    \[
        x(x+s)^{s-1}\le\frac{\Gamma(x+s)}{\Gamma(x)}\le x^s .
    \]
    If moreover $x\ge1$, then $x^s-s\le\Gamma(x+s)/\Gamma(x)$.
\end{lemma}

\begin{proof}\leanok
    Write $x+s=(1-s)x+s(x+1)$ and $x+1=s(x+s)+(1-s)(x+s+1)$, apply
    log-convexity of $\Gamma$ to each, and use $\Gamma(y+1)=y\Gamma(y)$. For
    the last claim, $(x+s)^{s-1}\ge x^s/(x+s)$, so the lower bound is at least
    $x^s-sx^s/(x+s)\ge x^s-s$ when $x\ge1$.
\end{proof}

\begin{lemma}[Stirling estimate for the Gamma function]
    \label{lem:log_gamma_stirling}
    \lean{Real.abs_log_Gamma_one_add_sub_le}
    \leanok
    \uses{lem:wendel_gamma_ratio, lem:stirling_log_factorial}
    For $u\ge0$,
    $|\log\Gamma(1+u)-(u\log u-u)|\le3\log(u+2)+3$.
\end{lemma}

\begin{proof}\leanok
    Let $n=\lfloor u\rfloor$ and $s=u-n$. Wendel's inequalities at $x=n+1$
    give $|\log\Gamma(1+u)-\log n!|\le\log(u+2)$. The Stirling bounds control
    $\log n!-(n\log n-n)$, and $u\log u-u$ differs from $n\log n-n$ by at most
    $\log(u+2)+2$.
\end{proof}

\section{The common label function}

\begin{lemma}[Positivity, normalization and marked ratio]
    \label{lem:replica_weight_ratio}
    \lean{Partition.replicaConst, Partition.replicaWeight, Partition.replicaRatio,
        Partition.replicaWeight_pos, Partition.replicaWeight_zero,
        Partition.replicaWeight_update_succ}
    \leanok
    The label function is positive, and $w_0=1$. If $\lambda\vdash k$ is
    obtained from $\nu\vdash k-1$ by adding a box in row $i$, then
    $w_k(\lambda)=r_{\lambda,i}\,w_{k-1}(\nu)$ with
    \[
        r_{\lambda,i}=\frac{\Gamma(c+t(k-1))}{\Gamma(c+tk)}
            \frac{\Gamma(1+tl_i)}{\Gamma(1+t(l_i-1))},
    \]
    where $l_i$ is the shifted part of $\lambda$.
\end{lemma}

\begin{proof}\leanok
    Every argument of $\Gamma$ is positive. Adding the box raises the number of
    copies by one and the $i$-th shifted part by one, leaving the other shifted
    parts unchanged.
\end{proof}

\begin{lemma}[Entropy asymptotics of the label function]
    \label{lem:replica_weight_entropy}
    \lean{Partition.exists_abs_log_replicaWeight_add_le}
    \leanok
    \uses{lem:log_gamma_stirling, lem:replica_weight_ratio}
    There is a constant $C_{d,t}$ such that every $\lambda\vdash k$ of length at
    most $d$ satisfies
    $|\log w_k(\lambda)+tkH(\lambda/k)|\le C_{d,t}\log(k+2)$.
\end{lemma}

\begin{proof}\leanok
    Apply the Stirling estimate to each of the $d+2$ Gamma factors. The terms
    containing $\log t$ cancel up to a constant because
    $\sum_il_i=k+\binom d2$. The shifts $0\le l_i-\lambda_i\le d-1$ change
    $\sum_il_i\log l_i$ by $O_d(\log(k+2))$, and
    $kH(\lambda/k)=k\log k-\sum_i\lambda_i\log\lambda_i$.
\end{proof}

Together with the entropy asymptotics of the dimension formula, this gives
$w_k(\lambda)=d_\lambda^{-t}$ up to factors polynomial in $k$, which is the
label-wise form of the comparison of $W_{Q,k}$ with $e^{-tF_{Q,k}}$.

\section{The marked ratio}

\begin{lemma}[Size of the marked ratio]
    \label{lem:replica_ratio_bounds}
    \lean{Partition.replicaRatio_eq_div, Partition.replicaRatio_wendel,
        Partition.exists_replicaRatio_bounds}
    \leanok
    \uses{lem:wendel_gamma_ratio, lem:replica_weight_ratio}
    Let $0<t<1$. There is $C\ge1$ such that, for $k\ge1$ and $1\le l\le k+d$,
    the ratio $r(k,l)$ of the previous lemma satisfies
    $C^{-1}(l/k)^t\le r(k,l)\le C$.
\end{lemma}

\begin{proof}\leanok
    The ratio is a quotient of two Wendel ratios, with $x=1+t(l-1)$ and
    $y=c+t(k-1)$, both at least one. Wendel's inequalities give
    $(1-t)(x/y)^t\le r(k,l)\le(x/y)^t/(1-t)$. On the stated range,
    $tl\le x\le(1+d)y$ and $y\le(c+t)k$.
\end{proof}

\begin{lemma}[Limit of the marked ratio]
    \label{lem:replica_ratio_limit}
    \lean{Real.abs_rpow_sub_rpow_le, Partition.exists_abs_replicaRatio_sub_le}
    \leanok
    \uses{lem:replica_ratio_bounds}
    Let $0<t<1$. There is a constant $C$ such that, for $k\ge1$ and
    $1\le l\le k+d$,
    \[
        \bigl|r(k,l)-\max\bigl((l-d)/k,0\bigr)^t\bigr|\le Ck^{-t}.
    \]
\end{lemma}

\begin{proof}\leanok
    Wendel's inequalities give $|r(k,l)-(x/y)^t|\le Ck^{-t}$, since $y\ge tk$
    and $(x/y)^t\le1+d$. Moreover $|x/y-l/k|=O(1/k)$ and
    $|l/k-\max((l-d)/k,0)|\le d/k$, and $|a^t-b^t|\le|a-b|^t$ for $a,b\ge0$.
\end{proof}

On a removable branch the star operator has eigenvalue $(\lambda_i-i)/k=(l_i-d)/k$,
so this lemma is the scalar content of the convergence of the marked ratio of
the metrics to the positive part of the star operator raised to the power $t$.

\section{Functional calculus on an orthogonal resolution}

\begin{lemma}[Functions of an orthogonal resolution]
    \label{lem:orthogonal_resolution_cfc}
    \lean{Matrix.IsOrthogonalResolution.aeval_hom,
        Matrix.IsOrthogonalResolution.spectrum_hom_subset,
        Matrix.IsOrthogonalResolution.cfc_hom,
        Matrix.IsOrthogonalResolution.l2_opNorm_hom_le,
        Matrix.IsOrthogonalResolution.re_star_dotProduct_hom_mulVec_self,
        Matrix.IsOrthogonalResolution.sum_weight}
    \leanok
    \uses{lem:orthogonal_resolution}
    Let $P_i$ be Hermitian idempotents with $P_iP_j=\delta_{ij}P_i$ and
    $\sum_iP_i=I$, and let $g$ be real. Then $f(\sum_ig_iP_i)=\sum_if(g_i)P_i$ for
    every real function $f$. If $|c_i|\le M$ whenever $P_i\ne0$, then
    $\|\sum_ic_iP_i\|\le M$ in operator norm. For every vector $z$,
    $\|\sum_ic_iP_iz\|^2=\sum_i|c_i|^2\|P_iz\|^2$ and $\sum_i\|P_iz\|^2=\|z\|^2$.
\end{lemma}

\begin{proof}\leanok
    Polynomials act pointwise, and the spectrum lies in the range of $g$, so
    interpolating $f$ on that finite set by a polynomial proves the first claim.
    For the second, $A^\ast A=\sum_i|c_i|^2P_i\le M^2I$ and
    $\|A\|^2=\|A^\ast A\|$.
\end{proof}

\section{The metrics}

Let $\mathcal V=\bigotimes_f\C^{\iota_f}$ have dimension $d$, and let a subsystem
$Q$ be a set of factors. A label occurring on $Q^{\otimes k}$ has at most $d_Q\le d$
rows.

\begin{lemma}[Labels of a subsystem]
    \label{lem:subsystem_label_transport}
    \lean{PermutationRepresentation.labelProj_of_intertwine,
        PermutationRepresentation.labelProj_prodLeft,
        TensorPower.labelProj_subsystemPerm_eq,
        TensorPower.labelPart_eq_zero_of_subsystemPerm,
        TensorPower.card_subConfig_le}
    \leanok
    Under $\mathcal V^{\otimes k}\cong Q^{\otimes k}\otimes(Q^c)^{\otimes k}$, the
    label projector $\pi^\lambda_{Q,k}$ is the label projector of $Q^{\otimes k}$
    tensored with the identity. Consequently a label with $\pi^\lambda_{Q,k}\ne0$
    has at most $d_Q$ rows, and $d_Q\le d$.
\end{lemma}

\begin{proof}\leanok
    Intertwined permutation actions have the same group-algebra image up to
    relabelling the basis, and an action on the first factor of a product gives
    the image tensored with the identity. The row bound is the corresponding
    statement for $(\C^{d_Q})^{\otimes k}$.
\end{proof}

\begin{theorem}[Positivity, centrality and complements]
    \label{thm:replica_metric_basic}
    \lean{TensorPower.replicaMetric, TensorPower.posDef_replicaMetric,
        TensorPower.replicaMetric_zero, TensorPower.commute_replicaMetric_permOp,
        TensorPower.commute_replicaMetric_of_subset,
        TensorPower.commute_replicaMetric_of_disjoint,
        TensorPower.replicaMetric_mul_symProj_compl}
    \leanok
    \uses{lem:replica_weight_ratio, thm:schur_label_commutation}
    Let $t\ge0$ and assume every factor is nonzero. The metric
    $W_{Q,k}=\sum_\lambda w_k(\lambda)\pi^\lambda_{Q,k}$ is positive definite,
    commutes with the copy permutations of $Q$ and with the metrics of nested and
    disjoint subsystems, and $W_{Q,0}=I$. On the symmetric subspace,
    $W_{Q,k}=W_{Q^c,k}$.
\end{theorem}

\begin{proof}\leanok
    The weights are positive and the label projectors form an orthogonal
    resolution of the identity. The other claims hold for every central label
    function, by the commutation and complement properties of the label
    projectors.
\end{proof}

\begin{theorem}[Comparison with the label observable]
    \label{thm:replica_metric_comparison}
    \lean{TensorPower.exists_abs_log_replicaLabelWeight_add_le,
        TensorPower.cfc_exp_labelEntropy, TensorPower.exists_replicaMetric_comparison}
    \leanok
    \uses{lem:replica_weight_entropy, lem:hook_formula_entropy,
        lem:subsystem_label_transport, lem:orthogonal_resolution_cfc}
    Let $t>0$. There is $C$, depending only on $d$ and $t$, such that for every
    subsystem $Q$ and every $k$,
    \[
        (k+2)^{-C}e^{-tF_{Q,k}}\le W_{Q,k}\le(k+2)^Ce^{-tF_{Q,k}}.
    \]
\end{theorem}

\begin{proof}\leanok
    Both sides are functions of the label projectors, so it suffices to compare
    the coefficients of the labels that occur. Such a label has at most $d$ rows,
    $d_\lambda$ equals the dimension formula, and the entropy asymptotics of the
    dimension formula and of the label function give
    $|\log w_k(\lambda)+t\log d_\lambda|\le C\log(k+2)$.
\end{proof}

\section{The marked ratio of the metrics}

Let $k=m+1$. Extend $W_{Q,k-1}$ by the identity on the last copy, put
$R_{Q,k}=W_{Q,k}W_{Q,k-1}^{-1}$, and let $J_{Q,k}$ be the normalized star
operator of $Q$.

\begin{lemma}[Branch decomposition of the marked ratio]
    \label{lem:marked_ratio_branches}
    \lean{TensorPower.snocEquiv, TensorPower.replicaMetricFirst,
        TensorPower.replicaMetricFirst_eq, TensorPower.markedRatio,
        TensorPower.starOp_subsystemPerm_eq_hom, TensorPower.markedRatio_eq_hom,
        TensorPower.exists_branch_scalars}
    \leanok
    \uses{thm:replica_metric_basic, lem:replica_weight_ratio}
    In the joint resolution by the label $\lambda$ of all copies and the label
    $\nu$ of the first $k-1$ copies, $R_{Q,k}$ and $J_{Q,k}$ are diagonal. On a
    nonzero block, $\nu=\lambda-e_i$, the ratio acts by $r(k,l_i)$ and the star
    operator by $(l_i-d)/k$, where $1\le l_i\le k+d$ is the shifted part of
    $\lambda$ padded to length $d$.
\end{lemma}

\begin{proof}\leanok
    The extended metric is the central label function of the copy permutations
    fixing the last copy. On a nonzero block the labels are compatible, so
    $\nu=\lambda-e_i$; the star eigenvalue is the one of the Schur-label
    estimates, and the ratio of label functions is the exact marked ratio.
\end{proof}

\begin{theorem}[Forward marked ratio]
    \label{thm:marked_ratio_forward}
    \lean{TensorPower.exists_l2_opNorm_markedRatio_sub_le,
        TensorPower.exists_l2_opNorm_markedRatio_le}
    \leanok
    \uses{lem:marked_ratio_branches, lem:replica_ratio_limit, lem:replica_ratio_bounds,
        lem:orthogonal_resolution_cfc}
    Let $0<t<1$. There are constants such that, on the entire tensor space and
    for every subsystem,
    $\|R_{Q,k}-(J_{Q,k})_+^t\|\le Ck^{-t}$ and $\|R_{Q,k}\|\le C$. In particular
    $\|R_{Q,k}-(J_{Q,k})_+^t\|\to0$.
\end{theorem}

\begin{proof}\leanok
    Both operators are diagonal in the branch decomposition; apply the limit and
    the bounds of the marked ratio blockwise and the operator-norm bound on an
    orthogonal resolution.
\end{proof}

\begin{lemma}[Branch weights of a symmetric vector]
    \label{lem:symmetric_branch_weights}
    \lean{TensorPower.complAverage, TensorPower.trace_complAverage_mul,
        TensorPower.commute_complAverage, TensorPower.weight_branch,
        TensorPower.card_branch_le, TensorPower.exists_branch_data}
    \leanok
    \uses{lem:marked_ratio_branches, lem:subsystem_label_transport}
    Let $z\in\mathcal S_k$. On a branch $\nu=\lambda-e_i$,
    $d_\lambda\|\pi^\lambda\pi^\nu z\|^2=d_\nu\|\pi^\lambda z\|^2$; hence
    $\|\pi^\lambda\pi^\nu z\|^2\le2^d(l_i/k)\|\pi^\lambda z\|^2$. Each label has at
    most $d$ branches with nonzero projector.
\end{lemma}

\begin{proof}\leanok
    The average of $|z\rangle\langle z|$ over the copy permutations of $Q^c$ commutes
    with the copy permutations of $Q$, because $z$ is symmetric, and pairs with every
    operator generated by them as $k!\,|z\rangle\langle z|$; this is the matrix form
    of the permutation invariance of the $Q$ marginal. The branch-probability
    identity of the Schur-label estimates applies to it. Distinct branches of one
    label differ in their removed row.
\end{proof}

\begin{theorem}[Inverse marked ratio on symmetric vectors]
    \label{thm:marked_ratio_inverse}
    \lean{TensorPower.markedRatio_inv_eq_hom,
        TensorPower.exists_norm_markedRatio_inv_mulVec_le,
        TensorPower.exists_markedRatio_inv_sub_clipped_sq_le,
        TensorPower.exists_norm_markedRatio_inv_sub_clipped_le}
    \leanok
    \uses{lem:symmetric_branch_weights, lem:replica_ratio_bounds, lem:replica_ratio_limit}
    Let $0<t<1/2$. There is $C$ such that $\|R_{Q,k}^{-1}z\|\le C\|z\|$ for every
    $z\in\mathcal S_k$, and, for every $0<\delta<1$,
    \[
        \limsup_{k\to\infty}\sup_{z\in\mathcal S_k,\ \|z\|=1}
        \bigl\|\bigl(R_{Q,k}^{-1}-\max(J_{Q,k},\delta)^{-t}\bigr)z\bigr\|
        \le C\delta^{(1-2t)/2}.
    \]
\end{theorem}

\begin{proof}\leanok
    Both operators are diagonal in the branch decomposition. On a branch,
    $r^{-2}\|\pi^\lambda\pi^\nu z\|^2\le C(l_i/k)^{1-2t}\|\pi^\lambda z\|^2$, and summing
    over at most $d$ branches and over labels gives the first bound. For the second,
    on $J\ge\delta$ the clipped inverse differs from $r^{-1}$ by $O(k^{-t}\delta^{-2t})$.
    On $J<\delta$, $l_i/k<\delta+d/k$, and both inverses contribute at most
    $C\bigl((\delta+d/k)^{1-2t}+\delta^{-2t}(\delta+d/k)\bigr)\|\pi^\lambda z\|^2$.
    Let $k\to\infty$ and take square roots.
\end{proof}

\section{Similarity transforms of copy means}

Let $\mathcal V=\bigotimes_v\C^{n_v}$ with every $n_v\ge1$, let $P,Y,F$ be disjoint
sets of factors, and let $A_k=(W_{P,k}^{-1}W_{F,k}^{-1}W_{Y,k})^2$ be the partition
metric. For a one-copy operator $h$ supported on $P\cup Y$, write $h^{(j)}$ for $h$
acting on the $j$-th copy, $\bar h=k^{-1}\sum_jh^{(j)}$, and
$O_k=A_k^{-1/2}\bar hA_k^{1/2}$.

\begin{lemma}[One-copy operators on replicas]
    \label{lem:replica_site_operators}
    \lean{PermutationRepresentation.commute_permOp_of_apply_eq,
        TensorPower.siteOp_apply_of_isSupportedOn,
        TensorPower.commute_permOp_subsystemPerm_siteOp_of_disjoint,
        TensorPower.commute_permOp_subsystemPerm_siteOp_of_fix,
        TensorPower.permOp_copyPerm_mul_siteOp, TensorPower.siteOp_eq_reindex,
        TensorPower.l2_opNorm_siteOp_le, Entropy.IsSupportedOn.exists_sum_mul}
    \leanok
    The operator $h^{(j)}$ commutes with the copy permutations of every subsystem
    disjoint from the support of $h$, and with every copy permutation of a subsystem
    that fixes the copy $j$. A permutation $\tau$ of entire copies conjugates $h^{(j)}$
    to $h^{(\tau j)}$, and $\|h^{(j)}\|\le\|h\|$. An operator supported on a disjoint
    union $X\cup T$ is a finite sum of products $a_ib_i$ with $a_i$ supported on $X$
    and $b_i$ supported on $T$.
\end{lemma}

\begin{proof}\leanok
    The first two statements compare matrix entries: $h^{(j)}$ is nonzero only
    between configurations that agree outside the support factors of copy $j$, and
    the permutations in question preserve this relation and these factors. The
    operator $h^{(j)}$ is $h\otimes I$ after splitting off copy $j$. For the
    decomposition, expand a local operator on $X\cup T$ in matrix units.
\end{proof}

\begin{theorem}[Uniform boundedness of the similarity transform]
    \label{thm:marked_similarity_bounded}
    \lean{TensorPower.leafRoot, TensorPower.leafMetric, TensorPower.copyMean,
        TensorPower.markedSimilarity, TensorPower.sqrt_leafMetric,
        TensorPower.star_dotProduct_copyMean_eq, TensorPower.replicaMetric_conj_siteOp_mul,
        TensorPower.exists_norm_markedSimilarity_mulVec_le}
    \leanok
    \uses{lem:replica_site_operators, thm:marked_ratio_forward, thm:marked_ratio_inverse,
        thm:replica_metric_basic}
    Let $0<t<1/2$. For fixed $P,Y,F$ and $h$ there is $C$, depending on $\mathcal V$, $t$,
    the partition and $h$, such that
    $\|O_kz\|\le C\|z\|$ for every $k$ and every $z\in\mathcal S_k$.
\end{theorem}

\begin{proof}\leanok
    The square root of $A_k$ is $W_{P,k}^{-1}W_{F,k}^{-1}W_{Y,k}$, the $F$ factors
    commute with $\bar h$ and cancel, and $O_k$ preserves $\mathcal S_k$. Between
    symmetric vectors, $h^{(k)}$ may replace $\bar h$. Write $h=\sum_ib_ia_i$ with $b_i$
    supported on $Y$ and $a_i$ on $P$. Since the metric of the first $k-1$ copies
    commutes with operators on the last copy,
    \[
        W_Y^{-1}W_P(b_ia_i)^{(k)}W_P^{-1}W_Y
        =\bigl(R_Y^{-1}b_i^{(k)}R_Y\bigr)\bigl(R_Pa_i^{(k)}R_P^{-1}\bigr).
    \]
    Hence $\langle w,O_kz\rangle=\sum_i\langle R_Yb_i^{(k)\ast}R_Y^{-1}w,
    R_Pa_i^{(k)}R_P^{-1}z\rangle$ for symmetric $w,z$, in which both inverses act on
    symmetric vectors. The forward and inverse bounds on the marked ratio bound each
    term by $C_i\|w\|\|z\|$; take $w=O_kz$.
\end{proof}
