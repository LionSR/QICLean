\section{Order of tensor products}

For complex matrices, $A\le B$ means that $B-A$ is positive semidefinite.
The index sets in this section are finite and may be empty.
These elementary comparisons give the tensor-order step in the sharp upper
comparison of \cite[Section~8]{OpenAI2026AreaLaw}.

\begin{theorem}[A two-factor order comparison]
    \label{thm:kronecker_order_two_factors}
    \lean{Matrix.kronecker_mono_of_posSemidef}
    \leanok
    Let $A,B$ be complex matrices indexed by $I$, and let $C,D$ be complex
    matrices indexed by $J$. Suppose that $A\le B$, $C\le D$, $B\ge0$, and
    $C\ge0$. Then $A\otimes C\le B\otimes D$.
    The matrix $A$ need not be positive semidefinite.
\end{theorem}

\begin{proof}
    \leanok
    Since $B$ and $C$ are positive semidefinite, the linear maps
    $X\mapsto X\otimes C$ and $Y\mapsto B\otimes Y$ are positive and hence
    preserve order. Consequently,
    \begin{align}
        A\otimes C\le B\otimes C\le B\otimes D.
    \end{align}
\end{proof}

\begin{theorem}[Order of finite tensor products]
    \label{thm:finite_kronecker_order}
    \lean{Matrix.finKronecker_mono}
    \leanok
    Let $N\ge0$. For each $0\le i<N$, let $I_i$ be a finite set and let
    $A_i,B_i$ be complex matrices indexed by $I_i$.
    If $0\le A_i\le B_i$ for every $i$, then
    \begin{align}
        \bigotimes_{i=0}^{N-1}A_i\le\bigotimes_{i=0}^{N-1}B_i.
    \end{align}
    The matrices act on $\bigotimes_{i=0}^{N-1}\mathbb C^{I_i}$.
    When $N=0$, both products are the identity on the one-dimensional
    empty tensor product.
\end{theorem}

\begin{proof}
    \leanok
    \uses{thm:kronecker_order_two_factors}
    Induct on $N$. The empty product gives equality. For the induction
    step, write $A'$ and $B'$ for the products of all factors after the
    first. The induction hypothesis gives $A'\le B'$, and positivity of
    tensor products gives $A'\ge0$. Since $B_0\ge A_0\ge0$,
    Theorem~\ref{thm:kronecker_order_two_factors} gives
    $A_0\otimes A'\le B_0\otimes B'$. Separating the first coordinate
    identifies these two matrices with the original tensor products.
\end{proof}
